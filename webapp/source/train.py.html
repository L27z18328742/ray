   9 def train(args):
  10     configure_logger()
  11     release_train = args.release_train
  12 
  13     # allocate the GPUs
  14     pgs = create_placement_groups(args)
  15     init_tracking(args)
  16 
  17     # create the rollout manager, with sglang engines inside.
  18     # need to initialize rollout manager first to calculate num_rollout
  19     rollout_manager, num_rollout_per_epoch = create_rollout_manager(args, pgs["rollout"])
  20 
  21     actor_model, critic_model = create_training_models(args, pgs, rollout_manager)
  22 
  23     if args.offload_rollout and not release_train:
  24         ray.get(rollout_manager.onload_weights.remote())
  25 
  26     # Always push actor weights to rollout once weights are loaded.
  27     actor_model.update_weights()
  28 
  29     if args.check_weight_update_equal:
  30         ray.get(rollout_manager.check_weights.remote(action="compare"))
  31 
  32     if args.offload_rollout:
  33         ray.get(rollout_manager.onload_kv.remote())
  34 
  35     # special case for eval-only
  36     if args.num_rollout == 0 and args.eval_interval is not None:
  37         ray.get(rollout_manager.eval.remote(rollout_id=0))
  38 
  39     def offload_train(actor_trains_this_step):
  40         # Each model auto-offloads after train() when offload_train is set,
  41         # so we only need clear_memory for the non-offload case.
  42         if not args.offload_train:
  43             if not args.use_critic or actor_trains_this_step:
  44                 actor_model.clear_memory()
  45             else:
  46                 critic_model.clear_memory()
  47 
  48     # train loop.
  49     for rollout_id in range(args.start_rollout_id, args.num_rollout):
  50         if args.eval_interval is not None and rollout_id == 0 and not args.skip_eval_before_train:
  51             ray.get(rollout_manager.eval.remote(rollout_id))
  52 
  53         rollout_data_ref = ray.get(rollout_manager.generate.remote(rollout_id))
  54 
  55         if args.offload_rollout:
  56             ray.get(rollout_manager.offload.remote())
  57 
  58         if release_train:
  59             actor_model.create()
  60 
  61         actor_trains = (not args.use_critic) or rollout_id >= args.num_critic_only_steps
  62         if args.use_critic:
  63             value_refs = critic_model.async_train(rollout_id, rollout_data_ref)
  64             if actor_trains:
  65                 ray.get(actor_model.async_train(rollout_id, rollout_data_ref, external_data=value_refs))
  66             else:
  67                 ray.get(value_refs)
  68         else:
  69             ray.get(actor_model.async_train(rollout_id, rollout_data_ref))
  70 
  71         if release_train or should_run_periodic_action(
  72             rollout_id, args.save_interval, num_rollout_per_epoch, args.num_rollout
  73         ):
  74             force_sync = release_train or rollout_id == args.num_rollout - 1
  75             if actor_trains:
  76                 actor_model.save_model(rollout_id, force_sync=force_sync)
  77             if args.use_critic:
  78                 critic_model.save_model(rollout_id, force_sync=force_sync)
  79             if args.rollout_global_dataset:
  80                 ray.get(rollout_manager.save.remote(rollout_id))
  81 
  82         ray.get(rollout_manager.cleanup_rollout_data.remote(rollout_id))
  83 
  84         offload_train(actor_trains)
  85         if args.offload_rollout and not release_train:
  86             ray.get(rollout_manager.onload_weights.remote())
  87         actor_model.update_weights()
  88 
  89         if args.offload_rollout:
  90             ray.get(rollout_manager.onload_kv.remote())
  91 
  92         if should_run_periodic_action(rollout_id, args.eval_interval, num_rollout_per_epoch):
  93             ray.get(rollout_manager.eval.remote(rollout_id))
  94 
  95     ray.get(rollout_manager.dispose.remote())
  96     finish_tracking(args)
  97 
  98 
  99 if __name__ == "__main__":
 100     args = parse_args()
 101     train(args)
