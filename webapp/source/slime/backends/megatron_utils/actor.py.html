 174     @timer
 175     def sleep(self) -> None:
 176         assert self.args.offload_train
 177 
 178         clear_memory(clear_host_memory=True)
 179         print_memory("before offload model")
 180         if (
 181             self.role == "actor"
 182             and self.args.use_critic
 183             and not self.args.colocate
 184             and hasattr(self.weight_updater, "disconnect_rollout_engines")
 185         ):
 186             self.weight_updater.disconnect_rollout_engines()
 187         destroy_process_groups()
 188 
 189         torch_memory_saver.pause()
 190         reset_cuda_stack_size()
 191 
 192         print_memory("after offload model")

 194     @timer
 195     def wake_up(self) -> None:
 196         assert self.args.offload_train
 197         print_memory("before wake_up model")
 198 
 199         torch_memory_saver.resume()
 200 
 201         clear_memory()
 202         reload_process_groups()
 203 
 204         if mpu.get_pipeline_model_parallel_world_size() > 2:
 205             # Megatron's patched batched pipeline P2P uses the default WORLD
 206             # group.  After reload, PP=4 starts with only the first two stages
 207             # entering batch_isend_irecv(), but PyTorch requires every rank when
 208             # that is the first NCCL operation on a group.  Prime WORLD here,
 209             # after the memory saver is resumed, so later stages cannot miss its
 210             # lazy initialization.  Sleep still destroys it completely.
 211             dist.barrier(device_ids=[accelerator.current_device()])
 212         if self.role == "actor":
 213             self._switch_model("actor")
 214         print_memory("after wake_up model")

 216     def _get_rollout_data(self, rollout_data_ref: Box) -> RolloutBatch:
 217         # Fetch data through ray on CPU, not sure if this will be performance bottleneck.
 218         # Both first pp stage and the last pp stage will receive the data.
 219         rollout_data = process_rollout_data(
 220             rollout_data_ref,
 221             mpu.get_data_parallel_rank(with_context_parallel=False),
 222             mpu.get_data_parallel_world_size(with_context_parallel=False),
 223         )
 224         # TODO: this is ugly, move to somewhere else?
 225         # move tokens to GPU in advance
 226         device = accelerator.current_device()
 227         rollout_data["tokens"] = [
 228             t.to(device=device, dtype=torch.long, non_blocking=True) for t in rollout_data["tokens"]
 229         ]
 230         rollout_data["loss_masks"] = [
 231             t.to(device=device, dtype=torch.int, non_blocking=True) for t in rollout_data["loss_masks"]
 232         ]
 233         if "rollout_mask_sums" in rollout_data:
 234             # Promote precomputed per-rollout mask totals to GPU tensors here
 235             # (matching loss_masks) so the loss reducer can just divide.
 236             rollout_data["rollout_mask_sums"] = rollout_data["rollout_mask_sums"].to(
 237                 device=device, dtype=torch.float32, non_blocking=True
 238             )
 239         if "multimodal_train_inputs" in rollout_data:
 240             # Move multimodal training tensors to GPU in advance
 241             rollout_data["multimodal_train_inputs"] = [
 242                 (
 243                     {
 244                         key: value.to(device=device, non_blocking=True) if isinstance(value, torch.Tensor) else value
 245                         for key, value in mm_dict.items()
 246                     }
 247                     if mm_dict is not None
 248                     else None
 249                 )
 250                 for mm_dict in rollout_data["multimodal_train_inputs"]
 251             ]
 252 
 253         for key in ["rollout_log_probs", "teacher_log_probs"]:
 254             if key not in rollout_data:
 255                 continue
 256             rollout_data[key] = [
 257                 slice_log_prob_with_cp(log_prob, total_length, response_length).to(
 258                     device=device,
 259                     dtype=torch.float32,
 260                     non_blocking=True,
 261                 )
 262                 for log_prob, total_length, response_length in zip(
 263                     rollout_data[key],
 264                     rollout_data["total_lengths"],
 265                     rollout_data["response_lengths"],
 266                     strict=False,
 267                 )
 268             ]
 269         return rollout_data

 277     def fill_routing_replay(self, data_iterator, num_microbatches, rollout_data):
 278         if "rollout_routed_experts" not in rollout_data:
 279             raise ValueError(
 280                 "rollout_routed_experts is required in rollout_data when use_rollout_routing_replay is set."
 281             )
 282 
 283         from megatron.core.transformer.transformer_block import get_num_layers_to_build
 284         from megatron.core.transformer.transformer_layer import get_transformer_layer_offset
 285 
 286         from slime.utils.routing_replay import RoutingReplay
 287 
 288         layer_ids = []
 289         for vp_stage, model in enumerate(self.model):
 290             config = model.module.config
 291             num_layers_to_build = get_num_layers_to_build(config, vp_stage=vp_stage)
 292             offset = get_transformer_layer_offset(config, vp_stage=vp_stage)
 293             for layer_id in range(offset, offset + num_layers_to_build):
 294                 if isinstance(config.moe_layer_freq, int):
 295                     if layer_id % config.moe_layer_freq != 0:
 296                         continue
 297                 elif isinstance(config.moe_layer_freq, list):
 298                     assert len(config.moe_layer_freq) == config.num_layers
 299                     if config.moe_layer_freq[layer_id] == 0:
 300                         continue
 301                 layer_ids.append(layer_id)
 302         assert len(layer_ids) == len(RoutingReplay.all_routing_replays)
 303 
 304         for iterator in data_iterator:
 305             iterator.reset()
 306 
 307         replay_source = rollout_data["rollout_routed_experts"]
 308         disk_prefetcher = None
 309         prepare_kwargs = {
 310             "num_experts": self.args.num_experts,
 311             "data_pad_size_multiplier": self.args.data_pad_size_multiplier,
 312             "sequence_parallel": self.args.sequence_parallel,
 313             "allgather_cp": self.args.allgather_cp,
 314         }
 315         for _ in range(sum(num_microbatches)):
 316             iterator = data_iterator[0]
 317             batch_indices = iterator.micro_batch_indices[iterator.offset]
 318             batch = iterator.get_next(["rollout_routed_experts", "tokens"])
 319             values = batch["rollout_routed_experts"]
 320 
 321             disk_backed = [isinstance(value, DiskTensorRef) for value in values]
 322             if any(disk_backed) and not all(disk_backed):
 323                 raise ValueError("A routing replay microbatch cannot mix disk-backed and resident route tensors.")
 324 
 325             if layer_ids and all(disk_backed):
 326                 if disk_prefetcher is None:
 327                     disk_prefetcher = RoutedExpertsMicrobatchPrefetcher(self.args.routing_replay_prefetch_microbatches)
 328                 source = RoutedExpertsMicrobatch(
 329                     values,
 330                     batch["tokens"],
 331                     consumer_count=len(layer_ids),
 332                     prepare_kwargs=prepare_kwargs,
 333                 )
 334                 disk_prefetcher.add(source)
 335                 for replay, layer_id in zip(RoutingReplay.all_routing_replays, layer_ids, strict=True):
 336                     replay.record(RoutedExpertsLayerRef(source, layer_id))
 337             elif layer_ids:
 338                 rollout_routed_experts = prepare_routed_experts_for_routing_replay(
 339                     values,
 340                     batch["tokens"],
 341                     **prepare_kwargs,
 342                 )
 343                 for replay, layer_id in zip(RoutingReplay.all_routing_replays, layer_ids, strict=True):
 344                     replay.record(rollout_routed_experts[:, layer_id])
 345 
 346             # Drop manager-owned references as soon as this microbatch has
 347             # been registered, bounding actor RSS during setup.
 348             for sample_idx in batch_indices:
 349                 replay_source[sample_idx] = None
 350 
 351         if disk_prefetcher is not None:
 352             disk_prefetcher.start()
 353             RoutingReplay.register_lazy_resource(disk_prefetcher)
 354             rss_gib, hwm_gib = get_process_host_memory_gib()
 355             logger.info(
 356                 "R3 lazy replay initialized: microbatches=%d layers=%d prefetch=%d rss=%.3f GiB hwm=%.3f GiB",
 357                 len(disk_prefetcher.sources),
 358                 len(layer_ids),
 359                 disk_prefetcher.prefetch_microbatches,
 360                 rss_gib,
 361                 hwm_gib,
 362             )
 363 
 364         del rollout_data["rollout_routed_experts"]
 365 
 366         for iterator in data_iterator:
 367             iterator.reset()

 386     def train(self, rollout_id: int, rollout_data_ref: Box, external_data=None):
 387         if self.args.debug_rollout_only:
 388             return None
 389 
 390         if self.args.offload_train:
 391             self.wake_up()
 392 
 393         with timer("data_preprocess"):
 394             rollout_data = self._get_rollout_data(rollout_data_ref)
 395 
 396         if self.role == "critic":
 397             result = self.train_critic(rollout_id, rollout_data)
 398         else:
 399             self.train_actor(rollout_id, rollout_data, external_data=external_data)
 400             result = None
 401 
 402         if self.args.offload_train:
 403             del rollout_data
 404             self.sleep()
 405 
 406         return result

 408     def train_critic(self, rollout_id: int, rollout_data: RolloutBatch):
 409         """Train critic and return CPU values (used as old-values for the next actor train)."""
 410         data_iterator = get_data_iterator(rollout_data)
 411         num_microbatches = rollout_data["num_microbatches"]
 412         global_batch_sizes = rollout_data["global_batch_sizes"]
 413 
 414         # Compute current critic values (used as old_values for value loss and for actor advantages).
 415         rollout_data.update(forward_only(get_values, self.args, self.model, data_iterator, num_microbatches))
 416 
 417         compute_advantages_and_returns(self.args, rollout_data)
 418 
 419         self.args.loss_type = "value_loss"
 420         train(
 421             rollout_id,
 422             self.model,
 423             self.optimizer,
 424             self.opt_param_scheduler,
 425             data_iterator,
 426             num_microbatches,
 427             global_batch_sizes,
 428         )
 429 
 430         if mpu.is_pipeline_last_stage() and "values" in rollout_data:
 431             from slime.backends.megatron_utils.data import tensors_to_cpu
 432 
 433             return {"values": tensors_to_cpu(rollout_data["values"])}
 434         return {}

 436     def train_actor(self, rollout_id: int, rollout_data: RolloutBatch, external_data=None) -> None:
 437         # Create data iterator for log_probs and train.
 438         data_iterator = get_data_iterator(rollout_data)
 439         num_microbatches = rollout_data["num_microbatches"]
 440         global_batch_sizes = rollout_data["global_batch_sizes"]
 441 
 442         if self.args.use_rollout_routing_replay:
 443             self.fill_routing_replay(data_iterator, num_microbatches, rollout_data)
 444 
 445         with inverse_timer("train_wait"), timer("train"):
 446             if self.args.compute_advantages_and_returns:
 447                 if "ref" in self.weights_backuper.backup_tags:
 448                     if self.args.use_routing_replay:
 449                         os.environ["ROUTING_REPLAY_STAGE"] = "fallthrough"
 450                     self._switch_model("ref")
 451                     rollout_data.update(
 452                         self.compute_log_prob(
 453                             data_iterator,
 454                             num_microbatches,
 455                             store_prefix="ref_",
 456                         )
 457                     )
 458 
 459                 # Forward teacher model to get teacher_log_probs for Megatron-based OPD
 460                 if "teacher" in self.weights_backuper.backup_tags:
 461                     if self.args.use_routing_replay:
 462                         os.environ["ROUTING_REPLAY_STAGE"] = "fallthrough"
 463                     self._switch_model("teacher")
 464                     rollout_data.update(
 465                         self.compute_log_prob(
 466                             data_iterator,
 467                             num_microbatches,
 468                             store_prefix="teacher_",
 469                         )
 470                     )
 471 
 472                 self._switch_model("actor")
 473                 can_reuse_log_probs_in_loss = (
 474                     len(num_microbatches) == 1
 475                     and self.args.loss_type == "policy_loss"
 476                     and self.args.kl_coef == 0
 477                     and not self.args.use_rollout_logprobs
 478                     and not self.args.get_mismatch_metrics
 479                     and not self.args.use_critic
 480                     and not self.args.use_opd
 481                     and (not self.args.use_routing_replay or self.args.use_rollout_routing_replay)
 482                     and self.args.advantage_estimator != "gspo"
 483                 )
 484                 if (
 485                     not self.args.use_rollout_logprobs or self.args.get_mismatch_metrics
 486                 ) and not can_reuse_log_probs_in_loss:
 487                     if self.args.use_routing_replay:
 488                         if self.args.use_rollout_routing_replay:
 489                             os.environ["ROUTING_REPLAY_STAGE"] = "replay_forward"
 490                             RoutingReplay.begin_lazy_pass("forward")
 491                         else:
 492                             os.environ["ROUTING_REPLAY_STAGE"] = "record"
 493                     rollout_data.update(
 494                         self.compute_log_prob(
 495                             data_iterator,
 496                             num_microbatches,
 497                             store_prefix="",
 498                         )
 499                     )
 500                     if self.args.use_rollout_routing_replay:
 501                         RoutingReplay.clear_all_forward()
 502 
 503                 if self.args.use_critic:
 504                     if external_data is not None and mpu.is_pipeline_last_stage():
 505                         values = external_data.get("values")
 506                         if values is not None:
 507                             from slime.backends.megatron_utils.data import tensors_to_gpu
 508 
 509                             rollout_data["values"] = tensors_to_gpu(values)
 510                 if self._active_model_tag != "actor":
 511                     self._switch_model("actor")
 512 
 513                 # Calculate adv and returns. Need to performed before training (instead of on the fly),
 514                 # because we may need normalize the whole rollout.
 515                 compute_advantages_and_returns(self.args, rollout_data)
 516 
 517             if self.rollout_data_postprocess is not None:
 518                 self.rollout_data_postprocess(self.args, rollout_id, rollout_data)
 519 
 520             train_metric_utils.log_rollout_data(
 521                 rollout_id,
 522                 self.args,
 523                 rollout_data,
 524             )
 525 
 526             # Train
 527             if self.args.use_routing_replay:
 528                 os.environ["ROUTING_REPLAY_STAGE"] = "replay_backward"
 529                 if self.args.use_rollout_routing_replay:
 530                     # Hold each microbatch across forward and recompute; release
 531                     # it after the backward replay has consumed every layer.
 532                     RoutingReplay.begin_lazy_pass("backward")
 533             # When dumping train debug data but the actor log_probs were not
 534             # recomputed separately (can_reuse_log_probs_in_loss / use_rollout_logprobs),
 535             # snapshot them from the training forward so the dump still carries
 536             # per-sample log_probs — at no extra forward pass.
 537             capture_log_probs = self.args.save_debug_train_data is not None and "log_probs" not in rollout_data
 538             if capture_log_probs:
 539                 enable_log_prob_capture()
 540             with timer("actor_train"):
 541                 train(
 542                     rollout_id,
 543                     self.model,
 544                     self.optimizer,
 545                     self.opt_param_scheduler,
 546                     data_iterator,
 547                     num_microbatches,
 548                     global_batch_sizes,
 549                 )
 550             if capture_log_probs:
 551                 captured = drain_captured_log_probs()
 552                 # `captured` is non-empty only on the last PP stage running a loss
 553                 # that snapshots log_probs (policy_loss), and then covers every
 554                 # local sample. Key it by this rank's `partition` to land in local
 555                 # sample order; skip otherwise (nothing to place).
 556                 if captured:
 557                     rollout_data["log_probs"] = [captured[pos] for pos in rollout_data["partition"]]
 558 
 559             self.prof.step(rollout_id=rollout_id)
 560 
 561         train_data_utils.save_debug_train_data(self.args, rollout_id=rollout_id, rollout_data=rollout_data)
 562 
 563         if self.args.use_routing_replay:
 564             RoutingReplay.clear_all()
 565 
 566         # update the cpu actor weight to the latest model
 567         self.weights_backuper.backup("actor")
 568 
 569         # Update ref model if needed
 570         if (
 571             self.args.ref_update_interval is not None
 572             and (rollout_id + 1) % self.args.ref_update_interval == 0
 573             and "ref" in self.weights_backuper.backup_tags
 574         ):
 575             with timer("ref_model_update"):
 576                 if is_megatron_main_rank():
 577                     logger.info(f"Updating ref model at rollout_id {rollout_id}")
 578                 self.weights_backuper.backup("ref")
 579 
 580         train_metric_utils.log_perf_data(
 581             rollout_id,
 582             self.args,
 583             extra_metrics=self.weight_updater.pop_metrics(),
 584         )

 586     @timer
 587     def save_model(self, rollout_id: int, force_sync: bool = False) -> None:
 588         if self.args.debug_rollout_only:
 589             return
 590 
 591         # torch dist may trigger nccl communication during saving.
 592         if self.args.offload_train:
 593             self.wake_up()
 594 
 595         if self.args.async_save:
 596             from megatron.training.async_utils import maybe_finalize_async_save
 597 
 598             maybe_finalize_async_save(blocking=True)
 599 
 600         save(rollout_id, self.model, self.optimizer, self.opt_param_scheduler)
 601 
 602         if force_sync and self.args.async_save:
 603             maybe_finalize_async_save(blocking=True)
 604 
 605         if self.args.save_hf is not None and self.role == "actor":
 606             save_hf_model_to_path(self.args, Path(self.args.save_hf.format(rollout_id=rollout_id)), self.model)
 607 
 608         if self.args.offload_train:
 609             self.sleep()
 610 
 611     @timer
 612     def update_weights(self) -> None:
 613         if self.args.debug_train_only or self.args.debug_rollout_only:
 614             return
 615 
 616         if self.args.use_fault_tolerance:
 617             if dist.get_rank() == 0:
 618                 ray.get(self.rollout_manager.recover_updatable_engines.remote())
 619             dist.barrier(group=get_gloo_group())
 620 
 621         (
 622             rollout_engines,
 623             rollout_engine_lock,
 624             num_new_engines,
 625             engine_gpu_counts,
 626             engine_gpu_offsets,
 627             engine_parallel_configs,
 628         ) = ray.get(self.rollout_manager.get_updatable_engines_and_lock.remote())
 629 
 630         reconnect_rollout_engines = self.args.offload_train and self.args.use_critic and not self.args.colocate
 631 
 632         if not rollout_engines and not reconnect_rollout_engines:
 633             if dist.get_rank() == 0:
 634                 logger.info("No updatable SGLang engines are running; skip weight update.")
 635             return
 636 
 637         if reconnect_rollout_engines:
 638             self.wake_up()
 639         elif self.args.offload_train:
 640             reload_process_groups()
 641 
 642         if num_new_engines > 0 or reconnect_rollout_engines:
 643             self.weight_updater.connect_rollout_engines(
 644                 rollout_engines,
 645                 rollout_engine_lock,
 646                 engine_gpu_counts=engine_gpu_counts,
 647                 engine_gpu_offsets=engine_gpu_offsets,
 648                 engine_parallel_configs=engine_parallel_configs,
 649             )
 650             dist.barrier(group=get_gloo_group())
 651             if dist.get_rank() == 0:
 652                 ray.get(self.rollout_manager.clear_updatable_num_new_engines.remote())
 653 
 654         with torch_memory_saver.disable() if self.args.offload_train else nullcontext():
 655             print_memory("before update_weights")
 656             self.weight_updater.update_weights()
 657             print_memory("after update_weights")
 658 
 659         if reconnect_rollout_engines:
 660             self.sleep()
 661         elif self.args.offload_train:
 662             destroy_process_groups()
 663 
 664     def load_other_checkpoint(self, model_tag: str, path: str) -> None:
 665         old_args = (
 666             self.args.load,
 667             self.args.no_load_optim,
 668             self.args.no_load_rng,
 669             self.args.finetune,
 670             self.args.ckpt_step,
 671         )

 673         self.args.no_load_optim = True
 674         self.args.no_load_rng = True
 675         self.args.finetune = True
 676 
 677         if model_tag == "ref" and self.args.ref_ckpt_step is not None:
 678             self.args.ckpt_step = self.args.ref_ckpt_step
 679         elif model_tag == "teacher" and self.args.opd_teacher_ckpt_step is not None:
 680             self.args.ckpt_step = self.args.opd_teacher_ckpt_step
 681 
 682         _, _ = load_checkpoint(
 683             self.model,
 684             None,
 685             None,
 686             checkpointing_context={},
 687         )
 688         (
 689             self.args.load,
 690             self.args.no_load_optim,
 691             self.args.no_load_rng,
 692             self.args.finetune,
 693             self.args.ckpt_step,
 694         ) = old_args
 695 
 696         self.weights_backuper.backup(model_tag)
 697         self._active_model_tag = model_tag
