  10 def create_weight_updater(
  11     args: Namespace,
  12     model: Sequence[torch.nn.Module],
  13     weights_getter: Callable[[], Mapping[str, torch.Tensor]],
  14     *,
  15     model_name: str,
  16     quantization_config: dict[str, Any] | None,
  17 ):
  18     """Select and construct the weight updater for the configured transport."""
  19     update_weight_mode = args.update_weight_mode
  20     update_weight_transport = args.update_weight_transport
  21 
  22     if update_weight_mode == "delta":
  23         # Delta sync is disk-transport only: each engine's /pull_weights applies the published
  24         # deltas into a host-local checkpoint on every host it spans, and the engines reload
  25         # via vanilla update_weights_from_disk.
  26         assert not args.colocate, "--update-weight-mode=delta is not supported with --colocate"
  27         assert update_weight_transport == "disk", "--update-weight-mode=delta requires --update-weight-transport=disk"
  28         from .update_weight_from_disk_delta import UpdateWeightFromDiskDelta
  29 
  30         update_weight_cls = UpdateWeightFromDiskDelta
  31     elif update_weight_transport == "disk":
  32         from .update_weight_from_disk import UpdateWeightFromDisk
  33 
  34         update_weight_cls = UpdateWeightFromDisk
  35     elif args.colocate:
  36         from .update_weight_from_tensor import UpdateWeightFromTensor
  37 
  38         update_weight_cls = UpdateWeightFromTensor
  39     else:
  40         assert update_weight_mode == "full"
  41         assert (
  42             update_weight_transport == "nccl"
  43         ), f"unsupported weight sync mode/transport: {update_weight_mode!r}/{update_weight_transport!r}"
  44         from .update_weight_from_distributed import UpdateWeightFromDistributed
  45 
  46         update_weight_cls = UpdateWeightFromDistributed
  47 
  48     updater = update_weight_cls(
  49         args,
  50         model,
  51         weights_getter,
  52         model_name=model_name,
  53         quantization_config=quantization_config,
  54     )
  55     updater.weight_version = getattr(args, "update_weight_start_version", 0)
  56     return updater
