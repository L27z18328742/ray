  24 class UpdateWeightFromDistributed:
  25     """
  26     Update distributed engines through a device process group. Each PP rank: group "slime-pp_{pp_rank}",
  27     only DP=TP=0 transfers. Non-expert (TP) and expert (EP) params separate.
  28     Subclasses override ``_send_weights`` / ``_on_chunk`` to inject per-mode behaviour.
  29     """
  30 
  31     def __init__(
  32         self,
  33         args: Namespace,
  34         model: Sequence[torch.nn.Module],
  35         weights_getter: Callable[[], Mapping[str, torch.Tensor]],
  36         *,
  37         model_name: str,
  38         quantization_config: dict[str, int | str | list[str]] | None,
  39     ) -> None:
  40         """
  41         Initialize. Groups created in connect_rollout_engines.
  42         """
  43         self.args = args
  44         self.model = model
  45         self.model_name = model_name
  46         self.quantization_config = quantization_config
  47         self.weight_version = 0
  48         self._model_update_groups = None
  49         self.update_weight_metrics: dict[str, float] = {}

  52         """

  58     def connect_rollout_engines(
  59         self,
  60         rollout_engines: Sequence[ActorHandle],
  61         rollout_engine_lock: ActorHandle,
  62         engine_gpu_counts: Sequence[int] | None = None,
  63         engine_gpu_offsets: Sequence[int] | None = None,
  64         engine_parallel_configs: Sequence[Mapping[str, object]] | None = None,
  65     ) -> None:
  66         """
  67         Create "slime-pp_{pp_rank}" if PP source (DP=TP=0). Lock prevents concurrent transfers.
  68         """
  69         self.rollout_engines = rollout_engines
  70         self.rollout_engine_lock = rollout_engine_lock
  71         self._engine_gpu_counts = engine_gpu_counts
  72 
  73         # For TP:
  74         #   1. AllGather parameters to rank 0
  75         #   2. Broadcast parameters from rank 0 to all sglang engines
  76         self._is_pp_src_rank = (
  77             mpu.get_data_parallel_rank(with_context_parallel=True) == 0 and mpu.get_tensor_model_parallel_rank() == 0
  78         )
  79         pp_rank = mpu.get_pipeline_model_parallel_rank()
  80         if self._is_pp_src_rank:
  81             self._group_name = f"slime-pp_{pp_rank}"
  82 
  83         if self._is_pp_src_rank:
  84             if self._model_update_groups is not None:
  85                 disconnect_rollout_engines_from_distributed(
  86                     self._group_name, self._model_update_groups, self.rollout_engines
  87                 )
  88             self._model_update_groups = connect_rollout_engines_from_distributed(
  89                 self.args,
  90                 self._group_name,
  91                 rollout_engines,
  92                 engine_gpu_counts=engine_gpu_counts,
  93             )
  94 
  95     def disconnect_rollout_engines(self) -> None:
  96         if not getattr(self, "_is_pp_src_rank", False) or self._model_update_groups is None:
  97             return
  98         disconnect_rollout_engines_from_distributed(self._group_name, self._model_update_groups, self.rollout_engines)
  99         self._model_update_groups = None

 101     @torch.no_grad()
 102     def update_weights(self) -> None:
 103         """
 104         Pause → flush → _send_weights → continue. Progress on PP source.
 105         """
 106         self.weight_version += 1
 107 
 108         if dist.get_rank() == 0:
 109             ray.get([engine.pause_generation.remote() for engine in self.rollout_engines])
 110             ray.get([engine.flush_cache.remote() for engine in self.rollout_engines])
 111 
 112             # int4/fp4 pre_process
 113             if self.quantization_config and self.quantization_config["quant_method"] in ["compressed-tensors"]:
 114                 post_process_weights(
 115                     restore_weights_before_load=True,
 116                     post_process_quantization=False,
 117                     rollout_engines=self.rollout_engines,
 118                 )
 119         dist.barrier(group=get_gloo_group())
 120 
 121         pbar = tqdm(desc=f"[{self._group_name}] Update weights", total=0) if self._is_pp_src_rank else None
 122         self._send_weights(pbar)
 123 
 124         if dist.get_rank() == 0:
 125             # int4/fp4 post_process
 126             if self.quantization_config and self.quantization_config["quant_method"] in ["compressed-tensors"]:
 127                 post_process_weights(
 128                     restore_weights_before_load=False,
 129                     post_process_quantization=True,
 130                     rollout_engines=self.rollout_engines,
 131                 )
 132             ray.get([engine.continue_generation.remote() for engine in self.rollout_engines])
 133         dist.barrier(group=get_gloo_group())

 135     def _send_weights(self, pbar: tqdm | None) -> None:
 136         """
 137         Non-expert (TP) pass → barrier → expert (EP) pass → barrier. Each iterator
 138         yields broadcast-ready chunks (bucketing happens internally); subclasses
 139         override ``_on_chunk`` to inject per-chunk behaviour.
 140         """
 141         for chunk_iter in (self._iter_non_expert_chunks(), self._iter_expert_chunks()):
 142             for hf_chunk in chunk_iter:
 143                 self._on_chunk(hf_chunk)
 144                 self._update_bucket_weights_from_distributed(hf_chunk, pbar=pbar)
 145             dist.barrier(group=get_gloo_group())
 146 
 147     def _on_chunk(self, hf_chunk: list[tuple[str, torch.Tensor]]) -> None:
 148         """
 149         Hook for each HF chunk in ``_send_weights`` before its broadcast. No-op by default.
 150         """
 151 
 152     def _iter_non_expert_chunks(self) -> Iterator[list[tuple[str, torch.Tensor]]]:
 153         """
 154         Yield broadcast-sized HF chunks of non-expert params: TP all-gather +
 155         HF convert per param, then bucket up to ``--update-weight-buffer-size``.
 156         Empty on non-PP-src ranks (they still join all_gather_param).
 157         """
 158         buffer_size = 0
 159         buffer: list[tuple[str, torch.Tensor]] = []
 160         for name, param in named_params_and_buffers(self.args, self.model):
 161             if ".experts." in name:
 162                 continue
 163             param = all_gather_param(name, param)
 164             if not self._is_pp_src_rank:
 165                 continue
 166             hf_chunk = convert_to_hf(self.args, self.model_name, name, param, self.quantization_config)
 167             chunk_bytes = sum(t.numel() * t.element_size() for _, t in hf_chunk)
 168             if buffer and buffer_size + chunk_bytes > self.args.update_weight_buffer_size:
 169                 yield buffer
 170                 buffer = []
 171                 buffer_size = 0
 172             buffer.extend(hf_chunk)
 173             buffer_size += chunk_bytes
 174         if buffer:
 175             yield buffer
 176 
 177     def _iter_expert_chunks(self) -> Iterator[list[tuple[str, torch.Tensor]]]:
 178         """
 179         Yield one HF chunk per EP-weighted batch of expert params: TP gather +
 180         buffer until threshold, then EP gather + HF convert.
 181         """
 182         params = ((n, p) for n, p in named_params_and_buffers(self.args, self.model) if ".experts." in n)
 183         buffer_size = 0
 184         batch: list[tuple[str, torch.Tensor]] = []
 185         for name, param in params:
 186             param = all_gather_param(name, param)
 187             param_size = param.numel() * param.element_size()
 188             if (
 189                 buffer_size + param_size
 190             ) * mpu.get_expert_model_parallel_world_size() > self.args.update_weight_buffer_size:
 191                 hf_chunk = self._ep_gather_and_convert(batch)
 192                 if hf_chunk:
 193                     yield hf_chunk
 194                 batch = []
 195                 buffer_size = 0
 196             batch.append((name, param))
 197             buffer_size += param_size
 198         if batch:
 199             hf_chunk = self._ep_gather_and_convert(batch)
 200             if hf_chunk:
 201                 yield hf_chunk
 202 
 203     def _ep_gather_and_convert(self, named_tensors: list[tuple[str, torch.Tensor]]) -> list[tuple[str, torch.Tensor]]:
 204         """
 205         EP all-gather a buffered batch + HF convert on PP source. Returns HF tensors on
 206         PP source, [] elsewhere. Clears ``named_tensors``.
 207         """
 208         names = [name for name, _ in named_tensors]
 209         all_names = [None] * mpu.get_expert_model_parallel_world_size()
 210         dist.all_gather_object(all_names, names, group=mpu.get_expert_model_parallel_group())
 211 
 212         for names in all_names:
 213             assert len(named_tensors) == len(names), f"mismatch names length: {len(named_tensors)} != {len(names)}"
 214 
 215         all_gathered_params = [[] for _ in range(mpu.get_expert_model_parallel_world_size())]
 216         handles = []
 217         for i, (_name, param) in enumerate(named_tensors):
 218             params = [
 219                 torch.empty_like(param.data, device=accelerator.current_device())
 220                 for _ in range(mpu.get_expert_model_parallel_world_size())
 221             ]
 222             handle = dist.all_gather(params, param.data, group=mpu.get_expert_model_parallel_group(), async_op=True)
 223             handles.append(handle)
 224             for ep_rank, names in enumerate(all_names):
 225                 all_gathered_params[ep_rank].append((names[i], params[ep_rank]))
 226         for handle in handles:
 227             handle.wait()
 228 
 229         named_tensors.clear()
 230         if not self._is_pp_src_rank:
 231             return []
 232 
 233         all_gathered_params = sum(all_gathered_params, [])
 234         converted_hf_tensors = []
 235         for name, param in all_gathered_params:
 236             converted_hf_tensors += convert_to_hf(self.args, self.model_name, name, param, self.quantization_config)
 237         return converted_hf_tensors

 239     def _update_bucket_weights_from_distributed(
 240         self,
 241         converted_named_tensors: list[tuple[str, torch.Tensor]],
 242         pbar: tqdm | None = None,
 243         load_format: str | None = None,
 244     ) -> None:
 245         """
 246         Lock → transfer → clear → unlock → pbar++. Lock prevents communication deadlock.
 247         """
 248         # Lock the rollout engines to prevent communication deadlock.
 249         while not ray.get(self.rollout_engine_lock.acquire.remote()):
 250             time.sleep(0.1)
 251 
 252         refs = update_weights_from_distributed(
 253             self._group_name,
 254             self._model_update_groups,
 255             self.weight_version,
 256             self.rollout_engines,
 257             converted_named_tensors,
 258             load_format=load_format,
 259         )
 260 
 261         ray.get(refs)
 262         converted_named_tensors.clear()
 263         ray.get(self.rollout_engine_lock.release.remote())
 264         pbar.update(1)

 267 def connect_rollout_engines_from_distributed(
 268     args: Namespace,
 269     group_name: str,
 270     rollout_engines: Sequence[ActorHandle],
 271     engine_gpu_counts: Sequence[int] | None = None,
 272 ) -> dist.ProcessGroup:
 273     """
 274     Create a device process group: training rank 0 + all engine GPUs. Blocks until joined.
 275 
 276     ``engine_gpu_counts`` gives the number of GPUs per engine.  When engines
 277     have heterogeneous TP sizes (e.g. prefill TP=2, decode TP=4), each engine
 278     occupies a different number of ranks in the process group.
 279     """
 280     if engine_gpu_counts is None:
 281         engine_gpu_counts = [args.rollout_num_gpus_per_engine] * len(rollout_engines)
 282 
 283     master_address = ray._private.services.get_node_ip_address()
 284     with socket.socket() as sock:
 285         sock.bind(("", 0))
 286         master_port = sock.getsockname()[1]
 287     world_size = sum(engine_gpu_counts) + 1  # +1 for training rank 0
 288 
 289     # Compute cumulative rank offsets: engine i starts at cumulative[i] + 1.
 290     cumulative = [0]
 291     for c in engine_gpu_counts:
 292         cumulative.append(cumulative[-1] + c)
 293 
 294     backend = accelerator.weight_update_backend()
 295     refs = [
 296         engine.init_weights_update_group.remote(
 297             master_address=master_address,
 298             master_port=master_port,
 299             rank_offset=cumulative[i] + 1,
 300             world_size=world_size,
 301             group_name=group_name,
 302             backend=backend,
 303         )
 304         for i, engine in enumerate(rollout_engines)
 305     ]
 306     model_update_groups = init_process_group(
 307         backend=backend,
 308         init_method=f"tcp://{_wrap_ipv6(master_address)}:{master_port}",
 309         world_size=world_size,
 310         rank=0,
 311         group_name=group_name,
 312     )
 313     ray.get(refs)
 314     return model_update_groups

 317 def disconnect_rollout_engines_from_distributed(group_name, model_update_groups, rollout_engines):
 318     """
 319     Destroy the weight-update process group on training and engines.
 320     """
 321     refs = [engine.destroy_weights_update_group.remote(group_name) for engine in rollout_engines]
 322     dist.destroy_process_group(model_update_groups)
 323     ray.get(refs)

 326 def update_weights_from_distributed(
 327     group_name: str,
 328     group: dist.ProcessGroup,
 329     weight_version: int,
 330     rollout_engines: Sequence[ActorHandle],
 331     converted_named_tensors: Sequence[tuple[str, torch.Tensor]],
 332     load_format: str | None = None,
 333 ) -> list[ObjectRef]:
 334     """
 335     Send metadata through Ray and tensors through the configured transport.
 336     """
 337     refs = [
 338         engine.update_weights_from_distributed.remote(
 339             names=[name for name, _ in converted_named_tensors],
 340             dtypes=[param.dtype for _, param in converted_named_tensors],
 341             shapes=[param.shape for _, param in converted_named_tensors],
 342             group_name=group_name,
 343             weight_version=str(weight_version),
 344             load_format=load_format,
 345         )
 346         for engine in rollout_engines
 347     ]
 348     handles = []
 349     for _, param in converted_named_tensors:
 350         handles.append(dist.broadcast(param.data, 0, group=group, async_op=True))
 351     for handle in handles:
 352         handle.wait()
 353 
 354     return refs
