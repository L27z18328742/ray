  30 def _build_flattened_tensor_data(
  31     named_tensors: list[tuple[str, torch.Tensor]],
  32 ) -> dict[str, Any]:
  33     if not named_tensors:
  34         return {
  35             "flattened_tensor": torch.empty(0, dtype=torch.uint8, device=accelerator.current_device()),
  36             "metadata": [],
  37         }
  38 
  39     # Do not reuse the IPC-facing flattened tensor. SGLang returns from the
  40     # HTTP/Ray request after enqueueing GPU copies into model weights, but it
  41     # does not guarantee a CUDA-device sync before the response. Reusing and
  42     # overwriting the same producer buffer immediately after ray.get can race
  43     # with the consumer-side copy and corrupt weights.
  44     flattened_tensor_bucket = FlattenedTensorBucket(named_tensors=named_tensors)
  45     return {
  46         "flattened_tensor": flattened_tensor_bucket.get_flattened_tensor(),
  47         "metadata": flattened_tensor_bucket.get_metadata(),
  48     }

  51 class UpdateWeightFromTensor:
  52     """
  53     Update rollout engines from tensor dict:
  54     load(dict→GPU) → broadcast PP/EP(GPU NCCL) → gather TP(GPU NCCL) → convert HF(GPU) → send.
  55     Colocated: GPU→CPU serialize → gather_object(Gloo CPU, collects from rollout_num_gpus_per_engine ranks) → Ray IPC to engine.
  56     Distributed: GPU NCCL broadcast to remote engines.
  57     """
  58 
  59     def __init__(
  60         self,
  61         args: Namespace,
  62         model: Sequence[torch.nn.Module],
  63         weights_getter: Callable[[], Mapping[str, torch.Tensor]],
  64         *,
  65         model_name: str,
  66         quantization_config: dict[str, int | str | list[str]] | None,
  67     ) -> None:
  68         """
  69         Compute param buckets.  IPC Gloo groups are created later in
  70         ``connect_rollout_engines`` once ``engine_gpu_counts`` is known.
  71         """
  72         self.args = args
  73         self.model = model
  74         self.weights_getter = weights_getter
  75         self.rank = dist.get_rank()
  76         self.model_name = model_name
  77         self.quantization_config = quantization_config
  78         self.weight_version = 0
  79         self.update_weight_metrics: dict[str, float] = {}
  80 
  81         self._hf_weight_iterator = HfWeightIteratorDirect(
  82             args=args, model=model, model_name=model_name, quantization_config=quantization_config
  83         )
  84         param_info_buckets = getattr(self._hf_weight_iterator, "megatron_local_param_info_buckets", None)
  85         self._full_param_info_buckets = (
  86             tuple(tuple(bucket) for bucket in param_info_buckets) if param_info_buckets is not None else None
  87         )
  88         self._non_expert_param_info_buckets: list[list[ParamInfo]] | None = None
  89 
  90         self._ipc_gather_group = None
  91         self._ipc_gather_src = None
  92         self._ipc_engine = None
  93         self._model_update_groups = None
  94         self._expert_transfer_plan = []

  96     def connect_rollout_engines(
  97         self,
  98         rollout_engines: Sequence[ActorHandle],
  99         rollout_engine_lock: ActorHandle,
 100         engine_gpu_counts: Sequence[int] | None = None,
 101         engine_gpu_offsets: Sequence[int] | None = None,
 102         engine_parallel_configs: Sequence[Mapping[str, Any]] | None = None,
 103     ) -> None:
 104         """
 105         Split colocated/distributed engines. Global source rank (DP=TP=PP=0) creates NCCL
 106         for distributed. Map ranks to colocated IPC engines.
 107         """
 108         self.rollout_engines = rollout_engines
 109 
 110         if engine_gpu_counts is None:
 111             engine_gpu_counts = [self.args.rollout_num_gpus_per_engine] * len(rollout_engines)
 112         if engine_gpu_offsets is None:
 113             # Fallback: assume engines are densely packed (no placeholder gaps).
 114             engine_gpu_offsets = []
 115             offset = 0
 116             for c in engine_gpu_counts:
 117                 engine_gpu_offsets.append(offset)
 118                 offset += c
 119 
 120         # Compute colocated engine count: engines whose GPUs fall within actor GPU range.
 121         total_actor_gpus = self.args.actor_num_nodes * self.args.actor_num_gpus_per_node
 122         colocate_engine_nums = 0
 123         for gpu_offset, gpu_count in zip(engine_gpu_offsets, engine_gpu_counts, strict=True):
 124             if gpu_offset + gpu_count > total_actor_gpus:
 125                 break
 126             colocate_engine_nums += 1
 127 
 128         self.use_distribute = len(rollout_engines) > colocate_engine_nums
 129 
 130         if self.use_distribute:
 131             self.rollout_engines = rollout_engines[:colocate_engine_nums]
 132             self.distributed_rollout_engines = rollout_engines[colocate_engine_nums:]
 133             distributed_gpu_counts = engine_gpu_counts[colocate_engine_nums:]
 134             self._is_distributed_src_rank = (
 135                 mpu.get_data_parallel_rank(with_context_parallel=True) == 0
 136                 and mpu.get_tensor_model_parallel_rank() == 0
 137                 and mpu.get_pipeline_model_parallel_rank() == 0
 138             )
 139             self._group_name = "slime"
 140             if self._is_distributed_src_rank:
 141                 if self._model_update_groups is not None:
 142                     disconnect_rollout_engines_from_distributed(
 143                         self._group_name, self._model_update_groups, self.distributed_rollout_engines
 144                     )
 145 
 146                 self._model_update_groups = connect_rollout_engines_from_distributed(
 147                     self.args,
 148                     self._group_name,
 149                     self.distributed_rollout_engines,
 150                     engine_gpu_counts=distributed_gpu_counts,
 151                 )
 152 
 153         colocate_gpu_offsets = engine_gpu_offsets[:colocate_engine_nums]
 154         colocate_gpu_counts = engine_gpu_counts[:colocate_engine_nums]
 155         colocate_parallel_configs = (
 156             engine_parallel_configs[:colocate_engine_nums] if engine_parallel_configs is not None else None
 157         )
 158 
 159         # Create IPC Gloo gather groups (only on first call; partitioning is
 160         # fixed across reconnects).
 161         if self._ipc_gather_group is None:
 162             for i in range(colocate_engine_nums):
 163                 group_ranks = list(range(colocate_gpu_offsets[i], colocate_gpu_offsets[i] + colocate_gpu_counts[i]))
 164                 new_group = dist.new_group(ranks=group_ranks, backend="gloo")
 165                 if self.rank in group_ranks:
 166                     self._ipc_gather_group = new_group
 167                     self._ipc_gather_src = colocate_gpu_offsets[i]
 168 
 169         # Map training ranks to colocated engine actors.
 170         for i, engine in enumerate(self.rollout_engines):
 171             start = colocate_gpu_offsets[i]
 172             end = start + colocate_gpu_counts[i]
 173             if start <= self.rank < end:
 174                 self._ipc_engine = engine
 175 
 176         self._non_expert_param_info_buckets, self._expert_transfer_plan = configure_expert_routing(
 177             args=self.args,
 178             full_param_info_buckets=self._full_param_info_buckets,
 179             get_local_weight_names=self.weights_getter,
 180             engine_gpu_counts=colocate_gpu_counts,
 181             engine_gpu_offsets=colocate_gpu_offsets,
 182             engine_parallel_configs=colocate_parallel_configs,
 183             use_distribute=self.use_distribute,
 184         )
 185 
 186     def pop_metrics(self) -> dict[str, float]:
 187         """
 188         Return and clear ``update_weight_metrics``. Empty under colocate today;
 189         kept symmetric with UpdateWeightFromDistributed so the actor can drain unconditionally.
 190         """
 191         out, self.update_weight_metrics = self.update_weight_metrics, {}
 192         return out

 194     def _prepare_expert_weight_batch(
 195         self,
 196         transfers: Sequence[Any],
 197         megatron_local_weights: Mapping[str, torch.Tensor],
 198         staging_buffers: dict[tuple[torch.dtype, tuple[int, ...]], list[torch.Tensor]],
 199     ) -> list[tuple[str, torch.Tensor]]:
 200         local_params = []
 201         p2p_ops = []
 202         buffer_offsets: dict[tuple[torch.dtype, tuple[int, ...]], int] = defaultdict(int)
 203         for transfer in transfers:
 204             for expert_param in transfer.params:
 205                 info = expert_param.info
 206                 if self.rank != transfer.source_rank and self.rank not in transfer.target_ranks:
 207                     continue
 208                 key = (info.dtype, tuple(info.shape))
 209                 pool = staging_buffers.setdefault(key, [])
 210                 offset = buffer_offsets[key]
 211                 buffer_offsets[key] = offset + 1
 212                 if offset == len(pool):
 213                     pool.append(torch.empty(info.shape, dtype=info.dtype, device=accelerator.device()))
 214                 tensor = pool[offset]
 215                 if self.rank == transfer.source_rank:
 216                     source = megatron_local_weights[info.name]
 217                     if source.shape != info.shape or source.dtype != info.dtype:
 218                         raise ValueError(f"expert metadata changed for {info.name}")
 219                     tensor.copy_(source, non_blocking=True)
 220                     p2p_ops.extend(
 221                         dist.P2POp(dist.isend, tensor, target_rank)
 222                         for target_rank in transfer.target_ranks
 223                         if target_rank != self.rank
 224                     )
 225                     if self.rank in expert_param.target_ranks:
 226                         local_params.append((expert_param, tensor))
 227                 else:
 228                     p2p_ops.append(dist.P2POp(dist.irecv, tensor, transfer.source_rank))
 229                     local_params.append((expert_param, tensor))
 230 
 231         for request in dist.batch_isend_irecv(p2p_ops) if p2p_ops else ():
 232             request.wait()
 233 
 234         hf_named_tensors = []
 235         for expert_param, tensor in local_params:
 236             hf_named_tensors.extend(
 237                 convert_to_hf(
 238                     self.args,
 239                     self.model_name,
 240                     expert_param.info.name,
 241                     tensor,
 242                     self.quantization_config,
 243                 )
 244             )
 245         return hf_named_tensors

 247     def _update_expert_weights(
 248         self,
 249         megatron_local_weights: Mapping[str, torch.Tensor],
 250     ) -> None:
 251         dist.barrier(group=get_gloo_group())
 252         # Initialize WORLD on all ranks before subset batched P2P.
 253         dist.barrier()
 254         # Reuse staging across layers instead of fragmenting the CUDA allocator.
 255         staging_buffers: dict[tuple[torch.dtype, tuple[int, ...]], list[torch.Tensor]] = {}
 256         for transfer_group in tqdm(
 257             self._expert_transfer_plan,
 258             disable=self.rank != 0,
 259             desc="Update expert weights",
 260         ):
 261             for transfer_batch in transfer_group:
 262                 hf_named_tensors = self._prepare_expert_weight_batch(
 263                     transfer_batch,
 264                     megatron_local_weights,
 265                     staging_buffers,
 266                 )
 267                 refs, long_lived_tensors = self._send_hf_params(hf_named_tensors)
 268                 ray.get(refs)
 269                 dist.barrier(group=get_gloo_group())
 270                 accelerator.synchronize()
 271                 del refs, long_lived_tensors, hf_named_tensors
 272                 accelerator.ipc_collect()
 273                 accelerator.empty_cache()
 274         del staging_buffers
 275         accelerator.empty_cache()

 277     @torch.no_grad()
 278     def update_weights(self) -> None:
 279         """
 280         version++, flush caches, process buckets. Progress on rank 0.
 281         """
 282         self.weight_version += 1
 283 
 284         if self.rank == 0:
 285             ray.get([engine.pause_generation.remote() for engine in self.rollout_engines])
 286             ray.get([engine.flush_cache.remote() for engine in self.rollout_engines])
 287             if self.quantization_config and self.quantization_config["quant_method"] in ["compressed-tensors"]:
 288                 post_process_weights(
 289                     restore_weights_before_load=True,
 290                     post_process_quantization=False,
 291                     rollout_engines=self.rollout_engines,
 292                 )
 293         dist.barrier(group=get_gloo_group())
 294 
 295         megatron_local_weights = self.weights_getter()
 296 
 297         param_info_buckets = (
 298             self._non_expert_param_info_buckets if self._expert_transfer_plan else self._full_param_info_buckets
 299         )
 300         for hf_named_tensors in self._hf_weight_iterator.get_hf_weight_chunks(
 301             megatron_local_weights,
 302             param_info_buckets=param_info_buckets,
 303         ):
 304             refs, long_lived_tensors = self._send_hf_params(hf_named_tensors)
 305             ray.get(refs)
 306             # Free GPU tensors so the caching allocator can reuse the blocks,
 307             # then release CUDA IPC cache entries whose consumers (sglang engines)
 308             # have already closed their IPC handles.
 309             del refs, long_lived_tensors, hf_named_tensors
 310             accelerator.ipc_collect()
 311             accelerator.empty_cache()
 312 
 313         if self._expert_transfer_plan:
 314             self._update_expert_weights(megatron_local_weights)
 315 
 316         del megatron_local_weights
 317         dist.barrier(group=get_gloo_group())
 318         # After the barrier all engines have returned, so every rank's last-chunk
 319         # IPC handles are now released by the consumers.  Clean them up.
 320         accelerator.ipc_collect()
 321         accelerator.empty_cache()
 322 
 323         # int4/fp4 post_process
 324         if self.rank == 0:
 325             if self.quantization_config and self.quantization_config["quant_method"] in ["compressed-tensors"]:
 326                 post_process_weights(
 327                     restore_weights_before_load=False,
 328                     post_process_quantization=True,
 329                     rollout_engines=self.rollout_engines,
 330                 )
 331             ray.get([engine.continue_generation.remote() for engine in self.rollout_engines])
 332         dist.barrier(group=get_gloo_group())

 334     def _send_hf_params(self, hf_named_tensors) -> tuple[list[ObjectRef], Any]:
 335         all_refs = []
 336 
 337         refs_colocated, long_lived_tensors = _send_to_colocated_engine(
 338             hf_named_tensors,
 339             ipc_engine=self._ipc_engine,
 340             ipc_gather_src=self._ipc_gather_src,
 341             ipc_gather_group=self._ipc_gather_group,
 342             weight_version=self.weight_version,
 343         )
 344         all_refs.extend(refs_colocated)
 345 
 346         if self.use_distribute and self._is_distributed_src_rank:
 347             refs_distributed = update_weights_from_distributed(
 348                 self._group_name,
 349                 self._model_update_groups,
 350                 self.weight_version,
 351                 self.distributed_rollout_engines,
 352                 hf_named_tensors,
 353             )
 354             if refs_distributed:
 355                 all_refs.extend(refs_distributed)
 356 
 357         return all_refs, long_lived_tensors

 360 def _send_to_colocated_engine(
 361     hf_named_tensors: list[tuple[str, torch.Tensor]],
 362     *,
 363     ipc_engine,
 364     ipc_gather_src,
 365     ipc_gather_group,
 366     weight_version,
 367 ) -> tuple[list[ObjectRef], Any]:
 368     # Placeholder ranks (GPU slots reserved but no engine) have no gather group.
 369     # gather_object is only collective among group members, so we skip entirely.
 370     if ipc_gather_group is None:
 371         return [], None
 372 
 373     long_live_tensors = []
 374 
 375     if getattr(FlattenedTensorBucket, "supports_multi_dtypes", False):
 376         converted_named_tensors_by_dtypes = {"dtype": hf_named_tensors} if hf_named_tensors else {}
 377     else:
 378         converted_named_tensors_by_dtypes = {}
 379         for name, tensor in hf_named_tensors:
 380             dtype = tensor.dtype
 381             if dtype not in converted_named_tensors_by_dtypes:
 382                 converted_named_tensors_by_dtypes[dtype] = []
 383             converted_named_tensors_by_dtypes[dtype].append((name, tensor))
 384 
 385     serialized_tensors = []
 386     for _dtype, named_tensors in converted_named_tensors_by_dtypes.items():
 387         flattened_tensor_data = _build_flattened_tensor_data(named_tensors)
 388         long_live_tensors.append(flattened_tensor_data)
 389         serialized_tensors.append(MultiprocessingSerializer.serialize(flattened_tensor_data, output_str=True))
 390 
 391     serialized_named_tensors = (
 392         [None] * dist.get_world_size(ipc_gather_group) if ipc_gather_src == dist.get_rank() else None
 393     )
 394     dist.gather_object(
 395         serialized_tensors,
 396         object_gather_list=serialized_named_tensors,
 397         dst=ipc_gather_src,
 398         group=ipc_gather_group,
 399     )
 400 
 401     refs = []
 402     if dist.get_rank() == ipc_gather_src:
 403         num_buckets = max(len(tensors) for tensors in serialized_named_tensors)
 404         empty_serialized_tensor = None
 405         for i in range(num_buckets):
 406             serialized_tensors_for_dtype = []
 407             for tensors in serialized_named_tensors:
 408                 if i < len(tensors):
 409                     serialized_tensors_for_dtype.append(tensors[i])
 410                     continue
 411 
 412                 if empty_serialized_tensor is None:
 413                     empty_tensor_data = _empty_flattened_tensor_data()
 414                     long_live_tensors.append(empty_tensor_data)
 415                     empty_serialized_tensor = MultiprocessingSerializer.serialize(empty_tensor_data, output_str=True)
 416                 serialized_tensors_for_dtype.append(empty_serialized_tensor)
 417 
 418             kwargs = {
 419                 "serialized_named_tensors": serialized_tensors_for_dtype,
 420                 "load_format": "flattened_bucket",
 421                 "weight_version": str(weight_version),
 422             }
 423             refs.append(ipc_engine.update_weights_from_tensor.remote(**kwargs))
 424 
 425     return refs, long_live_tensors
