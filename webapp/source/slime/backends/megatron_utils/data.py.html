  14 def get_batch(
  15     data_iterator: "DataIterator",
  16     keys: Sequence[str],
  17     pad_multiplier: int = 128,
  18     allgather_cp: bool = False,
  19 ) -> dict[str, torch.Tensor | PackedSeqParams | list[torch.Tensor] | None]:
  20     """
  21     Generate a CP-ready micro-batch with packed sequence parameters.
  22 
  23     Steps:
  24     - Fetch raw fields via iterator.
  25     - Save original token tensors under "unconcat_tokens".
  26     - Slice tokens into two chunks for Context Parallelism (CP), concatenate, and pad to a configurable multiple.
  27     - Build cu_seqlens and `PackedSeqParams` with T-H-D layout (T: sequence length, H: attention heads, D: head dimension).
  28 
  29     Args:
  30         data_iterator: Iterator providing micro-batch data.
  31         keys: List of keys to fetch from the iterator.
  32         pad_multiplier: Multiplier for padding size calculation (default: 128).
  33 
  34     Returns a dict including:
  35     - "tokens": torch.LongTensor of shape [1, T_padded] on the current CUDA device
  36     - "unconcat_tokens": list[torch.LongTensor] for the micro-batch before CP slicing/concat
  37     - "packed_seq_params": PackedSeqParams with T-H-D settings (cu_seqlens on CUDA, dtype=int)
  38     Plus any other requested keys forwarded from the iterator.
  39     """
  40 
  41     assert "tokens" in keys
  42     batch = data_iterator.get_next(keys)
  43 
  44     tokens = batch["tokens"]
  45     # use 0 as the pad token id should be fine?
  46     pad_token_id = 0
  47     pad_size = mpu.get_tensor_model_parallel_world_size() * pad_multiplier
  48 
  49     # for cp, we need all tokens to calculate logprob
  50     batch["unconcat_tokens"] = tokens
  51 
  52     cp_size = mpu.get_context_parallel_world_size()
  53     cp_rank = mpu.get_context_parallel_rank()
  54 
  55     if allgather_cp:
  56         # DSA mode: concatenate all sequences first, then slice once with CP.
  57         # We also pad the *global* concatenated stream to make per-rank chunks equal.
  58         cu_seqlens_list: list[int] = [0]
  59         for t in tokens:
  60             cu_seqlens_list.append(cu_seqlens_list[-1] + t.size(0))
  61 
  62         tokens = torch.cat(tokens, dim=0)
  63 
  64         # Pad global stream so (1) divisible by cp_size (equal chunks),
  65         # (2) divisible by pad_size (reduce fragmentation).
  66         global_pad_size = cp_size * pad_size
  67         pad = (global_pad_size - tokens.size(0) % global_pad_size) % global_pad_size
  68         if pad != 0:
  69             tokens = F.pad(tokens, (0, pad), value=pad_token_id)
  70             cu_seqlens_list.append(cu_seqlens_list[-1] + pad)
  71 
  72         cu_seqlens = torch.tensor(cu_seqlens_list, dtype=torch.int, device=accelerator.current_device())
  73         tokens = tokens.chunk(cp_size, dim=0)[cp_rank]
  74     else:
  75         tokens = [slice_with_cp(t, pad_token_id) for t in tokens]
  76 
  77         cu_seqlens = [0]
  78         for t in tokens:
  79             cu_seqlens.append(cu_seqlens[-1] + t.size(0))
  80 
  81         tokens = torch.cat(tokens)
  82 
  83         # Always pad to reduce memory fragmentation and maybe make the computation faster
  84         pad = (pad_size - tokens.size(0) % pad_size) % pad_size
  85         if pad != 0:
  86             tokens = F.pad(tokens, (0, pad), value=pad_token_id)
  87             cu_seqlens.append(cu_seqlens[-1] + pad)
  88 
  89         # thd requires the cu_seqlens to be of the origin length
  90         cu_seqlens = torch.tensor(cu_seqlens, dtype=torch.int, device=accelerator.device()) * cp_size
  91 
  92     max_seqlen = (cu_seqlens[1:] - cu_seqlens[:-1]).max().item()
  93     packed_seq_params = PackedSeqParams(
  94         cu_seqlens_q=cu_seqlens,
  95         cu_seqlens_kv=cu_seqlens,
  96         max_seqlen_q=max_seqlen,
  97         max_seqlen_kv=max_seqlen,
  98         qkv_format="thd",
  99     )
 100 
 101     tokens = tokens.unsqueeze(0)
 102 
 103     batch["tokens"] = tokens
 104     batch["packed_seq_params"] = packed_seq_params
 105 
 106     # loss masks
 107     loss_masks = []
 108     for loss_mask, total_length, response_length in zip(
 109         batch["loss_masks"],
 110         batch["total_lengths"],
 111         batch["response_lengths"],
 112         strict=True,
 113     ):
 114         prompt_length = total_length - response_length
 115         # Align mask to token stream positions (prompt_length-1 left pad, 1 right pad)
 116         loss_mask = F.pad(loss_mask, (prompt_length - 1, 1), value=0)
 117         if allgather_cp:
 118             loss_masks.append(loss_mask)
 119             continue
 120         loss_mask = slice_with_cp(loss_mask, 0)
 121         loss_masks.append(loss_mask)
 122 
 123     if allgather_cp:
 124         # DSA: concatenate first (same as tokens), pad globally (same pad as above), then slice once.
 125         loss_masks = torch.cat(loss_masks, dim=0)
 126         if pad != 0:
 127             loss_masks = F.pad(loss_masks, (0, pad), value=0)
 128         loss_masks = loss_masks.chunk(cp_size, dim=0)[cp_rank].unsqueeze(0)
 129     else:
 130         loss_masks = torch.cat(loss_masks)
 131         loss_masks = F.pad(loss_masks, (0, pad), value=0).unsqueeze(0)
 132 
 133     assert loss_masks.shape == tokens.shape, f"loss_masks.shape: {loss_masks.shape}, tokens.shape: {tokens.shape}"
 134     batch["full_loss_masks"] = loss_masks
 135 
 136     # Process multimodal training tensors if present
 137     multimodal_train_inputs = batch.get("multimodal_train_inputs", None)
 138     if multimodal_train_inputs is not None:
 139         multimodal_data = {}  # key -> concatenated tensor
 140         for mm_input_dict in multimodal_train_inputs:
 141             if mm_input_dict is not None:
 142                 for key, mm_tensor in mm_input_dict.items():
 143                     if key not in multimodal_data:
 144                         multimodal_data[key] = mm_tensor
 145                     else:
 146                         multimodal_data[key] = torch.cat([multimodal_data[key], mm_tensor], dim=0)
 147         batch["multimodal_train_inputs"] = multimodal_data
 148 
 149     return batch

 152 class DataIterator:
 153     """Iterator over a rollout dict following an explicit micro-batch index schedule."""
 154 
 155     def __init__(
 156         self,
 157         rollout_data: RolloutBatch,
 158         micro_batch_indices: list[list[int]],
 159     ) -> None:
 160         """Initialize an iterator over ``rollout_data``.
 161 
 162         Args:
 163             rollout_data: Dict of per-sample fields for this DP rank.
 164             micro_batch_indices: List of mbs, each mbs being the local sample indices to select.
 165         """
 166         self.rollout_data = rollout_data
 167         self.micro_batch_indices = micro_batch_indices
 168         self.offset = 0
 169 
 170     def get_next(self, keys: Sequence[str]) -> dict[str, list[object] | None]:
 171         """Return the next micro-batch for the requested keys.
 172 
 173         Returns a dict mapping each key to a list subset (or None if absent).
 174         """
 175         batch = {}
 176         indices = self.micro_batch_indices[self.offset]
 177         for key in keys:
 178             vals = self.rollout_data.get(key, None)
 179             if vals is None:
 180                 batch[key] = None
 181             else:
 182                 batch[key] = [vals[i] for i in indices]
 183         self.offset += 1
 184         return batch
 185 
 186     def reset(self) -> "DataIterator":
 187         """Reset internal offset to the start and return self."""
 188         self.offset = 0
 189         return self
 190 
 191 
 192 def get_data_iterator(rollout_data: RolloutBatch) -> list[DataIterator]:
 193     """Build one ``DataIterator`` per VPP stage from the pre-computed schedule in ``rollout_data``."""
 194     vpp_size = mpu.get_virtual_pipeline_model_parallel_world_size() or 1
 195     micro_batch_indices = rollout_data["micro_batch_indices"]
 196     return [DataIterator(rollout_data, micro_batch_indices) for _ in range(vpp_size)]

 199 def tensors_to_cpu(tensor_list):
 200     """Move a list of GPU tensors to CPU for Ray object store transfer.
 201 
 202     Args:
 203         tensor_list: List of GPU tensors, or None.
 204 
 205     Returns:
 206         List of CPU tensors (detached), or None if input is None.
 207     """
 208     if tensor_list is None:
 209         return None
 210     return [t.detach().cpu() for t in tensor_list]

 213 def tensors_to_gpu(tensor_list, device=None):
 214     """Move a list of CPU tensors back to GPU.
 215 
 216     Args:
 217         tensor_list: List of CPU tensors, or None.
 218         device: Target CUDA device. If None, uses current device.
 219 
 220     Returns:
 221         List of GPU tensors, or None if input is None.
 222     """
 223     if tensor_list is None:
 224         return None
 225     if device is None:
 226         device = accelerator.current_device()
 227     return [t.to(device=device, dtype=torch.float32) for t in tensor_list]
