 124 def _critic_output_layer_needs_reinit(args: Namespace, model: Sequence[DDP], role: str) -> bool:
 125     if role != "critic" or args.load is None:
 126         return False
 127 
 128     from megatron.core.dist_checkpointing.serialization import load_tensors_metadata
 129 
 130     checkpoint_path = Path(get_load_checkpoint_path_by_args(args))
 131     if not (checkpoint_path / ".metadata").is_file():
 132         return False
 133 
 134     checkpoint_metadata = load_tensors_metadata(str(checkpoint_path))
 135     for _chunk_id, output_layer in _iter_critic_output_layers(model):
 136         for name in ("weight", "bias"):
 137             param = getattr(output_layer, name, None)
 138             if param is None:
 139                 continue
 140 
 141             param_name = f"output_layer.{name}"
 142             ckpt_tensor_metadata = next(
 143                 (
 144                     tensor_metadata
 145                     for key, tensor_metadata in checkpoint_metadata.items()
 146                     if key == param_name or key.endswith(f".{param_name}")
 147                 ),
 148                 None,
 149             )
 150             expected_shape = tuple(param.shape)
 151             checkpoint_shape = tuple(ckpt_tensor_metadata.global_shape) if ckpt_tensor_metadata is not None else None
 152             if checkpoint_shape == expected_shape:
 153                 continue
 154 
 155             reason = (
 156                 "missing from checkpoint metadata"
 157                 if checkpoint_shape is None
 158                 else f"shape mismatch checkpoint={checkpoint_shape} runtime={expected_shape}"
 159             )
 160             logger.warning(
 161                 "Will reinitialize critic %s after checkpoint load because it is %s",
 162                 param_name,
 163                 reason,
 164             )
 165             return True
 166 
 167     return False

 181 def get_optimizer_param_scheduler(args: Namespace, optimizer: MegatronOptimizer) -> OptimizerParamScheduler:
 182     """Create and configure the optimizer learning-rate/weight-decay scheduler.
 183 
 184     This configures iteration-based schedules derived from the global batch size
 185     and run-time arguments.
 186 
 187     Args:
 188         args (Namespace): Training/runtime arguments (argparse namespace).
 189         optimizer (MegatronOptimizer): Megatron optimizer bound to the model.
 190 
 191     Returns:
 192         OptimizerParamScheduler: Initialized scheduler bound to ``optimizer``.
 193     """
 194     # Iteration-based training. ``train_iters`` is an estimate of the total
 195     # number of training steps — it's only used to size Megatron's LR decay
 196     # schedule (and ``lr_decay_iters`` defaults to it). With variable per-rollout
 197     # sample counts (dynamic sampling / filtering / custom step splitter) the
 198     # *actual* total can drift; the schedule still tracks the true progress via
 199     # ``opt_param_scheduler.num_steps`` (samples consumed, also persisted across
 200     # resume), so the worst case is the cosine/linear schedule reaches its
 201     # plateau slightly early or late. Pass ``--lr-decay-iters`` explicitly if you
 202     # need exact decay control.
 203     args.train_iters = args.num_rollout * args.rollout_batch_size * args.n_samples_per_prompt // args.global_batch_size
 204     if args.lr_decay_iters is None:
 205         args.lr_decay_iters = args.train_iters
 206     lr_decay_steps = args.lr_decay_iters * args.global_batch_size
 207     wd_incr_steps = args.train_iters * args.global_batch_size
 208     wsd_decay_steps = None
 209     if args.lr_wsd_decay_iters is not None:
 210         wsd_decay_steps = args.lr_wsd_decay_iters * args.global_batch_size
 211     if args.lr_warmup_fraction is not None:
 212         lr_warmup_steps = args.lr_warmup_fraction * lr_decay_steps
 213     else:
 214         lr_warmup_steps = args.lr_warmup_iters * args.global_batch_size
 215 
 216     opt_param_scheduler = OptimizerParamScheduler(
 217         optimizer,
 218         init_lr=args.lr_warmup_init,
 219         max_lr=args.lr,
 220         min_lr=args.min_lr,
 221         lr_warmup_steps=lr_warmup_steps,
 222         lr_decay_steps=lr_decay_steps,
 223         lr_decay_style=args.lr_decay_style,
 224         start_wd=args.start_weight_decay,
 225         end_wd=args.end_weight_decay,
 226         wd_incr_steps=wd_incr_steps,
 227         wd_incr_style=args.weight_decay_incr_style,
 228         use_checkpoint_opt_param_scheduler=args.use_checkpoint_opt_param_scheduler,
 229         override_opt_param_scheduler=args.override_opt_param_scheduler,
 230         wsd_decay_steps=wsd_decay_steps,
 231         lr_wsd_decay_style=args.lr_wsd_decay_style,
 232     )
 233 
 234     return opt_param_scheduler

 247 @contextmanager
 248 def _patch_megatron_adam(adam_cls):
 249     import megatron.core.optimizer as megatron_optimizer
 250     import megatron.core.optimizer.distrib_optimizer as megatron_distrib_optimizer
 251 
 252     missing = object()
 253     old_adam = megatron_optimizer.Adam
 254     old_cpu_adam = getattr(megatron_optimizer, "CPUAdam", missing)
 255     old_distrib_adam = megatron_distrib_optimizer.Adam
 256     try:
 257         megatron_optimizer.Adam = adam_cls
 258         if old_cpu_adam is not missing:
 259             megatron_optimizer.CPUAdam = adam_cls
 260         megatron_distrib_optimizer.Adam = adam_cls
 261         yield
 262     finally:
 263         megatron_optimizer.Adam = old_adam
 264         if old_cpu_adam is not missing:
 265             megatron_optimizer.CPUAdam = old_cpu_adam
 266         megatron_distrib_optimizer.Adam = old_distrib_adam

 269 def setup_model_and_optimizer(
 270     args: Namespace,
 271     role: str = "actor",
 272 ) -> tuple[list[DDP], MegatronOptimizer | None, OptimizerParamScheduler | None]:
 273     """Build model(s), wrap with DDP, and construct optimizer and scheduler.
 274 
 275     Args:
 276         args (Namespace): Training/runtime arguments (argparse namespace).
 277         role (str): Logical role of the model (e.g., "actor", "critic").
 278         no_wd_decay_cond (Callable[..., bool] | None): Predicate to exclude
 279             parameters from weight decay.
 280         scale_lr_cond (Callable[..., bool] | None): Predicate to scale LR for
 281             selected parameter groups.
 282         lr_mult (float): Global learning-rate multiplier for the optimizer.
 283 
 284     Returns:
 285         tuple[list[DDP], MegatronOptimizer | None, OptimizerParamScheduler | None]:
 286             - List of model chunks wrapped by ``DDP``.
 287             - The constructed ``MegatronOptimizer`` instance.
 288             - The learning-rate/weight-decay scheduler tied to the optimizer.
 289     """
 290     assert not args.moe_use_upcycling
 291     assert args.load is not None or args.pretrained_checkpoint is not None
 292 
 293     model = get_model(get_model_provider_func(args, role), ModelType.encoder_or_decoder)
 294 
 295     if args.num_rollout == 0:
 296         args.no_load_optim = True
 297         return model, None, None
 298 
 299     # Optimizer
 300     kwargs = {}
 301     for f in dataclasses.fields(OptimizerConfig):
 302         if hasattr(args, f.name):
 303             kwargs[f.name] = getattr(args, f.name)
 304     config = OptimizerConfig(**kwargs)
 305     config.timers = None
 306 
 307     if args.use_stateless_adam:
 308         assert config.optimizer == "adam", "Stateless Adam only supports --optimizer adam."
 309         assert args.no_save_optim, "Stateless Adam does not save Adam moment states. Please set --no-save-optim."
 310 
 311     optimizer_context = _patch_megatron_adam(StatelessAdam) if args.use_stateless_adam else nullcontext()
 312     with optimizer_context:
 313         optimizer = get_megatron_optimizer(
 314             config=config,
 315             model_chunks=model,
 316             use_gloo_process_groups=args.enable_gloo_process_groups,
 317         )
 318     if args.use_stateless_adam:
 319         _disable_distributed_optimizer_state_initialization(optimizer)
 320     opt_param_scheduler = get_optimizer_param_scheduler(args, optimizer)
 321     return model, optimizer, opt_param_scheduler

 347 @torch.no_grad()
 348 def forward_only(
 349     f: Callable[..., dict[str, list[torch.Tensor]]],
 350     args: Namespace,
 351     model: Sequence[DDP],
 352     data_iterator: Sequence[DataIterator],
 353     num_microbatches: Sequence[int],
 354     store_prefix: str = "",
 355     use_rollout_top_p_replay: bool = False,
 356 ) -> dict[str, list[torch.Tensor]]:
 357     """Run forward passes only and collect non-loss outputs (e.g., logprobs).
 358 
 359     The model is put into evaluation mode, a forward-only pipeline pass is
 360     executed, and relevant outputs are aggregated and returned.
 361 
 362     Args:
 363         f (Callable[..., dict[str, list[torch.Tensor]]]): Post-forward callback used to
 364             compute and package outputs to collect. This should accept a logits
 365             tensor as its first positional argument and additional keyword-only
 366             arguments; see ``get_log_probs_and_entropy``/``get_values`` in
 367             ``megatron_utils.loss`` for examples. It will be partially applied
 368             so that the callable returned from the internal forward step only
 369             requires the logits tensor.
 370         args (Namespace): Runtime arguments.
 371         model (Sequence[DDP]): Sequence of DDP-wrapped model chunks.
 372         data_iterator (Sequence[DataIterator]): Iterable(s) yielding batches for inference.
 373         num_microbatches (Sequence[int]): Number of microbatches per rollout step.
 374         store_prefix (str): Prefix to prepend to stored output keys.
 375         use_rollout_top_p_replay (bool): Whether to pass rollout top-p token sets
 376             to the post-forward log-prob callback when top-p rollout is enabled.
 377 
 378     Returns:
 379         dict[str, list[torch.Tensor]]: Aggregated outputs keyed by ``store_prefix + key``.
 380     """
 381 
 382     # reset data iterator
 383     for iterator in data_iterator:
 384         iterator.reset()
 385 
 386     config = get_model_config(model[0])
 387     batch_keys = [
 388         "tokens",
 389         "loss_masks",
 390         "multimodal_train_inputs",
 391         "total_lengths",
 392         "response_lengths",
 393     ]
 394     if use_rollout_top_p_replay:
 395         batch_keys = _with_rollout_top_p_token_keys(args, batch_keys)
 396 
 397     def forward_step(
 398         data_iterator: DataIterator, model: GPTModel, return_schedule_plan: bool = False
 399     ) -> tuple[torch.Tensor, Callable[[torch.Tensor], dict[str, list[torch.Tensor]]]]:
 400         """Forward step used by Megatron's pipeline engine.
 401 
 402         Args:
 403             data_iterator (DataIterator): Input data iterator.
 404             model (GPTModel): The GPT model chunk to execute.
 405 
 406         Returns:
 407             tuple[torch.Tensor, Callable[[torch.Tensor], dict[str, list[torch.Tensor]]]]:
 408             Output tensor(s) and a callable that computes and packages results
 409             to be collected by the engine.
 410         """
 411 
 412         assert not return_schedule_plan, "forward_only step should never return schedule plan"
 413 
 414         # Get the batch.
 415         batch = get_batch(
 416             data_iterator,
 417             batch_keys,
 418             args.data_pad_size_multiplier,
 419             args.allgather_cp,
 420         )
 421         unconcat_tokens = batch["unconcat_tokens"]
 422         tokens = batch["tokens"]
 423         packed_seq_params = batch["packed_seq_params"]
 424         total_lengths = batch["total_lengths"]
 425         response_lengths = batch["response_lengths"]
 426         forward_kwargs = {
 427             "input_ids": tokens,
 428             "position_ids": None,
 429             "attention_mask": None,
 430             "labels": None,
 431             "packed_seq_params": packed_seq_params,
 432             "loss_mask": batch["full_loss_masks"],
 433         }
 434         if batch["multimodal_train_inputs"] is not None:
 435             forward_kwargs.update(batch["multimodal_train_inputs"])
 436         output_tensor = model(**forward_kwargs)
 437 
 438         output_kwargs = {
 439             "args": args,
 440             "unconcat_tokens": unconcat_tokens,
 441             "total_lengths": total_lengths,
 442             "response_lengths": response_lengths,
 443             "with_entropy": args.use_rollout_entropy,
 444         }
 445         if use_rollout_top_p_replay:
 446             output_kwargs.update(get_rollout_top_p_logprob_kwargs(args, batch))
 447 
 448         return output_tensor, partial(f, **output_kwargs)
 449 
 450     # Turn on evaluation mode which disables dropout.
 451     for model_module in model:
 452         model_module.eval()
 453 
 454     if args.custom_megatron_before_log_prob_hook_path:
 455         from slime.utils.misc import load_function
 456 
 457         custom_before_log_prob_hook = load_function(args.custom_megatron_before_log_prob_hook_path)
 458         custom_before_log_prob_hook(args, model, store_prefix)
 459 
 460     forward_backward_func = get_forward_backward_func()
 461     # Don't care about timing during evaluation
 462     config.timers = None
 463     forward_data_store = []
 464     num_steps_per_rollout = len(num_microbatches)
 465     microbatch_pbar = tqdm(
 466         total=sum(num_microbatches),
 467         desc=f"{(store_prefix or getattr(model[0], 'role', 'actor')).rstrip('_')} forward",
 468         unit="microbatch",
 469         dynamic_ncols=True,
 470         leave=False,
 471         disable=_disable_tqdm_for_non_main_rank(),
 472     )
 473     forward_step_with_progress = _wrap_forward_step_with_microbatch_pbar(forward_step, microbatch_pbar)
 474     for step_id in range(num_steps_per_rollout):
 475         forward_data_store += forward_backward_func(
 476             forward_step_func=forward_step_with_progress,
 477             data_iterator=data_iterator,
 478             model=model,
 479             num_microbatches=num_microbatches[step_id],
 480             seq_length=args.seq_length,
 481             micro_batch_size=args.micro_batch_size,
 482             forward_only=True,
 483         )
 484     microbatch_pbar.close()
 485 
 486     # Move model back to the train mode.
 487     for model_module in model:
 488         model_module.train()
 489 
 490     rollout_data = {}
 491     # Store the results on the last stage
 492     if mpu.is_pipeline_last_stage():
 493         keys = forward_data_store[0].keys()
 494         for key in keys:
 495             values = []
 496             for value in forward_data_store:
 497                 assert isinstance(value[key], list)
 498                 values += value[key]
 499 
 500             if args.use_dynamic_batch_size:
 501                 # TODO: This is ugly... Find a better way to make the data have the same order.
 502                 # TODO: move this out of the loop.
 503                 origin_values = [None] * len(values)
 504                 origin_indices = sum(data_iterator[0].micro_batch_indices, [])
 505                 for value, origin_index in zip(values, origin_indices, strict=False):
 506                     origin_values[origin_index] = value
 507                 values = origin_values
 508             rollout_data[f"{store_prefix}{key}"] = values
 509     return rollout_data

 512 def train_one_step(
 513     args: Namespace,
 514     rollout_id: int,
 515     step_id: int,
 516     data_iterator: Sequence[DataIterator],
 517     model: Sequence[DDP],
 518     optimizer: MegatronOptimizer,
 519     opt_param_scheduler: OptimizerParamScheduler,
 520     num_microbatches: int,
 521     step_global_batch_size: int,
 522     microbatch_pbar=None,
 523 ) -> tuple[dict[str, float], float]:
 524     """Execute a single pipeline-parallel training step.
 525 
 526     Runs forward/backward over ``num_microbatches``, applies optimizer step and
 527     one scheduler step when gradients are valid.
 528 
 529     Args:
 530         args (Namespace): Runtime arguments.
 531         rollout_id (int): Rollout identifier.
 532         step_id (int): Step index within the current rollout.
 533         data_iterator (Sequence[DataIterator]): Iterable(s) yielding training batches.
 534         model (Sequence[DDP]): Sequence of DDP-wrapped model chunks.
 535         optimizer (MegatronOptimizer): Optimizer instance.
 536         opt_param_scheduler (OptimizerParamScheduler): LR/WD scheduler.
 537         num_microbatches (int): Number of microbatches to process.
 538         step_global_batch_size (int): Rollout count for this training step
 539             (total across DP; one "rollout" = one execution of one of the
 540             ``n_samples_per_prompt`` rollouts, which may emit >1 training
 541             sample under compact / subagent). Used both as the loss
 542             normalizer inside the closure and as the LR scheduler
 543             ``increment``. In the common case (1 rollout = 1 sample) this
 544             equals the per-step sample count, so behavior is unchanged.
 545 
 546     Returns:
 547         tuple[dict[str, float], float]: Reduced loss dictionary (last stage only)
 548         and gradient norm for logging.
 549     """
 550     args = get_args()
 551 
 552     # Set grad to zero.
 553     for model_chunk in model:
 554         model_chunk.zero_grad_buffer()
 555     optimizer.zero_grad()
 556 
 557     if args.custom_megatron_before_train_step_hook_path:
 558         from slime.utils.misc import load_function
 559 
 560         custom_before_train_step_hook = load_function(args.custom_megatron_before_train_step_hook_path)
 561         custom_before_train_step_hook(args, rollout_id, step_id, model, optimizer, opt_param_scheduler)
 562 
 563     def forward_step(data_iterator: DataIterator, model: GPTModel, return_schedule_plan: bool = False) -> tuple[
 564         torch.Tensor,
 565         Callable[[torch.Tensor], tuple[torch.Tensor, int, dict[str, torch.Tensor | list[str]]]],
 566     ]:
 567         """Forward step used by Megatron's pipeline engine during training.
 568 
 569         Args:
 570             data_iterator (DataIterator): Input data iterator.
 571             model (GPTModel): The GPT model chunk to execute.
 572 
 573         Returns:
 574             tuple[torch.Tensor, Callable[[torch.Tensor], tuple[torch.Tensor, int, dict[str, torch.Tensor | list[str]]]]]:
 575             Output tensor(s) and the loss function, which returns
 576             (loss, num_elems, {"keys": list[str], "values": torch.Tensor}).
 577         """
 578 
 579         # Get the batch.
 580         batch = get_batch(
 581             data_iterator,
 582             _with_rollout_top_p_token_keys(
 583                 args,
 584                 [
 585                     "tokens",
 586                     "multimodal_train_inputs",
 587                     "packed_seq_params",
 588                     "total_lengths",
 589                     "response_lengths",
 590                     "loss_masks",
 591                     "log_probs",
 592                     "ref_log_probs",
 593                     "values",
 594                     "advantages",
 595                     "returns",
 596                     "rollout_log_probs",
 597                     "teacher_log_probs",
 598                     "rollout_mask_sums",
 599                     # Only present when dumping train debug data; lets the loss
 600                     # snapshot each sample's log_probs keyed by rollout position.
 601                     *(["partition"] if args.save_debug_train_data is not None else []),
 602                 ],
 603             ),
 604             args.data_pad_size_multiplier,
 605             args.allgather_cp,
 606         )
 607 
 608         if os.environ.get("ENABLE_ROUTING_REPLAY", "0") == "1":
 609             old_stage = os.environ["ROUTING_REPLAY_STAGE"]
 610             os.environ["ROUTING_REPLAY_STAGE"] = "replay_forward"
 611 
 612         if return_schedule_plan:
 613             assert not args.enable_mtp_training, "MTP training should not be enabled when using combined 1f1b"
 614             position_ids = None
 615             output_tensor = model.build_schedule_plan(
 616                 input_ids=batch["tokens"],
 617                 position_ids=position_ids,
 618                 attention_mask=None,
 619                 labels=None,
 620                 packed_seq_params=batch["packed_seq_params"],
 621                 loss_mask=batch["full_loss_masks"],
 622             )
 623         else:
 624             forward_kwargs = {
 625                 "input_ids": batch["tokens"],
 626                 "position_ids": None,
 627                 "attention_mask": None,
 628                 "labels": None,
 629                 "packed_seq_params": batch["packed_seq_params"],
 630                 "loss_mask": batch["full_loss_masks"],
 631             }
 632 
 633             if batch["multimodal_train_inputs"] is not None:
 634                 forward_kwargs.update(batch["multimodal_train_inputs"])
 635 
 636             if args.enable_mtp_training:
 637                 forward_kwargs["mtp_kwargs"] = {"mtp_labels": batch["tokens"]}
 638 
 639             output_tensor = model(**forward_kwargs)
 640 
 641         if os.environ.get("ENABLE_ROUTING_REPLAY", "0") == "1":
 642             os.environ["ROUTING_REPLAY_STAGE"] = old_stage
 643 
 644         return output_tensor, partial(loss_function, args, batch, num_microbatches, step_global_batch_size)
 645 
 646     # Forward pass.
 647     forward_backward_func = get_forward_backward_func()
 648     losses_reduced = forward_backward_func(
 649         forward_step_func=_wrap_forward_step_with_microbatch_pbar(forward_step, microbatch_pbar),
 650         data_iterator=data_iterator,
 651         model=model,
 652         num_microbatches=num_microbatches,
 653         seq_length=args.seq_length,
 654         micro_batch_size=args.micro_batch_size,
 655         decoder_seq_length=args.decoder_seq_length,
 656         forward_only=False,
 657     )
 658 
 659     valid_step = True
 660     grad_norm = float("nan")
 661     if not getattr(args, "check_for_nan_in_loss_and_grad", True):
 662         found_inf_flag = optimizer.prepare_grads()
 663         if found_inf_flag:
 664             valid_step = False
 665         else:
 666             grad_norm = optimizer.get_grad_norm()
 667             if isinstance(grad_norm, torch.Tensor):
 668                 valid_step = not (torch.isnan(grad_norm) or torch.isinf(grad_norm))
 669             else:
 670                 valid_step = not (math.isnan(grad_norm) or math.isinf(grad_norm))
 671 
 672     # CI check: verify only MTP parameters have non-zero gradients when truncation happens
 673     # This check must happen before optimizer.step() as gradients may be modified during step
 674     if args.ci_test and args.enable_mtp_training:
 675         from slime.backends.megatron_utils.ci_utils import check_mtp_only_grad
 676 
 677         check_mtp_only_grad(model, step_id)
 678 
 679     if valid_step:
 680         # Update parameters.
 681         update_successful, grad_norm, num_zeros_in_grad = optimizer.step()
 682 
 683         # Update learning rate. Use the per-step global_batch_size when dynamic
 684         # batching is on so the scheduler's samples-seen counter tracks reality.
 685         assert update_successful
 686         opt_param_scheduler.step(increment=step_global_batch_size)
 687 
 688     # release grad
 689     for model_chunk in model:
 690         model_chunk.zero_grad_buffer()
 691     optimizer.zero_grad()
 692 
 693     if mpu.is_pipeline_last_stage(ignore_virtual=True):
 694         loss_reduced = train_metric_utils.reduce_train_step_metrics(
 695             losses_reduced,
 696             calculate_per_token_loss=args.calculate_per_token_loss,
 697             step_global_batch_size=step_global_batch_size,
 698             cp_size=mpu.get_context_parallel_world_size(),
 699             dp_with_cp_group=mpu.get_data_parallel_group(with_context_parallel=True),
 700         )
 701         return loss_reduced, grad_norm
 702     return {}, grad_norm

 706     """Block forward pre-hook for certain configurations."""
 707     return args.use_distributed_optimizer and args.overlap_param_gather

 710 def train(
 711     rollout_id: int,
 712     model: Sequence[DDP],
 713     optimizer: MegatronOptimizer,
 714     opt_param_scheduler: OptimizerParamScheduler,
 715     data_iterator: Sequence[DataIterator],
 716     num_microbatches: Sequence[int],
 717     global_batch_sizes: Sequence[int],
 718 ) -> None:
 719     """Run training over a rollout consisting of multiple steps.
 720 
 721     The model is switched to train mode, training hooks are configured, and
 722     ``train_one_step`` is invoked for each step in the rollout.
 723 
 724     Args:
 725         rollout_id (int): Rollout identifier.
 726         model (Sequence[DDP]): Sequence of DDP-wrapped model chunks.
 727         optimizer (MegatronOptimizer): Optimizer instance.
 728         opt_param_scheduler (OptimizerParamScheduler): LR/WD scheduler.
 729         data_iterator (Sequence[DataIterator]): Iterable(s) yielding training batches.
 730         num_microbatches (Sequence[int]): Microbatches per step in the rollout.
 731         global_batch_sizes (Sequence[int]): Rollout count per step (total
 732             across DP; one "rollout" = one execution of one of the
 733             ``n_samples_per_prompt`` rollouts of a prompt). Same length as
 734             ``num_microbatches``; consumed by ``train_one_step`` for loss
 735             scaling and LR scheduler increments. Equals per-step sample count
 736             in the common case (1 rollout = 1 sample).
 737     """
 738     args = get_args()
 739 
 740     assert len(num_microbatches) == len(global_batch_sizes), (
 741         f"num_microbatches and global_batch_sizes must have the same length, "
 742         f"got {len(num_microbatches)} vs {len(global_batch_sizes)}"
 743     )
 744 
 745     for iterator in data_iterator:
 746         iterator.reset()
 747 
 748     # Turn on training mode which enables dropout.
 749     for model_module in model:
 750         model_module.train()
 751 
 752     # Setup some training config params.
 753     config = get_model_config(model[0])
 754     config.grad_scale_func = optimizer.scale_loss
 755     config.timers = None
 756     if isinstance(model[0], DDP) and args.overlap_grad_reduce:
 757         assert config.no_sync_func is None, (
 758             "When overlap_grad_reduce is True, config.no_sync_func must be None; "
 759             "a custom no_sync_func is not supported when overlapping grad-reduce"
 760         )
 761         config.no_sync_func = [model_chunk.no_sync for model_chunk in model]
 762         if len(model) == 1:
 763             config.no_sync_func = config.no_sync_func[0]
 764         if args.align_grad_reduce:
 765             config.grad_sync_func = [model_chunk.start_grad_sync for model_chunk in model]
 766             if len(model) == 1:
 767                 config.grad_sync_func = config.grad_sync_func[0]
 768     if args.overlap_param_gather and args.align_param_gather:
 769         config.param_sync_func = [model_chunk.start_param_sync for model_chunk in model]
 770         if len(model) == 1:
 771             config.param_sync_func = config.param_sync_func[0]
 772     config.finalize_model_grads_func = finalize_model_grads
 773 
 774     pre_hook_enabled = False
 775 
 776     if args.reset_optimizer_states:
 777         if (
 778             mpu.get_data_parallel_rank(with_context_parallel=True) == 0
 779             and mpu.get_tensor_model_parallel_rank() == 0
 780             and mpu.get_pipeline_model_parallel_rank() == mpu.get_pipeline_model_parallel_world_size() - 1
 781         ):
 782             logger.info("Reset optimizer states")
 783         for chained_optimizer in optimizer.chained_optimizers:
 784             for group in chained_optimizer.optimizer.param_groups:
 785                 if "step" in group:
 786                     group["step"] = 0
 787             for state in chained_optimizer.optimizer.state.values():
 788                 if "step" in state:
 789                     if isinstance(state["step"], torch.Tensor):
 790                         state["step"].zero_()
 791                     else:
 792                         state["step"] = 0
 793                 if "exp_avg" in state:
 794                     state["exp_avg"].zero_()
 795                 if "exp_avg_sq" in state:
 796                     state["exp_avg_sq"].zero_()
 797 
 798     if args.manual_gc:
 799         # Disable the default garbage collector and perform the collection manually.
 800         # This is to align the timing of garbage collection across ranks.
 801         assert args.manual_gc_interval >= 0, "Manual garbage collection interval should be larger than or equal to 0"
 802         gc.disable()
 803         gc.collect()
 804 
 805     # Disable forward pre-hook to start training to ensure that errors in checkpoint loading
 806     # or random initialization don't propagate to all ranks in first all-gather (which is a
 807     # no-op if things work correctly).
 808     if should_disable_forward_pre_hook(args):
 809         disable_forward_pre_hook(model, param_sync=False)
 810         # Also remove param_sync_func temporarily so that sync calls made in
 811         # `forward_backward_func` are no-ops.
 812         param_sync_func = config.param_sync_func
 813         config.param_sync_func = None
 814         pre_hook_enabled = False
 815 
 816     num_steps_per_rollout = len(num_microbatches)
 817     microbatch_pbar = tqdm(
 818         total=sum(num_microbatches),
 819         desc=f"{getattr(model[0], 'role', 'actor')} train",
 820         unit="microbatch",
 821         dynamic_ncols=True,
 822         leave=False,
 823         disable=_disable_tqdm_for_non_main_rank(),
 824     )
 825 
 826     # Run training iterations till done.
 827     for step_id in range(num_steps_per_rollout):
 828 
 829         # Run training step.
 830         loss_dict, grad_norm = train_one_step(
 831             args,
 832             rollout_id,
 833             step_id,
 834             data_iterator,
 835             model,
 836             optimizer,
 837             opt_param_scheduler,
 838             num_microbatches[step_id],
 839             global_batch_sizes[step_id],
 840             microbatch_pbar=microbatch_pbar,
 841         )
 842 
 843         if step_id == 0:
 844             # Enable forward pre-hook after training step has successfully run. All subsequent
 845             # forward passes will use the forward pre-hook / `param_sync_func` in
 846             # `forward_backward_func`.
 847             if should_disable_forward_pre_hook(args):
 848                 enable_forward_pre_hook(model)
 849                 config.param_sync_func = param_sync_func
 850                 pre_hook_enabled = True
 851 
 852         if args.enable_mtp_training:
 853             from megatron.core.transformer.multi_token_prediction import MTPLossLoggingHelper
 854 
 855             mtp_loss_scale = 1 / num_microbatches[step_id]
 856             tracker = MTPLossLoggingHelper.tracker
 857             if "values" in tracker:
 858                 values = tracker["values"]
 859                 if tracker.get("reduce_group") is not None:
 860                     torch.distributed.all_reduce(values, group=tracker.get("reduce_group"))
 861                 if tracker.get("avg_group") is not None:
 862                     torch.distributed.all_reduce(values, group=tracker["avg_group"], op=torch.distributed.ReduceOp.AVG)
 863                 # Multi-head MTP: tracker["values"] is [num_mtp_layers]; aggregate below.
 864                 mtp_losses = tracker["values"] * mtp_loss_scale
 865                 MTPLossLoggingHelper.clean_loss_in_tracker()
 866 
 867                 # CI check: verify MTP loss is within expected bounds
 868                 if args.ci_test:
 869                     from slime.backends.megatron_utils.ci_utils import check_mtp_loss
 870 
 871                     check_mtp_loss(mtp_losses.sum().item())
 872 
 873         # per train step log.
 874         if (
 875             mpu.get_data_parallel_rank(with_context_parallel=True) == 0
 876             and mpu.get_tensor_model_parallel_rank() == 0
 877             and mpu.get_pipeline_model_parallel_rank() == mpu.get_pipeline_model_parallel_world_size() - 1
 878         ):
 879             accumulated_step_id = rollout_id * num_steps_per_rollout + step_id
 880             role = getattr(model[0], "role", "actor")
 881             role_tag = "" if role == "actor" else f"{role}-"
 882             log_dict = {
 883                 f"train/{role_tag}{key}": val.mean().item() if isinstance(val, torch.Tensor) else val
 884                 for key, val in loss_dict.items()
 885             }
 886             log_dict[f"train/{role_tag}grad_norm"] = grad_norm
 887             if args.enable_mtp_training:
 888                 for _i in range(mtp_losses.shape[0]):
 889                     log_dict[f"train/{role_tag}mtp_{_i + 1}_loss"] = mtp_losses[_i].item()
 890                 log_dict[f"train/{role_tag}mtp_loss"] = mtp_losses.sum().item()
 891 
 892             for param_group_id, param_group in enumerate(optimizer.param_groups):
 893                 log_dict[f"train/{role_tag}lr-pg_{param_group_id}"] = opt_param_scheduler.get_lr(param_group)
 894 
 895             # Per-step gbs — uneven step sizes are easy to miss without this.
 896             log_dict[f"train/{role_tag}global_batch_size"] = global_batch_sizes[step_id]
 897             log_dict["train/step"] = accumulated_step_id
 898             logging_utils.log(args, log_dict, step_key="train/step")
 899 
 900             if args.ci_test and "train/train_rollout_logprob_abs_diff" in log_dict:
 901                 threshold = args.ci_train_rollout_logprob_abs_diff_threshold
 902                 assert log_dict["train/train_rollout_logprob_abs_diff"] <= threshold, f"{threshold=} {log_dict=}"
 903 
 904             if args.ci_test and not args.ci_disable_kl_checker:
 905                 if step_id == 0 and "train/ppo_kl" in log_dict and "train/pg_clipfrac" in log_dict:
 906                     # TODO: figure out why KL is not exactly zero when using PPO loss with KL clipping, and whether this is expected behavior or a bug.
 907                     assert log_dict["train/ppo_kl"] < 1e-8, f"{log_dict=}"
 908                 # R3 replays rollout routing for the actor path, while ref
 909                 # log-probs are computed with normal routing. The initial
 910                 # actor/ref KL is therefore not expected to be exactly zero.
 911                 if (
 912                     accumulated_step_id == 0
 913                     and not getattr(args, "use_rollout_routing_replay", False)
 914                     and "train/kl_loss" in log_dict
 915                 ):
 916                     assert log_dict["train/kl_loss"] < 1e-8, f"{log_dict=}"
 917 
 918             logger.info(f"{role_tag}step {accumulated_step_id}: {log_dict}")
 919 
 920             if args.ci_save_grad_norm is not None:
 921                 ci_save_grad_norm_path = args.ci_save_grad_norm.format(
 922                     role=role,
 923                     rollout_id=rollout_id,
 924                     step_id=step_id,
 925                 )
 926                 torch.save(grad_norm, ci_save_grad_norm_path)
 927             elif args.ci_load_grad_norm is not None:
 928                 ci_load_grad_norm_path = args.ci_load_grad_norm.format(
 929                     role=role,
 930                     rollout_id=rollout_id,
 931                     step_id=step_id,
 932                 )
 933                 expected_grad_norm = torch.load(ci_load_grad_norm_path)
 934                 assert math.isclose(
 935                     grad_norm,
 936                     expected_grad_norm,
 937                     rel_tol=0.01,
 938                     abs_tol=0.01,
 939                 ), f"grad norm mismatch: {grad_norm} != {expected_grad_norm}"
 940     microbatch_pbar.close()
 941     # Close out pre-hooks if using distributed optimizer and overlapped param gather.
 942     if pre_hook_enabled:
 943         disable_forward_pre_hook(model)

 947     iteration: int,
 948     model: Sequence[DDP],
 949     optimizer: MegatronOptimizer,
 950     opt_param_scheduler: OptimizerParamScheduler,
 951 ) -> None:
 952     """Persist a training checkpoint safely with forward hooks disabled.

 954     Args:

 956         model (Sequence[DDP]): Sequence of DDP-wrapped model chunks.
 957         optimizer (MegatronOptimizer): Optimizer instance.

 961     if should_disable_forward_pre_hook(args):
 962         disable_forward_pre_hook(model)
 963     save_checkpoint(
 964         iteration,
 965         model,
 966         optimizer,
 967         opt_param_scheduler,
 968         num_floating_point_operations_so_far=0,
 969         checkpointing_context=None,
 970         train_data_iterator=None,
 971         preprocess_common_state_dict_fn=None,
 972     )
 973     if should_disable_forward_pre_hook(args):
 974         enable_forward_pre_hook(model)

 977 def initialize_model_and_optimizer(
 978     args: Namespace, role: str = "actor"
 979 ) -> tuple[list[DDP], MegatronOptimizer | None, OptimizerParamScheduler | None, int]:
 980     """Initialize model(s), optimizer, scheduler, and load from checkpoint.
 981 
 982     Args:
 983         args (Namespace): Runtime arguments.
 984         role (str): Logical role of the model (e.g., "actor", "critic").
 985 
 986     Returns:
 987         tuple[list[DDP], MegatronOptimizer | None, OptimizerParamScheduler | None, int]:
 988             DDP-wrapped model chunks, optimizer, scheduler, and iteration index.
 989     """
 990 
 991     if torch.version.hip:
 992         import megatron.core.dist_checkpointing.strategies.filesystem_async as filesystem_async_module
 993 
 994         from slime.utils.rocm_checkpoint_writer import ROCmFileSystemWriterAsync
 995 
 996         filesystem_async_module.FileSystemWriterAsync = ROCmFileSystemWriterAsync
 997         print("[ROCm] Applied FileSystemWriterAsync patch for HIP compatibility")
 998 
 999     model, optimizer, opt_param_scheduler = setup_model_and_optimizer(args, role)
1000     model[0].role = role
1001     reinit_critic_output_layer = _critic_output_layer_needs_reinit(args, model, role)
1002     clear_memory()
1003     iteration, _ = load_checkpoint(
1004         model,
1005         optimizer,
1006         opt_param_scheduler,
1007         checkpointing_context={},
1008     )
1009     if reinit_critic_output_layer:
1010         _reinitialize_critic_output_layer(args, model)
1011         if (args.fp16 or args.bf16) and optimizer is not None:
1012             optimizer.reload_model_params()
1013     clear_memory()
1014 
1015     return model, optimizer, opt_param_scheduler, iteration
