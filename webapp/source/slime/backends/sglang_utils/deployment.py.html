  18 def _start_router(args, *, has_pd_disaggregation: bool = False, force_new: bool = False) -> tuple[str, int]:
  19     """Start sglang_router and return (router_ip, router_port)."""
  20     if not force_new and args.sglang_router_ip is not None:
  21         return args.sglang_router_ip, args.sglang_router_port
  22 
  23     router_ip = _wrap_ipv6(get_host_info()[1])
  24     if force_new:
  25         router_port = find_available_port(random.randint(3000, 4000))
  26     else:
  27         router_port = args.sglang_router_port
  28         if router_port is None:
  29             router_port = find_available_port(random.randint(3000, 4000))
  30 
  31     from sglang_router.launch_router import RouterArgs
  32 
  33     from slime.utils.http_utils import run_router
  34 
  35     router_args = RouterArgs.from_cli_args(args, use_router_prefix=True)
  36     router_args.host = router_ip
  37     router_args.port = router_port
  38     router_args.prometheus_port = find_available_port(random.randint(4000, 5000))
  39     router_args.request_timeout_secs = args.sglang_router_request_timeout_secs
  40 
  41     if has_pd_disaggregation:
  42         router_args.pd_disaggregation = True
  43 
  44     # RDMA transfer timeouts are transient and should not mark decode workers dead.
  45     router_args.disable_circuit_breaker = True
  46 
  47     # RolloutHealthMonitor owns engine health checks.
  48     if hasattr(router_args, "disable_health_check"):
  49         router_args.disable_health_check = True
  50 
  51     logger.info(f"Launch router with args: {router_args}")
  52 
  53     process = multiprocessing.Process(
  54         target=run_router,
  55         args=(router_args,),
  56     )
  57     process.daemon = True
  58     process.start()
  59     time.sleep(3)
  60     assert process.is_alive()
  61     logger.info(f"Router launched at {router_ip}:{router_port}, Prometheus port: {router_args.prometheus_port}")
  62     return router_ip, router_port

  65 def _compute_rollout_offset(args) -> int:
  66     """Offset (in placement-group bundle slots) where rollout GPUs start."""
  67     if args.debug_train_only or args.debug_rollout_only or args.colocate:

  79 def start_rollout_servers(args, pg) -> tuple[dict[str, Any], list[Any]]:
  80     """Start configured rollout servers without waiting for final engine initialization."""
  81     if args.rollout_external:
  82         return start_external_rollout_servers(args, start_router=_start_router)
  83 
  84     config = resolve_sglang_config(args)
  85     placement = ServerGroupPlacement(
  86         args=args,
  87         pg=pg,
  88         rollout_pg_offset=_compute_rollout_offset(args),
  89         megatron_num_gpus=_compute_megatron_num_gpus(args),
  90     )
  91 
  92     servers: dict[str, RolloutServer] = {}
  93     pending_init_handles: list[Any] = []
  94 
  95     for model_idx, model_config in enumerate(config.models):
  96         model_config.resolve(args)
  97 
  98         router_ip, router_port = _start_router(
  99             args,
 100             has_pd_disaggregation=model_config.has_pd_disaggregation,
 101             force_new=(model_idx > 0),
 102         )
 103 
 104         if model_idx == 0:
 105             args.sglang_router_ip = router_ip
 106             args.sglang_router_port = router_port
 107 
 108         if model_config.has_encoder_disaggregation:
 109             server_groups, init_handles = start_epd_server_groups(
 110                 model_config,
 111                 placement,
 112                 router_ip,
 113                 router_port,
 114             )
 115         elif model_config.has_pd_disaggregation:
 116             server_groups, init_handles = start_pd_server_groups(
 117                 model_config,
 118                 placement,
 119                 router_ip,
 120                 router_port,
 121             )
 122         else:
 123             server_groups = []
 124             init_handles = []
 125             port_cursors: dict[int, int] = {}
 126             for group_config in model_config.server_groups:
 127                 group = placement.create(group_config, router_ip, router_port)
 128                 handles, port_cursors = group.start_engines(port_cursors)
 129                 init_handles.extend(handles)
 130                 server_groups.append(group)
 131 
 132         pending_init_handles.extend(init_handles)
 133         servers[model_config.name] = RolloutServer(
 134             server_groups=server_groups,
 135             router_ip=router_ip,
 136             router_port=router_port,
 137             model_name=model_config.name,
 138             update_weights=model_config.update_weights,
 139         )
 140 
 141     args.sglang_model_routers = {name: (server.router_ip, server.router_port) for name, server in servers.items()}
 142     return servers, pending_init_handles
