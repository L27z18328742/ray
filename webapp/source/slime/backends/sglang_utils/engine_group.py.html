  20 class ServerGroup:
  21     """A group of homogeneous SGLang engines with the same configuration.
  22 
  23     All engines in a group share the same tp_size / nodes_per_engine / pg.
  24     A RolloutServer may contain multiple ServerGroups (e.g. prefill vs decode
  25     in PD disaggregation).
  26     """

  29     pg: Any  # (placement_group, reordered_bundle_indices, reordered_gpu_ids)
  30     all_engines: list
  31     num_gpus_per_engine: int
  32     num_new_engines: int
  33     worker_type: str = "regular"  # "regular", "prefill", "decode", or "placeholder"
  34     rank_offset: int = 0  # cumulative engine count before this group
  35     gpu_offset: int = 0  # cumulative GPU count before this group

  51     def parallel_config(self) -> dict[str, Any]:
  52         """Return the SGLang parallel args that affect rank-local expert routing."""
  53         overrides = {key.replace("-", "_"): value for key, value in self.sglang_overrides.items()}
  54         pp_size = int(overrides.get("pp_size", getattr(self.args, "sglang_pp_size", 1)))
  55         tp_size = int(overrides.get("tp_size", self.num_gpus_per_engine // pp_size))
  56         return {
  57             "tp_size": tp_size,
  58             "pp_size": pp_size,

  63     def start_engines(self, port_cursors: dict[int, int] | None = None) -> tuple[list, dict[int, int]]:
  64         """Create Ray actors, allocate ports, and fire ``engine.init()`` without waiting.
  65 
  66         Returns ``(init_handles, port_cursors)`` where *init_handles* is a list
  67         of Ray ObjectRefs and *port_cursors* maps node index → next free port.
  68         The caller should ``ray.get()`` on the handles to block until the
  69         engines are healthy, and pass *port_cursors* to the next server group
  70         so that different groups on the same node don't race for ports.
  71 
  72         Placeholder groups (worker_type="placeholder") skip engine creation entirely.
  73         """
  74         if port_cursors is None:
  75             port_cursors = {}
  76         if self.args.debug_train_only or self.worker_type == "placeholder":
  77             self.num_new_engines = 0
  78             return [], port_cursors
  79 
  80         num_gpus_per_engine_on_node = min(self.num_gpus_per_engine, self.args.num_gpus_per_node)
  81 
  82         pg, reordered_bundle_indices, reordered_gpu_ids = self.pg
  83         num_engines = len(self.all_engines)
  84         required_gpu_slots = self.gpu_offset + num_engines * num_gpus_per_engine_on_node
  85         if num_engines and not (
  86             self.gpu_offset >= 0 and num_gpus_per_engine_on_node > 0 and required_gpu_slots <= len(reordered_gpu_ids)
  87         ):
  88             raise ValueError(
  89                 "Invalid rollout server group GPU placement: "
  90                 f"worker_type={self.worker_type}, "
  91                 f"gpu_offset={self.gpu_offset}, "
  92                 f"num_gpus_per_engine={self.num_gpus_per_engine}, "
  93                 f"num_gpus_per_engine_on_node={num_gpus_per_engine_on_node}, "
  94                 f"num_engines={num_engines}, "
  95                 f"required_gpu_slots={required_gpu_slots}, "
  96                 f"len(reordered_gpu_ids)={len(reordered_gpu_ids)}, "
  97                 f"rollout_num_gpus={self.args.rollout_num_gpus}, "
  98                 f"rollout_num_gpus_per_engine={self.args.rollout_num_gpus_per_engine}. "
  99                 "Please align --rollout-num-gpus, --rollout-num-gpus-per-engine, "
 100                 "and --sglang-config server_groups."
 101             )
 102 
 103         RolloutRayActor = ray.remote(SGLangEngine)
 104 
 105         rollout_engines = []
 106         for i in range(len(self.all_engines)):
 107             if self.all_engines[i] is not None:
 108                 continue
 109 
 110             global_rank = self.rank_offset + i
 111             num_gpus = 0.2
 112             num_cpus = num_gpus
 113 
 114             # Get the base GPU ID from placement group using gpu_offset.
 115             gpu_index = self.gpu_offset + i * num_gpus_per_engine_on_node
 116             base_gpu_id = int(reordered_gpu_ids[gpu_index])
 117 
 118             scheduling_strategy = PlacementGroupSchedulingStrategy(
 119                 placement_group=pg,
 120                 placement_group_capture_child_tasks=True,
 121                 placement_group_bundle_index=reordered_bundle_indices[gpu_index],
 122             )
 123 
 124             env_vars = {name: "1" for name in NOSET_VISIBLE_DEVICES_ENV_VARS_LIST} | {
 125                 key: os.environ.get(key, default_val)
 126                 for key, default_val in {
 127                     "SGLANG_JIT_DEEPGEMM_PRECOMPILE": "false",
 128                     "SGLANG_JIT_DEEPGEMM_FAST_WARMUP": "true",
 129                     "SGL_DISABLE_TP_MEMORY_INBALANCE_CHECK": "true",
 130                     "SGLANG_DISABLE_TP_MEMORY_INBALANCE_CHECK": "true",
 131                     "SGLANG_MEMORY_SAVER_CUDA_GRAPH": "true",
 132                     "SGLANG_BATCH_INVARIANT_OPS_ENABLE_MM_FALLBACK_VARIANT": "true",
 133                     "SGLANG_ENABLE_HEALTH_ENDPOINT_GENERATION": "false",
 134                     "SGLANG_ENABLE_STRICT_MEM_CHECK_DURING_IDLE": "false",
 135                 }.items()
 136             }
 137             rollout_engine = RolloutRayActor.options(
 138                 num_cpus=num_cpus,
 139                 num_gpus=num_gpus,
 140                 scheduling_strategy=scheduling_strategy,
 141                 runtime_env={
 142                     "env_vars": add_default_ray_env_vars(env_vars),
 143                 },
 144             ).remote(
 145                 self.args,
 146                 rank=global_rank,
 147                 worker_type=self.worker_type,
 148                 base_gpu_id=base_gpu_id,
 149                 sglang_overrides=self.sglang_overrides,
 150                 num_gpus_per_engine=self.num_gpus_per_engine,
 151             )
 152 
 153             rollout_engines.append((global_rank, rollout_engine))
 154             self.all_engines[i] = rollout_engine
 155 
 156         self.num_new_engines = len(rollout_engines)
 157 
 158         if self.num_new_engines == 0:
 159             return [], port_cursors
 160 
 161         # Compute base_port from the maximum cursor across all nodes that
 162         # this group's engines may land on (conservative: just use global max).
 163         base_port = max(port_cursors.values()) if port_cursors else 15000
 164         addr_and_ports, port_cursors = _allocate_rollout_engine_addr_and_ports_normal(
 165             args=self.args,
 166             rollout_engines=rollout_engines,
 167             worker_type=self.worker_type,
 168             num_gpus_per_engine=self.num_gpus_per_engine,
 169             rank_offset=self.rank_offset,
 170             base_port=base_port,
 171         )
 172 
 173         init_handles = [
 174             engine.init.remote(
 175                 **(addr_and_ports[rank]),
 176                 router_ip=self.router_ip,
 177                 router_port=self.router_port,
 178             )
 179             for rank, engine in rollout_engines
 180         ]
 181         return init_handles, port_cursors
 182 
 183     def offload(self):
 184         """Fire release_memory_occupation on all engines (non-blocking).
 185 
 186         Returns a list of Ray ObjectRefs.  Skipped for groups that do not
 187         overlap with megatron GPUs (``needs_offload=False``).
 188         """
 189         if not self.needs_offload:
 190             return []
 191         return [engine.release_memory_occupation.remote() for engine in self.engines if engine is not None]
 192 
 193     def onload(self, tags: list[str] | None = None):
 194         """Fire resume_memory_occupation on all engines (non-blocking).
 195 
 196         Returns a list of Ray ObjectRefs.  Skipped for groups that do not
 197         overlap with megatron GPUs (``needs_offload=False``).
 198         """
 199         if not self.needs_offload:
 200             return []
 201         return [engine.resume_memory_occupation.remote(tags=tags) for engine in self.engines if engine is not None]

 217         return [e for g in self.server_groups for e in g.engines]

 222         return [e for g in self.server_groups for e in g.all_engines]

 233     @property
 234     def engine_gpu_counts(self) -> list[int]:
 235         """Per-engine GPU count for all node-0 engines, parallel to ``engines``."""
 236         return [g.num_gpus_per_engine for g in self.server_groups for _ in g.engines]
 237 
 238     @property
 239     def engine_gpu_offsets(self) -> list[int]:
 240         """Per-engine GPU offset for all node-0 engines, parallel to ``engines``.
 241 
 242         Accounts for placeholder groups that occupy GPU slots without creating engines.
 243         """
 244         offsets = []
 245         for g in self.server_groups:
 246             for j in range(len(g.engines)):
 247                 offsets.append(g.gpu_offset + j * g.num_gpus_per_engine)
 248         return offsets
 249 
 250     @property
 251     def engine_parallel_configs(self) -> list[dict[str, Any]]:
 252         """Per-engine SGLang parallel config, parallel to ``engines``."""
 253         return [g.parallel_config() for g in self.server_groups for _ in g.engines]
 254 
 255     @property
 256     def nodes_per_engine(self):
 257         """Nodes per engine.  Only valid when all active groups share the same value."""
 258         values = {g.nodes_per_engine for g in self.server_groups if g.worker_type != "placeholder"}
 259         if len(values) != 1:
 260             raise ValueError(f"Heterogeneous nodes_per_engine across groups: {values}")
 261         return values.pop()

 263     def recover(self):
 264         """Recover dead engines across all active groups, overlapping init."""
 265         dead_per_group = [[i for i, engine in enumerate(g.all_engines) if engine is None] for g in self.server_groups]
 266 
 267         all_handles = []
 268         port_cursors: dict[int, int] = {}
 269         for g in self.server_groups:
 270             handles, port_cursors = g.start_engines(port_cursors)
 271             all_handles.extend(handles)
 272         if all_handles:
 273             ray.get(all_handles)
 274 
 275         release_handles = []
 276         updatable_new_engines = []
 277         non_updatable_groups_engines: list[tuple[str, list]] = []
 278         for g, dead_indices in zip(self.server_groups, dead_per_group, strict=True):
 279             logger.info(f"Recovered {g.num_new_engines} dead rollout engines (worker_type={g.worker_type})")
 280             assert g.num_new_engines == len(dead_indices), "num_new_engines does not match dead_indices length"
 281             if g.needs_offload and dead_indices:
 282                 new_engines = [g.all_engines[i] for i in dead_indices]
 283                 release_handles.extend(engine.release_memory_occupation.remote() for engine in new_engines)
 284                 if self.update_weights:
 285                     updatable_new_engines.extend(new_engines)
 286                 elif g.model_path:
 287                     non_updatable_groups_engines.append((g.model_path, new_engines))
 288 
 289         if release_handles:
 290             ray.get(release_handles)
 291             all_resume_engines = updatable_new_engines[:]
 292             for _model_path, engines in non_updatable_groups_engines:
 293                 all_resume_engines.extend(engines)
 294             if all_resume_engines:
 295                 ray.get(
 296                     [
 297                         engine.resume_memory_occupation.remote(tags=[GPU_MEMORY_TYPE_WEIGHTS])
 298                         for engine in all_resume_engines
 299                     ]
 300                 )
 301 
 302     def offload(self):
 303         """Release memory occupation across all groups (concurrent)."""
 304         handles = []
 305         for g in self.server_groups:
 306             handles.extend(g.offload())
 307         return ray.get(handles) if handles else []
 308 
 309     def onload(self, tags: list[str] | None = None):
 310         """Resume memory occupation across all groups (concurrent)."""
 311         handles = []
 312         for g in self.server_groups:
 313             handles.extend(g.onload(tags))
 314         return ray.get(handles) if handles else []
 315 
 316     def onload_weights(self):
 317         """Restore weights for offloaded groups."""
 318         handles = []
 319         for g in self.server_groups:
 320             if not g.needs_offload:
 321                 continue
 322             handles.extend(g.onload(tags=[GPU_MEMORY_TYPE_WEIGHTS]))
 323         return ray.get(handles) if handles else []
 324 
 325     def onload_kv(self):
 326         """Resume KV cache and CUDA graphs for offloaded groups."""
 327         handles = []
 328         for g in self.server_groups:
 329             handles.extend(g.onload(tags=[GPU_MEMORY_TYPE_KV_CACHE, GPU_MEMORY_TYPE_CUDA_GRAPH]))
 330         return ray.get(handles) if handles else []

 334 class ServerGroupPlacement:
 335     """Track rank and GPU offsets while materializing configured server groups."""
 336 
 337     args: Any
 338     pg: Any
 339     rollout_pg_offset: int
 340     megatron_num_gpus: int
 341     engine_offset: int = 0
 342     gpu_offset: int = 0
 343 
 344     def create(
 345         self,
 346         group_config: ServerGroupConfig,
 347         router_ip: str,
 348         router_port: int,
 349         *,
 350         overrides_extra: dict | None = None,
 351     ) -> ServerGroup:
 352         gpus_per_engine = group_config.num_gpus_per_engine
 353         assert gpus_per_engine is not None, "ModelConfig.resolve() must set num_gpus_per_engine before deployment"
 354         num_gpus_per_engine_on_node = min(gpus_per_engine, self.args.num_gpus_per_node)
 355         num_engines = group_config.num_gpus // num_gpus_per_engine_on_node
 356 
 357         group_abs_start = self.rollout_pg_offset + self.gpu_offset
 358         needs_offload = self.args.offload_rollout and group_abs_start < self.megatron_num_gpus
 359         overrides = dict(group_config.overrides)
 360         if overrides_extra:
 361             for key, value in overrides_extra.items():
 362                 overrides.setdefault(key, value)
 363         if self.args.offload_rollout and not needs_offload:
 364             overrides.setdefault("enable_memory_saver", False)
 365         logger.info(
 366             f"Engine group '{group_config.worker_type}' gpu_offset={self.gpu_offset} "
 367             f"(abs={group_abs_start}): needs_offload={needs_offload}"
 368         )
 369 
 370         group = ServerGroup(
 371             args=self.args,
 372             pg=self.pg,
 373             all_engines=[None] * num_engines if group_config.worker_type != "placeholder" else [],
 374             num_gpus_per_engine=gpus_per_engine,
 375             num_new_engines=0,
 376             worker_type=group_config.worker_type,
 377             rank_offset=self.engine_offset,
 378             gpu_offset=self.gpu_offset,
 379             sglang_overrides=overrides,
 380             needs_offload=needs_offload,
 381             model_path=overrides.get("model_path", self.args.hf_checkpoint),
 382             router_ip=router_ip,
 383             router_port=router_port,
 384         )
 385         self.engine_offset += num_engines
 386         self.gpu_offset += group_config.num_gpus
 387         return group

 390 def _allocate_rollout_engine_addr_and_ports_normal(
 391     *,
 392     args,
 393     rollout_engines,
 394     worker_type="regular",
 395     num_gpus_per_engine=None,
 396     rank_offset=0,
 397     base_port=15000,
 398 ):
 399     """Allocate rank-local SGLang and distributed-init ports for one group."""
 400     gpus_per_engine = num_gpus_per_engine or args.rollout_num_gpus_per_engine
 401     num_engines_per_node = max(1, args.num_gpus_per_node // gpus_per_engine)
 402     addr_and_ports: dict[int, dict] = {}
 403     node_port_cursor: dict[int, int] = {}
 404 
 405     visited_nodes = set()
 406     for rank, engine in rollout_engines:
 407         local_rank = rank - rank_offset
 408         node_index = local_rank // num_engines_per_node
 409         if node_index in visited_nodes:
 410             continue
 411         visited_nodes.add(node_index)
 412         num_engines_on_this_node = num_engines_per_node - (local_rank % num_engines_per_node)
 413 
 414         def get_addr_and_ports(engine, node_idx):
 415             start_port = node_port_cursor.get(node_idx, base_port)
 416 
 417             def port(consecutive=1):
 418                 nonlocal start_port
 419                 _, allocated_port = ray.get(
 420                     engine._get_current_node_ip_and_free_port.remote(
 421                         start_port=start_port,
 422                         consecutive=consecutive,
 423                     )
 424                 )
 425                 start_port = allocated_port + consecutive
 426                 node_port_cursor[node_idx] = start_port
 427                 return allocated_port
 428 
 429             def addr():
 430                 address, _ = ray.get(engine._get_current_node_ip_and_free_port.remote())
 431                 return address
 432 
 433             return addr, port
 434 
 435         get_addr, get_port = get_addr_and_ports(engine, node_index)
 436 
 437         for i in range(num_engines_on_this_node):
 438             current_rank = rank + i
 439             addr_and_ports.setdefault(current_rank, {})
 440             addr_and_ports[current_rank]["host"] = get_addr()
 441             addr_and_ports[current_rank]["port"] = get_port()
 442             addr_and_ports[current_rank]["nccl_port"] = get_port()
 443 
 444             if worker_type == "prefill":
 445                 addr_and_ports[current_rank]["disaggregation_bootstrap_port"] = get_port()
 446 
 447         if gpus_per_engine > args.num_gpus_per_node:
 448             num_node_per_engine = gpus_per_engine // args.num_gpus_per_node
 449             if local_rank % num_node_per_engine == 0:
 450                 dist_init_addr = f"{get_addr()}:{get_port(30 + args.sglang_dp_size)}"
 451                 for i in range(num_node_per_engine):
 452                     addr_and_ports.setdefault(rank + i, {})
 453                     addr_and_ports[rank + i]["dist_init_addr"] = dist_init_addr
 454         else:
 455             for i in range(num_engines_on_this_node):
 456                 addr_and_ports[rank + i]["dist_init_addr"] = f"{get_addr()}:{get_port(30 + args.sglang_dp_size)}"
 457 
 458     for rank, _ in rollout_engines:
 459         for key in ["port", "nccl_port", "dist_init_addr"]:
 460             assert key in addr_and_ports[rank], f"Engine {rank} {key} is not set."
 461         logger.info(f"Ports for engine {rank}: {addr_and_ports[rank]}")
 462 
 463     return addr_and_ports, node_port_cursor
