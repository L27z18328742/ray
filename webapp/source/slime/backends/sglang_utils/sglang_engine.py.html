  21 def get_base_gpu_id(args, rank):
  22     num_gpus = min(args.num_gpus_per_node, args.rollout_num_gpus_per_engine)
  23     if args.colocate:
  24         start_index = (rank * num_gpus) % args.num_gpus_per_node
  25     else:
  26         num_actor_gpus = 0 if args.debug_rollout_only else args.actor_num_gpus_per_node * args.actor_num_nodes
  27         start_index = (num_actor_gpus + rank * num_gpus) % args.num_gpus_per_node
  28     return start_index

  31 def launch_server_process(server_args: ServerArgs) -> multiprocessing.Process:
  32     # Expandable segments help the colocated training actor tolerate repeated
  33     # cache releases, but SGLang's allocator/sleep path does not support them.
  34     # The rollout Ray actor inherits the job environment, so remove the option
  35     # before spawning every SGLang server and its children.
  36     os.environ.pop("PYTORCH_CUDA_ALLOC_CONF", None)
  37     os.environ.pop("PYTORCH_ALLOC_CONF", None)
  38 
  39     if getattr(server_args, "encoder_only", False):
  40         from sglang.srt.disaggregation.encode_server import launch_server_process as sglang_launch_server_process
  41 
  42         return sglang_launch_server_process(
  43             server_args,
  44             start_method="spawn",
  45             wait_for_server=True,
  46         )
  47 
  48     from sglang.srt.entrypoints.http_server import launch_server
  49 
  50     multiprocessing.set_start_method("spawn", force=True)
  51     server_args.host = server_args.host.strip("[]")
  52     p = multiprocessing.Process(target=launch_server, args=(server_args,))
  53     p.start()
  54 
  55     if getattr(server_args, "node_rank", 0) != 0:
  56         return p
  57 
  58     _wait_server_healthy(
  59         base_url=server_args.url(),
  60         api_key=server_args.api_key,
  61         is_process_alive=lambda: p.is_alive(),
  62     )
  63 
  64     return p

  67 def _wait_server_healthy(base_url, api_key, is_process_alive):
  68     headers = {
  69         "Content-Type": "application/json; charset=utf-8",
  70         "Authorization": f"Bearer {api_key}",
  71     }
  72 
  73     with requests.Session() as session:
  74         while True:
  75             try:
  76                 response = session.get(f"{base_url}/health_generate", headers=headers)
  77                 if response.status_code == 200:
  78                     break
  79             except requests.RequestException:
  80                 pass
  81 
  82             if not is_process_alive():
  83                 raise Exception("Server process terminated unexpectedly.")
  84 
  85             time.sleep(2)

 157     def _init_external(self, expect_server_args, external_engine_need_check_fields):
 158         logger.info(f"Use external SGLang engine (rank={self.rank}, expect_server_args={expect_server_args})")
 159 
 160         def _sanity_check_server_args(actual_server_args, expect_server_args):
 161             for name in external_engine_need_check_fields:
 162                 expect_value = expect_server_args.get(name)
 163                 actual_value = actual_server_args.get(name)

 165                     actual_value == expect_value

 177     def _register_to_router(self, server_args_dict):
 178         if self.worker_type == "encoder":
 179             return
 180 
 181         if self.node_rank == 0 and self.router_ip and self.router_port:
 182             worker_url = f"http://{self.server_host}:{self.server_port}"
 183             payload = {
 184                 "url": worker_url,
 185                 "worker_type": self.worker_type,
 186             }
 187             if self.worker_type == "prefill":
 188                 bootstrap_port = server_args_dict.get("disaggregation_bootstrap_port")
 189                 if bootstrap_port is None:
 190                     raise RuntimeError(
 191                         f"Prefill worker {worker_url} does not have disaggregation_bootstrap_port; "
 192                         "cannot register it to the PD router."
 193                     )
 194                 payload["bootstrap_port"] = bootstrap_port
 195             response = requests.post(
 196                 f"http://{self.router_ip}:{self.router_port}/workers",
 197                 json=payload,
 198             )
 199             response.raise_for_status()

 201     def _make_request(self, endpoint: str, payload: dict | None = None):
 202         """Make a POST request to the specified endpoint with the given payload.
 203 
 204         Args:
 205             endpoint: The API endpoint to call
 206             payload: The JSON payload to send (default: empty dict)
 207 
 208         Returns:
 209             The JSON response from the server
 210         """
 211         if self.node_rank != 0:
 212             return
 213 
 214         url = f"http://{self.server_host}:{self.server_port}/{endpoint}"
 215         response = requests.post(url, json=payload or {})
 216         try:
 217             response.raise_for_status()
 218         except requests.exceptions.HTTPError as e:
 219             e.add_note(f"{response.text=}")
 220             raise
 221         return response.json()

 270     def flush_cache(self):
 271         """Flush the cache of the server."""
 272         if self.node_rank != 0:
 273             return
 274         # flush cache will not return status_code 200 when there are pending requests
 275         for _ in range(60):
 276             try:
 277                 response = requests.get(f"http://{self.server_host}:{self.server_port}/flush_cache")
 278                 if response.status_code == 200:
 279                     break
 280                 logger.info(f"Error flushing cache: HTTP {response.status_code} {response.text!r}")
 281                 time.sleep(1)
 282             except NewConnectionError as e:
 283                 raise e
 284             except Exception as e:
 285                 logger.info(f"Error flushing cache: {e}")
 286                 time.sleep(1)
 287                 continue
 288         else:
 289             raise TimeoutError("Timeout while flushing cache.")

 296     def shutdown(self):
 297         if self.args.rollout_external:
 298             return
 299 
 300         logger.info(f"Shutdown engine {self.server_host}:{self.server_port}...")
 301         if self.worker_type != "encoder" and self.node_rank == 0:
 302             worker_url = f"http://{self.server_host}:{self.server_port}"
 303             response = None
 304             try:
 305                 all_workers = requests.get(f"http://{self.router_ip}:{self.router_port}/workers").json()["workers"]
 306                 for worker in all_workers:
 307                     if worker["url"] == worker_url:
 308                         worker_id = worker["id"]
 309                         response = requests.delete(f"http://{self.router_ip}:{self.router_port}/workers/{worker_id}")
 310                         break
 311                 else:
 312                     logger.warning(f"Worker {worker_url} not found in router during shutdown.")
 313             except Exception as e:
 314                 logger.warning(f"Failed to fetch workers list or remove worker: {e}")
 315 
 316             if response is not None:
 317                 response.raise_for_status()
 318         kill_process_tree(self.process.pid)

 506 def _compute_server_args(
 507     args,
 508     rank,
 509     dist_init_addr,
 510     nccl_port,
 511     host,
 512     port,
 513     worker_type: str = "regular",
 514     disaggregation_bootstrap_port: int | None = None,
 515     base_gpu_id: int | None = None,
 516     sglang_overrides: dict | None = None,
 517     num_gpus_per_engine: int | None = None,
 518 ):
 519     _gpus_per_engine = num_gpus_per_engine or args.rollout_num_gpus_per_engine
 520     normalized_overrides = {key.replace("-", "_"): value for key, value in (sglang_overrides or {}).items()}
 521     pp_size = int(normalized_overrides.get("pp_size", args.sglang_pp_size))
 522     tp_size = int(normalized_overrides.get("tp_size", _gpus_per_engine // pp_size))
 523     nnodes = max(1, _gpus_per_engine // args.num_gpus_per_node)
 524     node_rank = rank % nnodes
 525     base = base_gpu_id if base_gpu_id is not None else get_base_gpu_id(args, rank)
 526     base = accelerator.resolve_visible_device_id(base)
 527     kwargs = {
 528         "model_path": args.hf_checkpoint,
 529         "trust_remote_code": True,
 530         "random_seed": args.seed + rank * args.num_gpus_per_node,
 531         # memory
 532         "enable_memory_saver": args.offload_rollout,
 533         # distributed
 534         "host": host,
 535         "port": port,
 536         "nccl_port": nccl_port,
 537         "nnodes": nnodes,
 538         "node_rank": node_rank,
 539         "dist_init_addr": dist_init_addr,
 540         "gpu_id_step": 1,
 541         "base_gpu_id": base,
 542         # parallel
 543         "tp_size": tp_size,
 544         "dp_size": args.sglang_dp_size,
 545         "pp_size": pp_size,
 546         "ep_size": args.sglang_ep_size,
 547         # always skip warmup to prevent warmup timeout.
 548         "skip_server_warmup": True,
 549         # always enable draft weights cpu backup so that we run training without mtp weights.
 550         "enable_draft_weights_cpu_backup": True,
 551         # Always enable Prometheus metrics so the router /engine_metrics endpoint
 552         # is available for external scraping.
 553         "enable_metrics": True,
 554     }
 555 
 556     if worker_type == "prefill":
 557         kwargs["disaggregation_mode"] = "prefill"
 558         kwargs["load_balance_method"] = "follow_bootstrap_room"
 559         assert (
 560             disaggregation_bootstrap_port is not None
 561         ), "disaggregation_bootstrap_port must be set for prefill worker"
 562         kwargs["disaggregation_bootstrap_port"] = disaggregation_bootstrap_port
 563     elif worker_type == "decode":
 564         kwargs["disaggregation_mode"] = "decode"
 565         kwargs["prefill_round_robin_balance"] = True
 566     elif worker_type == "encoder":
 567         kwargs["encoder_only"] = True
 568 
 569     if args.use_rollout_routing_replay:
 570         kwargs["enable_return_routed_experts"] = True
 571     if args.fp16:
 572         kwargs["dtype"] = "float16"
 573     external_engine_need_check_fields = [k for k in kwargs.keys() if k not in _EXTERNAL_ENGINE_SKIP_CHECK_FIELDS]
 574 
 575     server_arg_fields = dataclasses.fields(ServerArgs)
 576     server_arg_field_names = {attr.name for attr in server_arg_fields}
 577     unused_keys = set(kwargs.keys())
 578     for attr in server_arg_fields:
 579         if worker_type == "decode" and attr.name == "enable_hierarchical_cache":
 580             continue
 581         if hasattr(args, f"sglang_{attr.name}") and attr.name not in kwargs:
 582             kwargs[attr.name] = getattr(args, f"sglang_{attr.name}")
 583         unused_keys.discard(attr.name)
 584 
 585     # Per-server-group overrides from --sglang-config YAML.
 586     # Applied after base args so they take highest priority.
 587     if sglang_overrides:
 588         for key, value in sglang_overrides.items():
 589             normalized_key = key.replace("-", "_")
 590             if normalized_key != key:
 591                 logger.warning(
 592                     f"sglang_overrides key '{key}' normalized to '{normalized_key}' (rank={rank}). "
 593                     "Please use underscore style in YAML overrides."
 594                 )
 595             if normalized_key in kwargs:
 596                 logger.info(
 597                     f"sglang_overrides: overriding {normalized_key}={kwargs[normalized_key]} -> {value} (rank={rank})"
 598                 )
 599             kwargs[normalized_key] = value
 600             if normalized_key in server_arg_field_names:
 601                 unused_keys.discard(normalized_key)
 602             else:
 603                 unused_keys.add(normalized_key)
 604 
 605     if (
 606         "cuda_graph_backend_prefill" in server_arg_field_names
 607         and kwargs.get("enable_memory_saver")
 608         and kwargs.get("cuda_graph_backend_prefill") is None
 609     ):
 610         # Breakable is SGLang's default prefill backend on CUDA, but it is incompatible with memory saver mode.
 611         kwargs["cuda_graph_backend_prefill"] = "disabled"
 612 
 613     # for compatibility with old args
 614     if len(unused_keys) > 0:
 615         logger.info(f"Warning: The following arguments is not supported in the current sglang: {unused_keys}.")
 616         for key in unused_keys:
 617             kwargs.pop(key)
 618 
 619     return kwargs, external_engine_need_check_fields
