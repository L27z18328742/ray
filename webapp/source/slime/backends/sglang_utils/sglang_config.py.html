  11 @dataclasses.dataclass
  12 class ServerGroupConfig:
  13     """Configuration for a single server group.
  14 
  15     Attributes:
  16         worker_type: One of "regular", "prefill", "decode", "placeholder",
  17                      or "encoder".
  18                      "placeholder" reserves GPU slots without creating engines.
  19                      "encoder" creates encoder-only engines for EPD
  20                      (Encoder-Prefill-Decode) disaggregation; encoder engines
  21                      are started first and their URLs are automatically
  22                      injected into prefill groups as ``encoder_urls``.
  23         num_gpus: Total number of GPUs for this group.
  24         num_gpus_per_engine: GPUs per engine for this group.  Overrides the
  25                              model-level or global ``--rollout-num-gpus-per-engine``.
  26         overrides: Optional dict of SGLang ``ServerArgs`` field overrides.
  27                    These are applied on top of the base CLI ``--sglang-*``
  28                    arguments in ``_compute_server_args``.
  29     """
  30 
  31     worker_type: str
  32     num_gpus: int
  33     num_gpus_per_engine: int | None = None
  34     overrides: dict = dataclasses.field(default_factory=dict)
  35 
  36     def __post_init__(self):
  37         valid_types = {"regular", "prefill", "decode", "placeholder", "encoder"}
  38         assert (
  39             self.worker_type in valid_types
  40         ), f"Invalid worker_type '{self.worker_type}', must be one of {valid_types}"
  41         assert self.num_gpus > 0, f"num_gpus must be > 0, got {self.num_gpus}"

  44 @dataclasses.dataclass
  45 class ModelConfig:
  46     """Configuration for a single model deployment.
  47 
  48     Attributes:
  49         name: Unique name for this model (e.g. "actor", "reward").
  50         model_path: HF checkpoint path.  Falls back to ``args.hf_checkpoint``.
  51         num_gpus_per_engine: Default GPUs per engine for all groups in this
  52                              model.  Individual groups can override.
  53         server_groups: Server group configurations for this model.
  54         update_weights: Whether this model receives weight updates from
  55                         training.  Set to ``False`` for frozen models
  56                         (reference, reward, etc.).  When ``None`` (default),
  57                         automatically inferred in ``resolve()``: ``True`` if
  58                         model_path matches ``args.hf_checkpoint``, ``False``
  59                         otherwise.
  60     """
  61 
  62     name: str
  63     model_path: str | None = None
  64     num_gpus_per_engine: int | None = None
  65     server_groups: list[ServerGroupConfig] = dataclasses.field(default_factory=list)
  66     update_weights: bool | None = None
  67 
  68     def resolve(self, args) -> None:
  69         """Resolve per-group defaults from model-level then args-level values."""
  70         default_gpus_per_engine = self.num_gpus_per_engine or args.rollout_num_gpus_per_engine
  71         default_model_path = self.model_path or args.hf_checkpoint
  72         for g in self.server_groups:
  73             if g.num_gpus_per_engine is None:
  74                 g.num_gpus_per_engine = default_gpus_per_engine
  75             # Inject model_path into overrides so _compute_server_args picks it up.
  76             if "model_path" not in g.overrides:
  77                 g.overrides["model_path"] = default_model_path
  78 
  79         # Validate: all server groups within a model must share the same model_path.
  80         if self.server_groups:
  81             model_paths = {g.overrides["model_path"] for g in self.server_groups}
  82             assert len(model_paths) == 1, (
  83                 f"Model '{self.name}' has server groups with different model_path values: "
  84                 f"{model_paths}. All server groups within a model must use the same model_path."
  85             )
  86             effective_model_path = model_paths.pop()
  87         else:
  88             effective_model_path = default_model_path
  89 
  90         # Auto-infer update_weights when not explicitly set.
  91         if self.update_weights is None:
  92             if effective_model_path != args.hf_checkpoint:
  93                 logger.warning(
  94                     f"Model '{self.name}' uses model_path='{effective_model_path}' which differs "
  95                     f"from hf_checkpoint='{args.hf_checkpoint}'. Defaulting update_weights to False. "
  96                     f"Set update_weights explicitly in the config to suppress this warning."
  97                 )
  98                 self.update_weights = False
  99             else:
 100                 self.update_weights = True
 101 
 102     @property
 103     def has_pd_disaggregation(self) -> bool:
 104         return any(g.worker_type in ("prefill", "decode") for g in self.server_groups)
 105 
 106     @property
 107     def has_encoder_disaggregation(self) -> bool:
 108         return any(g.worker_type == "encoder" for g in self.server_groups)
 109 
 110     @property
 111     def total_num_gpus(self) -> int:
 112         return sum(g.num_gpus for g in self.server_groups)

 115 @dataclasses.dataclass
 116 class SglangConfig:
 117     """Configuration for SGLang engine deployment.
 118 
 119     Loaded from ``--sglang-config`` YAML file.
 120 
 121     **Config format**::
 122 
 123         sglang:
 124           - name: actor
 125             model_path: /path/to/actor
 126             update_weights: true          # receives training weight updates (default)
 127             num_gpus_per_engine: 2
 128             server_groups:
 129               - worker_type: prefill
 130                 num_gpus: 4
 131                 num_gpus_per_engine: 2
 132               - worker_type: decode
 133                 num_gpus: 8
 134                 num_gpus_per_engine: 4
 135           - name: ref
 136             model_path: /path/to/ref
 137             update_weights: false          # frozen, no weight updates
 138             server_groups:
 139               - worker_type: regular
 140                 num_gpus: 4
 141 
 142     Each model gets its own router.  ``placeholder`` groups reserve GPU
 143     slots without creating engines.  ``overrides`` are ``ServerArgs``
 144     field names applied on top of the base ``--sglang-*`` CLI args.
 145 
 146     Set ``update_weights: false`` for frozen models (reference, reward,
 147     etc.) that should not receive weight updates from training.
 148 
 149     .. note::
 150 
 151        ``engine_groups`` is accepted as a backward-compatible alias for
 152        ``server_groups`` in the YAML config.
 153     """
 154 
 155     models: list[ModelConfig]
 156 
 157     @staticmethod
 158     def from_yaml(path: str) -> "SglangConfig":
 159         with open(path) as f:
 160             data = yaml.safe_load(f)
 161 
 162         assert "sglang" in data, (
 163             f"sglang config must have a 'sglang' key, got {list(data.keys())}. "
 164             f"Wrap your server_groups inside a model entry under 'sglang'."
 165         )
 166         models = []
 167         for m in data["sglang"]:
 168             # Accept both "server_groups" and legacy "engine_groups".
 169             raw_groups = m.get("server_groups") or m.get("engine_groups") or []
 170             groups = [ServerGroupConfig(**g) for g in raw_groups]
 171             models.append(
 172                 ModelConfig(
 173                     name=m["name"],
 174                     model_path=m.get("model_path"),
 175                     num_gpus_per_engine=m.get("num_gpus_per_engine"),
 176                     server_groups=groups,
 177                     update_weights=m.get("update_weights"),
 178                 )
 179             )
 180         return SglangConfig(models=models)
 181 
 182     @staticmethod
 183     def from_prefill_num_servers(args) -> "SglangConfig":
 184         """Build a config equivalent to the legacy --prefill-num-servers flag."""
 185         total_gpus = args.rollout_num_gpus
 186         prefill_gpus = args.prefill_num_servers * args.rollout_num_gpus_per_engine
 187         decode_gpus = total_gpus - prefill_gpus
 188         assert decode_gpus > 0, f"No decode GPUs: total {total_gpus}, prefill {prefill_gpus}"
 189         return SglangConfig(
 190             models=[
 191                 ModelConfig(
 192                     name="default",
 193                     server_groups=[
 194                         ServerGroupConfig(worker_type="prefill", num_gpus=prefill_gpus),
 195                         ServerGroupConfig(worker_type="decode", num_gpus=decode_gpus),
 196                     ],
 197                 )
 198             ]
 199         )
 200 
 201     @property
 202     def has_pd_disaggregation(self) -> bool:
 203         return any(m.has_pd_disaggregation for m in self.models)
 204 
 205     @property
 206     def total_num_gpus(self) -> int:
 207         return sum(m.total_num_gpus for m in self.models)

 210 def resolve_sglang_config(args) -> SglangConfig:
 211     """Resolve the configured, legacy PD, or default SGLang deployment."""
 212     if getattr(args, "sglang_config", None) is not None:
 213         config = SglangConfig.from_yaml(args.sglang_config)
 214         expected = args.rollout_num_gpus
 215         actual = config.total_num_gpus
 216         assert actual == expected, f"sglang_config total GPUs ({actual}) != rollout_num_gpus ({expected})"
 217         return config
 218 
 219     if args.rollout_num_gpus == 0:
 220         return SglangConfig(models=[ModelConfig(name="default", server_groups=[])])
 221 
 222     if args.prefill_num_servers is not None:
 223         return SglangConfig.from_prefill_num_servers(args)
 224 
 225     return SglangConfig(
 226         models=[
 227             ModelConfig(
 228                 name="default",
 229                 server_groups=[ServerGroupConfig(worker_type="regular", num_gpus=args.rollout_num_gpus)],
 230             )
 231         ]
 232     )
