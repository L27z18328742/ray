  43 def _try_load_glm4v_processor(name_or_path: str, **kwargs):
  44     """Fallback: manually construct a Glm4vProcessor for GLM-4.6V / GLM-4.5V models.
  45 
  46     AutoProcessor fails for these models on transformers < 5.0 because
  47     the Glm46VProcessor / Glm4vMoeProcessor classes are not registered.
  48     The underlying Glm4vProcessor (non-MoE) works for both variants since
  49     they share the same vision architecture.
  50     """
  51     try:
  52         from transformers.models.glm4v.image_processing_glm4v import Glm4vImageProcessor
  53         from transformers.models.glm4v.processing_glm4v import Glm4vProcessor
  54         from transformers.models.glm4v.video_processing_glm4v import Glm4vVideoProcessor
  55     except ImportError:
  56         return None
  57 
  58     pp_path = Path(name_or_path) / "preprocessor_config.json"
  59     vp_path = Path(name_or_path) / "video_preprocessor_config.json"
  60     if not pp_path.exists():
  61         return None
  62 
  63     skip_keys = {"image_processor_type", "processor_class", "video_processor_type"}
  64     with open(pp_path) as f:
  65         pp_cfg = {k: v for k, v in json.load(f).items() if k not in skip_keys}
  66     image_processor = Glm4vImageProcessor(**pp_cfg)
  67 
  68     video_processor = None
  69     if vp_path.exists():
  70         with open(vp_path) as f:
  71             vp_cfg = {k: v for k, v in json.load(f).items() if k not in skip_keys}
  72         video_processor = Glm4vVideoProcessor(**vp_cfg)
  73 
  74     tokenizer = AutoTokenizer.from_pretrained(name_or_path, **kwargs)
  75     proc = Glm4vProcessor(
  76         image_processor=image_processor,
  77         tokenizer=tokenizer,
  78         video_processor=video_processor,
  79         chat_template=tokenizer.chat_template,
  80     )
  81     logger.info(f"Loaded Glm4vProcessor manually for {name_or_path}")
  82     return proc

  85 def load_processor(name_or_path: str, **kwargs):
  86     try:
  87         proc = AutoProcessor.from_pretrained(name_or_path, **kwargs)
  88     except (OSError, ValueError) as e:
  89         logger.warning(f"Failed to load processor from {name_or_path}: {e}")
  90         proc = None
  91 
  92     # If HF returned a tokenizer instead of a proper processor, discard it.
  93     if isinstance(proc, PreTrainedTokenizerBase) or not isinstance(proc, ProcessorMixin):
  94         # Fallback: try to construct a GLM-4.6V / GLM-4.5V processor manually.
  95         proc = _try_load_glm4v_processor(name_or_path, **kwargs)
  96 
  97     return proc

 100 def _extract_images_from_messages(messages):
 101     """Extract PIL images from chat messages containing multimodal content.
 102 
 103     Handles base64 strings (with or without data: URI prefix), file paths,
 104     and PIL Image objects embedded in message content dicts.
 105     """
 106     images = []
 107     for msg in messages:
 108         content = msg.get("content", [])
 109         if not isinstance(content, list):
 110             continue
 111         for item in content:
 112             if not isinstance(item, dict) or item.get("type") != "image":
 113                 continue
 114             image_data = item.get("image")
 115             if image_data is None:
 116                 continue
 117             if isinstance(image_data, Image.Image):
 118                 images.append(image_data)
 119             elif isinstance(image_data, str):
 120                 if image_data.startswith("data:"):
 121                     _, encoded = image_data.split(",", 1)
 122                     images.append(Image.open(io.BytesIO(base64.b64decode(encoded))))
 123                 else:
 124                     try:
 125                         raw = base64.b64decode(image_data)
 126                         images.append(Image.open(io.BytesIO(raw)))
 127                     except Exception:
 128                         # Not base64 — try as file path
 129                         images.append(Image.open(image_data))
 130     return images

 133 def process_vision_info(prompt, processor):
 134     """Extract PIL images (and videos) from the message list for training.
 135 
 136     Tries qwen_vl_utils first (Qwen VL family), falls back to generic
 137     extraction for other models (e.g. GLM-4.6V).
 138     """
 139     try:
 140         from qwen_vl_utils import process_vision_info as qwen_process_vision_info
 141 
 142         if hasattr(processor.image_processor, "patch_size"):
 143             image_patch_size = processor.image_processor.patch_size
 144         else:
 145             image_patch_size = DEFAULT_PATCH_SIZE
 146         images, videos = qwen_process_vision_info(prompt, image_patch_size=image_patch_size)
 147     except Exception:
 148         # Fallback: generic extraction for non-Qwen models
 149         images = _extract_images_from_messages(prompt) or None
 150         videos = None
 151 
 152     return {"images": images, "videos": videos}

 155 def encode_image_for_rollout_engine(image) -> str:
 156     """Load an image from path, ensure RGB, encode as PNG base64 string."""
 157     buffer = io.BytesIO()
 158     if image.mode != "RGB":
 159         image = image.convert("RGB")
 160     image.save(buffer, format="PNG")
 161     image_base64 = base64.b64encode(buffer.getvalue()).decode("utf-8")
 162     return f"data:image/png;base64,{image_base64}"
