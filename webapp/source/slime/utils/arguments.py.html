  19 def reset_arg(parser, name, **kwargs):
  20     """
  21     Reset the default value of a Megatron argument.
  22     :param parser: The argument parser.
  23     :param name: The name of the argument to reset.
  24     :param default: The new default value.
  25     """
  26     for action in parser._actions:
  27         if name in action.option_strings:
  28             if "default" in kwargs:
  29                 action.default = kwargs["default"]
  30             break
  31     else:
  32         parser.add_argument(name, **kwargs)

1633     sglang_ns = None

1635         sglang_ns = sglang_parse_args()

1638     # Uses ignore_unknown_args=True so that --sglang-* and pre-parsed CLI flags

1640     from slime.backends.megatron_utils.arguments import megatron_parse_args

1668 def _apply_megatron_role_overrides(base_args, overrides, role):
1669     role_args = copy.deepcopy(base_args)
1670     ignored_keys = {"num_nodes", "num_gpus_per_node"}
1671 
1672     # Apply overrides from the YAML config.
1673     # Unspecified keys inherit from base_args via deepcopy.
1674     for key, value in overrides.items():
1675         if key in ignored_keys:
1676             logger.info(f"Ignoring {role} config key '{key}'; GPU allocation always follows CLI args.")
1677             continue
1678         if not hasattr(role_args, key):
1679             logger.warning(f"{role.capitalize()} config key '{key}' is not a known argument, setting it anyway.")
1680         else:
1681             # YAML safe_load doesn't parse scientific notation (e.g. 1e-5) as float.
1682             # Coerce the value to match the type of the existing attribute.
1683             original = getattr(role_args, key)
1684             if original is not None and isinstance(value, str) and isinstance(original, (int, float)):
1685                 try:
1686                     value = type(original)(value)
1687                 except (ValueError, TypeError):
1688                     pass
1689         setattr(role_args, key, value)
1690 
1691     if role == "critic":
1692         # Critic-specific: disable features that only apply to actors.
1693         role_args.kl_coef = 0
1694         role_args.use_opd = False
1695         role_args.custom_advantage_function_path = None
1696         role_args.untie_embeddings_and_output_weights = True
1697         if "disable_param_buffers_cpu_backup" not in overrides:
1698             role_args.disable_param_buffers_cpu_backup = False
1699 
1700     return role_args

1703 def parse_megatron_role_args(base_args, megatron_config_path, role):
1704     """Parse role-specific arguments from a unified Megatron YAML config.
1705 
1706     The config must contain a top-level ``megatron`` list with per-role entries.
1707     Missing roles inherit the base args unchanged.
1708     """
1709     assert role in {"actor", "critic"}, f"Unsupported Megatron config role: {role}"
1710 
1711     with open(megatron_config_path) as f:
1712         raw_config = yaml.safe_load(f) or {}
1713 
1714     assert "megatron" in raw_config, (
1715         "megatron config must contain a top-level 'megatron' list, e.g. "
1716         "megatron: [{name: default, role: actor, overrides: {...}}]"
1717     )
1718 
1719     overrides = {}
1720     megatron_entries = raw_config["megatron"]
1721     assert isinstance(megatron_entries, list), (
1722         "megatron config 'megatron' field must be a list, e.g. "
1723         "megatron: [{name: default, role: actor, overrides: {...}}]"
1724     )
1725     role_entries = [entry for entry in megatron_entries if entry.get("role") == role]
1726     assert len(role_entries) <= 1, (
1727         f"megatron config must contain at most one entry with role={role}, e.g. "
1728         f"megatron: [{{name: default, role: {role}, overrides: {{...}}}}]"
1729     )
1730     if role_entries:
1731         role_entry = role_entries[0]
1732         overrides = role_entry.get("overrides") or role_entry.get("args") or {}
1733     else:
1734         logger.info(
1735             f"No megatron config entry with role={role} found in {megatron_config_path}; using inherited args."
1736         )
1737 
1738     role_args = _apply_megatron_role_overrides(base_args, overrides, role)
1739     logger.info(
1740         f"Parsed megatron config for role={role} from {megatron_config_path}: overrides = {list(overrides.keys())}"
1741     )
1742 
1743     return role_args
