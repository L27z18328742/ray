  14 def decode_meta_array(
  15     meta_info: dict[str, Any],
  16     keys: str | Iterable[str],
  17     *,
  18     dtype: torch.dtype,
  19 ) -> torch.Tensor | None:
  20     if isinstance(keys, str):
  21         keys = (keys,)
  22     for key in keys:
  23         if key in meta_info:
  24             value = meta_info[key]
  25             break
  26     else:
  27         return None
  28 
  29     if value is None:
  30         return None
  31     if isinstance(value, str):
  32         import pybase64
  33 
  34         value = pybase64.b64decode(value.encode("ascii"))
  35     if isinstance(value, bytes | bytearray | memoryview):
  36         with warnings.catch_warnings():
  37             warnings.filterwarnings("ignore", message="The given buffer is not writable", category=UserWarning)
  38             return torch.frombuffer(value, dtype=dtype)
  39     if torch.is_tensor(value):
  40         return value.detach().to(device="cpu", dtype=dtype).reshape(-1)
  41     if hasattr(value, "flags") and not value.flags.writeable:
  42         value = value.copy()
  43     return torch.as_tensor(value, dtype=dtype).reshape(-1)

  64     A metaclass for creating singleton classes.
  65     """
  66 
  67     _instances = {}
  68 
  69     def __call__(cls, *args, **kwargs):
  70         if cls not in cls._instances:
  71             instance = super().__call__(*args, **kwargs)
  72             cls._instances[cls] = instance
  73         return cls._instances[cls]

  76         cls._instances = {}

  79 def exec_command(cmd: str, capture_output: bool = False) -> str | None:
  80     print(f"EXEC: {cmd}", flush=True)
  81

  83         result = subprocess.run(
  84             ["bash", "-c", cmd],

 111 def get_free_port(start_port=10000, consecutive=1):
 112     # find the port where port, port + 1, port + 2, ... port + consecutive - 1 are all available
 113     port = start_port
 114     while not all(is_port_available(port + i) for i in range(consecutive)):
 115         port += 1
 116     return port

 119 def should_run_periodic_action(
 120     rollout_id: int,
 121     interval: int | None,
 122     num_rollout_per_epoch: int | None = None,
 123     num_rollout: int | None = None,
 124 ) -> bool:
 125     """
 126     Return True when a periodic action (eval/save/checkpoint) should run.
 127 
 128     Args:
 129         rollout_id: The current rollout index (0-based).
 130         interval: Desired cadence; disables checks when None.
 131         num_rollout_per_epoch: Optional epoch boundary to treat as a trigger.
 132     """
 133     if interval is None:
 134         return False
 135 
 136     if num_rollout is not None and rollout_id == num_rollout - 1:
 137         return True
 138 
 139     step = rollout_id + 1
 140     return (step % interval == 0) or (num_rollout_per_epoch is not None and step % num_rollout_per_epoch == 0)

 152 # details: https://stackoverflow.com/questions/773/how-do-i-use-itertools-groupby
 153 def group_by(iterable, key=None):
 154     """Similar to itertools.groupby, but do not require iterable to be sorted"""
 155     ret = defaultdict(list)
 156     for item in iterable:
 157         ret[key(item) if key is not None else item].append(item)
 158     return dict(ret)
