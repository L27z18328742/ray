slime/utils/dp_schedule.py

   1 """Per-rollout DP/microbatch scheduling.
   2 
   3 Pure-Python logic that decides, for one rollout's worth of sample lengths,
   4 how to group samples into micro-batches and which DP rank owns each mbs.
   5 Lives outside the ray/sglang-importing modules so it can be unit-tested
   6 under CPU-only CI.
   7 
   8 The scheduling philosophy is **pack first, distribute second**:
   9 
  10   1. Group samples by rollout id (``rollout_indices[i]`` =
  11      ``samples[i].index``) and split rollouts into steps of
  12      ``global_batch_size`` rollouts each. In the common case one rollout
  13      emits one training sample so this is the same as a contiguous chunk;
  14      under compact / subagent one rollout may emit multiple training
  15      samples, in which case all of those samples stay in the same step.
  16   2. For each step, pack its samples into ``K`` micro-batches with a
  17      single first-fit pass (dynamic batch) or fixed-size chunking
  18      (static batch).
  19   3. Adjust ``K`` to a multiple of ``dp_size * (mb_group if vpp>1 else 1)``
  20      by splitting the largest multi-sample bins (dynamic only).
  21   4. Distribute the ``K`` mbs across ``dp_size`` ranks, ``K / dp_size``
  22      each, with either a strided round-robin or a Karmarkar-Karp pass on
  23      estimated mbs FLOPs.
  24 
  25 Invariants guaranteed by :func:`build_dp_schedule` (asserted by the tests):
  26   - every DP rank runs the **same** ``num_microbatches`` per training step
  27     (required for PP sync);
  28   - every mbs (dynamic path without ``balance_by_flops``) holds
  29     ``<= max_tokens_per_gpu * cp_size`` tokens, with one exception — an
  30     individual sample larger than that cap lands alone in its own mbs (and
  31     that mbs is the only one allowed to exceed the cap);
  32   - the union of per-rank sample indices equals the set of samples kept
  33     after trimming trailing rollouts (every kept sample placed exactly
  34     once);
  35   - flattening ``micro_batch_indices`` for a rank yields
  36     ``range(num_samples_rank)`` (each rank's samples are tiled exactly
  37     once by its mbs schedule).
  38 """

  51 def _calculate_workloads(step_lengths, args):
  52     return [calculate_fwd_flops([sl], args) for sl in step_lengths]

  55 def _pack_step_into_mbs(
  56     step_lengths: list[int],
  57     *,
  58     args: Any,
  59     use_dynamic_batch_size: bool,
  60     max_per_bin: int | None,
  61     micro_batch_size: int | None,
  62     balance_by_flops: bool = False,
  63 ) -> list[list[int]]:
  64     """Group a step's samples into mbs. Returns ``mbs[k]`` = local indices into ``step_lengths``."""
  65     if use_dynamic_batch_size:
  66         assert max_per_bin is not None
  67         if balance_by_flops:
  68             total_tokens = sum(step_lengths)
  69             num_mbs = max(1, (total_tokens + max_per_bin - 1) // max_per_bin)
  70             if num_mbs >= len(step_lengths):
  71                 return [[i] for i in range(len(step_lengths))]
  72             workloads = _calculate_workloads(step_lengths, args)
  73             # FLOPs balancing does not enforce the token cap per mbs. A
  74             # partition can exceed max_per_bin and may OOM if the cap is tight.
  75             return get_seqlen_balanced_partitions(workloads, num_mbs, equal_size=False)
  76         return first_fit_pack(step_lengths, max_per_bin)
  77     assert micro_batch_size is not None
  78     n = len(step_lengths)
  79     return [list(range(i, min(i + micro_batch_size, n))) for i in range(0, n, micro_batch_size)]

  82 def build_dp_schedule(
  83     args: Any,
  84     train_parallel_config: dict,
  85     total_lengths: list[int],
  86     *,
  87     global_batch_size: int,
  88     rollout_indices: list[int],
  89 ) -> tuple[list[list[int]], list[list[list[int]]], list[int], list[int]]:
  90     """Compute the per-rank DP partition and micro-batch schedule.
  91 
  92     See module docstring for the pack-first-distribute-second strategy.
  93 
  94     Args:
  95         args: Namespace with ``micro_batch_size``, ``use_dynamic_batch_size``,
  96             ``max_tokens_per_gpu``, ``balance_data``.
  97         train_parallel_config: ``{"dp_size", "cp_size", "vpp_size",
  98             "microbatch_group_size_per_vp_stage"}``.
  99         total_lengths: token count per sample, indexed globally.
 100         global_batch_size: number of rollouts (NOT training samples) per
 101             training step. Number of training steps =
 102             ``num_rollouts // global_batch_size``; trailing rollouts whose
 103             samples don't fit are dropped.
 104         rollout_indices: rollout id for each sample (``samples[i].index``).
 105             Samples sharing the same id are kept together in one step.
 106 
 107     Returns:
 108         ``(partitions, micro_batch_indices, num_microbatches, global_batch_sizes)``.
 109         ``global_batch_sizes[s]`` = rollout count for step s (constant
 110         ``global_batch_size`` for every step).
 111     """
 112     dp_size = train_parallel_config["dp_size"]
 113     cp_size = train_parallel_config["cp_size"]
 114     vpp_size = train_parallel_config["vpp_size"]
 115     mb_group = train_parallel_config["microbatch_group_size_per_vp_stage"]
 116 
 117     max_per_bin = None
 118     if args.use_dynamic_batch_size:
 119         assert args.max_tokens_per_gpu is not None
 120         max_per_bin = args.max_tokens_per_gpu * cp_size
 121 
 122     # mbs count per step must be divisible by (dp_size * mb_group_for_vpp) so
 123     # every rank ends up with the same num_mbs and (for VPP) the per-rank mbs
 124     # count is a multiple of mb_group.
 125     align_to = dp_size * (mb_group if vpp_size > 1 else 1)
 126 
 127     # Group samples by rollout id (preserve first-occurrence order). All
 128     # samples from one rollout stay in a single step so the per-rollout loss
 129     # reducer is well-defined.
 130     rollout_id_to_samples: dict[int, list[int]] = {}
 131     for sample_pos, rid in enumerate(rollout_indices):
 132         rollout_id_to_samples.setdefault(rid, []).append(sample_pos)
 133     rollout_ids = list(rollout_id_to_samples.keys())
 134 
 135     num_steps = len(rollout_ids) // global_batch_size
 136     assert num_steps >= 1, (
 137         f"num_rollouts ({len(rollout_ids)}) < global_batch_size ({global_batch_size}); "
 138         f"need at least one rollout per step."
 139     )
 140 
 141     partitions: list[list[int]] = [[] for _ in range(dp_size)]
 142     micro_batch_indices: list[list[list[int]]] = [[] for _ in range(dp_size)]
 143     num_microbatches: list[int] = []
 144     global_batch_sizes: list[int] = []
 145 
 146     for step_i in range(num_steps):
 147         step_rollouts = rollout_ids[step_i * global_batch_size : (step_i + 1) * global_batch_size]
 148         sample_indices = [pos for rid in step_rollouts for pos in rollout_id_to_samples[rid]]
 149         step_lengths = [total_lengths[i] for i in sample_indices]
 150         global_batch_sizes.append(global_batch_size)
 151         assert len(sample_indices) >= dp_size, (
 152             f"step {step_i}: {len(sample_indices)} samples < dp_size {dp_size}; "
 153             f"each step needs at least one sample per rank."
 154         )
 155 
 156         # 1. Pack samples in this step into mbs with one global pass.
 157         # ``step_mbs`` indices are LOCAL into ``sample_indices``.
 158         step_mbs = _pack_step_into_mbs(
 159             step_lengths,
 160             args=args,
 161             use_dynamic_batch_size=args.use_dynamic_batch_size,
 162             max_per_bin=max_per_bin,
 163             micro_batch_size=getattr(args, "micro_batch_size", None),
 164             balance_by_flops=args.balance_by_flops,
 165         )
 166 
 167         # 2. Align mbs count to a multiple of ``align_to``.
 168         target_K = max(((len(step_mbs) + align_to - 1) // align_to) * align_to, align_to)
 169         if target_K != len(step_mbs):
 170             if args.use_dynamic_batch_size:
 171                 expand_bins_by_splitting(step_mbs, target_K, step_lengths)
 172                 assert len(step_mbs) == target_K, (
 173                     f"dynamic path: could only produce {len(step_mbs)} mbs after maximal splitting; "
 174                     f"need {target_K}. step {step_i} has {len(sample_indices)} samples, below the "
 175                     f"alignment threshold ({align_to})."
 176                 )
 177             else:
 178                 raise AssertionError(
 179                     f"static path: num_mbs ({len(step_mbs)}) is not a multiple of "
 180                     f"dp_size * mb_group ({align_to}); got "
 181                     f"step_size={len(sample_indices)}, micro_batch_size={args.micro_batch_size}, "
 182                     f"dp_size={dp_size}, mb_group={mb_group if vpp_size > 1 else 1}. "
 183                     f"Splitting static mbs would break the fixed-size invariant; adjust the config "
 184                     f"so step_size % (dp_size * micro_batch_size * mb_group) == 0."
 185                 )
 186 
 187         K = len(step_mbs)
 188         num_mbs_per_rank = K // dp_size
 189         num_microbatches.append(num_mbs_per_rank)
 190 
 191         # 3. Distribute mbs across ranks: KK on estimated FLOPs when rank
 192         # workload balancing is enabled, otherwise a strided round-robin.
 193         if args.balance_data:
 194             step_workloads = _calculate_workloads(step_lengths, args)
 195             mbs_weights = [sum(step_workloads[i] for i in bin_) for bin_ in step_mbs]
 196             rank_mbs_idx = get_seqlen_balanced_partitions(mbs_weights, dp_size, equal_size=True)
 197         else:
 198             rank_mbs_idx = [list(range(r, K, dp_size)) for r in range(dp_size)]
 199 
 200         # 4. Build per-rank partitions (global sample indices) and micro_batch_indices
 201         # (local indices into partitions[r]).
 202         for r in range(dp_size):
 203             for mbs_idx in rank_mbs_idx[r]:
 204                 mbs_locals = step_mbs[mbs_idx]  # local indices into sample_indices
 205                 local_start = len(partitions[r])
 206                 partitions[r].extend(sample_indices[i] for i in mbs_locals)
 207                 micro_batch_indices[r].append(list(range(local_start, local_start + len(mbs_locals))))
 208 
 209     return partitions, micro_batch_indices, num_microbatches, global_batch_sizes
