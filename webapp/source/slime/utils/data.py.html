  24 def read_file(path):
  25     path, row_slice = _parse_generalized_path(path)
  26     reader = None
  27 
  28     if not os.path.exists(path):
  29         raise FileNotFoundError(f"Prompt dataset path '{path}' does not exist.")
  30 
  31     if path.endswith(".jsonl"):
  32 
  33         def jsonl_reader(p):
  34             with open(p, encoding="utf-8") as f:
  35                 for line_num, line in enumerate(f):
  36                     line = line.strip()
  37                     if not line:
  38                         continue
  39                     try:
  40                         yield json.loads(line)
  41                     except json.JSONDecodeError as e:
  42                         print(f"JSON decode error at line {line_num}: {e}")
  43                         continue
  44 
  45         reader = jsonl_reader(path)
  46 
  47     elif path.endswith(".parquet"):
  48         if pq is None:
  49             raise ImportError("pyarrow is required for parquet support")
  50 
  51         def parquet_reader(p):
  52             pf = pq.ParquetFile(p)
  53 
  54             for batch in pf.iter_batches():
  55                 yield from batch.to_pylist()
  56 
  57         reader = parquet_reader(path)
  58 
  59     else:
  60         raise ValueError(f"Unsupported file format: {path}. Supported formats are .jsonl and .parquet.")
  61 
  62     if row_slice is not None:
  63 
  64         logger.info("read_file path=%s applying slice row_slice=%s", path, row_slice)
  65         if (row_slice.start or 0) < 0 or (row_slice.stop or 0) < 0:
  66             # islice forbids negative indices, but the @[...] syntax accepts
  67             # them (e.g. "@[-100:]" = the last 100 rows). Resolving a negative
  68             # bound needs the total row count, so materialize for this case and
  69             # keep the streaming islice for plain non-negative slices.
  70             reader = iter(list(reader)[row_slice])
  71         else:
  72             reader = itertools.islice(reader, row_slice.start, row_slice.stop, row_slice.step)
  73 
  74     yield from reader

  77 def _parse_generalized_path(s: str):
  78     if (m := re.match(r"^(?P<real_path>.*)@\[(?P<start>-?\d*):(?P<end>-?\d*)\]$", s)) is not None:
  79         path = m.group("real_path")
  80         start = int(x) if (x := m.group("start")) != "" else None
  81         end = int(x) if (x := m.group("end")) != "" else None
  82         return path, slice(start, end)
  83 
  84     return s, None

  87 def filter_long_prompt(origin_samples: list[Sample], tokenizer, processor, max_length: int | None) -> list[Sample]:
  88     if max_length is None:
  89         return origin_samples
  90 
  91     if not isinstance(origin_samples[0].prompt, str):
  92         logger.warning(
  93             "Skipping max_length check for list prompt. Set apply_chat_template=True to enable length filtering."
  94         )
  95         return origin_samples
  96 
  97     if processor:
  98         # Use processor only for samples with actual multimodal content; use batched tokenizer for text-only.
  99         text_only = []
 100         multimodal = []
 101         for position, sample in enumerate(origin_samples):
 102             if sample.multimodal_inputs and any(v is not None for v in sample.multimodal_inputs.values()):
 103                 multimodal.append((position, sample))
 104             else:
 105                 text_only.append((position, sample))
 106         kept = []
 107         if text_only:
 108             prompts = [s.prompt for _, s in text_only]
 109             input_ids_list = tokenizer(prompts, add_special_tokens=False)["input_ids"]
 110             for (position, sample), input_ids in zip(text_only, input_ids_list, strict=True):
 111                 if len(input_ids) <= max_length:
 112                     kept.append((position, sample))
 113         if multimodal:
 114             from slime.utils.processing_utils import process_vision_info
 115 
 116             for position, sample in multimodal:
 117                 multimodal_inputs = process_vision_info(sample.prompt, processor)
 118                 processor_output = processor(text=sample.prompt, **multimodal_inputs)
 119                 input_ids = processor_output["input_ids"][0]
 120                 if len(input_ids) <= max_length:
 121                     kept.append((position, sample))
 122         # The two groups are scored separately for throughput, so restore the
 123         # dataset order here: without --rollout-shuffle the samples are consumed
 124         # in this order, and training should not depend on which of them happen
 125         # to carry multimodal content.
 126         kept.sort(key=lambda position_and_sample: position_and_sample[0])
 127         filtered_samples = [sample for _, sample in kept]
 128     else:
 129         prompts = [sample.prompt for sample in origin_samples]
 130         input_ids_list = tokenizer(prompts, add_special_tokens=False)["input_ids"]
 131         filtered_samples = [
 132             sample
 133             for sample, input_ids in zip(origin_samples, input_ids_list, strict=True)
 134             if len(input_ids) <= max_length
 135         ]
 136 
 137     logger.info(f"Filtered {len(origin_samples) - len(filtered_samples)} samples longer than max_length={max_length}.")
 138 
 139     return filtered_samples

 142 def _build_messages(data: dict, prompt_key: str, as_conversation: bool, multimodal_keys: dict = None):
 143     prompt = data.get(prompt_key)
 144 
 145     if isinstance(prompt, str):
 146         # If prompt is a string and we don't apply chat template, return the prompt as is.
 147         if not as_conversation:
 148             return prompt
 149         else:
 150             prompt = [{"role": "user", "content": prompt}]
 151 
 152     if multimodal_keys:
 153         # Build mapping: placeholder -> (MultimodalType, content_list)
 154         multimodals = {}
 155         for type_name, data_key in multimodal_keys.items():
 156             mt = MultimodalTypes.get(type_name)
 157             if mt:
 158                 multimodal_data = data.get(data_key)
 159                 if multimodal_data is not None:
 160                     multimodals[mt.placeholder] = (mt, list(multimodal_data))
 161 
 162         pattern = "(" + "|".join(re.escape(p) for p in multimodals.keys()) + ")"
 163 
 164         for message in prompt:
 165             if isinstance(message["content"], str):
 166                 content_list = []
 167                 for segment in re.split(pattern, message["content"]):
 168                     if not segment:
 169                         continue
 170                     if segment in multimodals:
 171                         mt, content = multimodals[segment]
 172                         assert len(content) > 0, (
 173                             f"Not enough {mt.name} data: more '{mt.placeholder}' placeholders in prompt "
 174                             f"than {mt.name}s provided in data"
 175                         )
 176                         item = content.pop(0)
 177                         # Support rich image config from https://github.com/QwenLM/Qwen3-VL/blob/main/README.md
 178                         # "images": [{"type": "image", "image": "path/to/img/01.jpeg", "max_pixels": 50176, "min_pixels": 50176}, {...}]
 179                         if isinstance(item, dict):
 180                             content_list.append(item)
 181                         # "images": ["path/to/img/01.jpeg", "url", "base64enc"]
 182                         else:
 183                             content_list.append({"type": mt.name, mt.name: item})
 184                     else:
 185                         content_list.append({"type": "text", "text": segment})
 186                 message["content"] = content_list
 187 
 188             elif isinstance(message["content"], list):
 189                 # TODO: handle more general cases. where message['content'] is a dict and contains multiple types of content.
 190                 # e.g.
 191                 #  "content": [
 192                 #     {
 193                 #         "type": "image",
 194                 #         "image": "https://qianwen-res.oss-cn-beijing.aliyuncs.com/Qwen-VL/assets/demo.jpeg",
 195                 #     },
 196                 #     {"type": "text", "text": "Describe this image."},
 197                 # ],
 198                 logger.warning("message['content'] is a list of dicts, no processing will be done.")
 199                 continue
 200             else:
 201                 raise ValueError(
 202                     f"Unsupported content type: {type(message['content'])}, expected str or list of dicts"
 203                 )
 204 
 205         for placeholder, (mt, remaining) in multimodals.items():
 206             assert len(remaining) == 0, (
 207                 f"Multimodal data count mismatch: {len(remaining)} more {mt.name}(s)"
 208                 f"than '{placeholder}' placeholders in prompt"
 209             )
 210 
 211     return prompt

 214 class Dataset:
 215     def __init__(
 216         self,
 217         path,
 218         tokenizer,
 219         processor,
 220         max_length,
 221         *,
 222         prompt_key="text",
 223         multimodal_keys=None,
 224         label_key=None,
 225         tool_key=None,
 226         metadata_key="metadata",
 227         seed=42,
 228         apply_chat_template=False,
 229         apply_chat_template_kwargs=None,
 230     ):
 231         origin_samples = []
 232         for data in read_file(path):
 233             # Both chat templates and multimodal inputs require conversation format (list of message dicts)
 234             as_conversation = apply_chat_template or (multimodal_keys is not None)
 235             prompt = _build_messages(data, prompt_key, as_conversation, multimodal_keys)
 236 
 237             metadata = data.get(metadata_key) or {}
 238             tools = None
 239             if tool_key is not None and tool_key in data:
 240                 tools = data[tool_key]
 241                 if isinstance(tools, str):
 242                     tools = json.loads(tools)
 243                 elif isinstance(tools, np.ndarray):
 244                     tools = tools.tolist()
 245                 assert isinstance(tools, list), f"tools must be a list, got {type(tools)} instead"
 246                 metadata["tools"] = tools
 247 
 248             if apply_chat_template:
 249                 output_prompt = tokenizer.apply_chat_template(
 250                     prompt,
 251                     tools=tools,
 252                     tokenize=False,
 253                     add_generation_prompt=True,
 254                     **(apply_chat_template_kwargs or {}),
 255                 )
 256             else:
 257                 output_prompt = prompt
 258 
 259             if processor:
 260                 from slime.utils.processing_utils import process_vision_info
 261 
 262                 assert isinstance(
 263                     prompt, list
 264                 ), f"prompt must be a list when processor is not None, got {type(prompt)} instead"
 265                 multimodal_inputs = process_vision_info(prompt, processor)
 266             else:
 267                 multimodal_inputs = None
 268 
 269             origin_samples.append(
 270                 Sample(
 271                     prompt=output_prompt,
 272                     label=data[label_key] if label_key is not None else None,
 273                     metadata=metadata,
 274                     multimodal_inputs=multimodal_inputs,
 275                 )
 276             )
 277 
 278         if max_length is not None:
 279             self.origin_samples = filter_long_prompt(origin_samples, tokenizer, processor, max_length)
 280         else:
 281             self.origin_samples = origin_samples
 282 
 283         self.epoch_id = -1
 284         self.seed = seed
 285         self.samples = self.origin_samples
 286 
 287     def shuffle(self, new_epoch_id):
 288         if self.epoch_id == new_epoch_id:
 289             return
 290 
 291         random.seed(self.seed + new_epoch_id)
 292         permutation = list(range(len(self.samples)))
 293         random.shuffle(permutation)
 294         self.samples = [self.origin_samples[i] for i in permutation]
 295         self.epoch_id = new_epoch_id
 296 
 297     def __getitem__(self, idx):
 298         return self.samples[idx]
 299 
 300     def __len__(self):
 301         return len(self.samples)

 304 def process_rollout_data(rollout_data_ref, dp_rank, dp_size):
 305     assert len(rollout_data_ref) == dp_size
 306     rollout_data = ray.get(rollout_data_ref[dp_rank].inner)
 307 
 308     # Keep `partition` in rollout_data: each local sample's position in the
 309     # flattened rollout batch (== its index in the rollout debug dump's
 310     # `samples`). It's a small list of ints, only read by the train debug dump /
 311     # log-prob capture, and ignored by training (the data iterator only fetches
 312     # requested keys), so there's no need to drop it.
 313     partition = rollout_data["partition"]
 314     total_lengths = rollout_data["total_lengths"]
 315 
 316     # save the seqlen of the whole rollout batch
 317     Timer().seq_lens = total_lengths
 318     rollout_data["total_lengths"] = [total_lengths[i] for i in partition]
 319 
 320     # `raw_reward` is shipped whole on purpose: log_passrate reshapes it into
 321     # [rollout_batch_size, n_samples_per_prompt] groups, which only works on the
 322     # full rollout batch. Metrics that pair a reward with this rank's per-sample
 323     # lists (response_lengths, loss_masks, log_probs, ...) need the DP-local
 324     # view instead, otherwise sample i's reward is matched against another
 325     # sample's data.
 326     if "raw_reward" in rollout_data:
 327         rollout_data["local_raw_reward"] = [rollout_data["raw_reward"][i] for i in partition]
 328 
 329     return rollout_data

 332 def get_source(sample: Sample) -> str:
 333     metadata = getattr(sample, "metadata", None) or {}
 334     if getattr(sample, "source", None):
 335         return sample.source
 336     if metadata.get("source_name"):
 337         return metadata["source_name"]
 338     return "unknown"
