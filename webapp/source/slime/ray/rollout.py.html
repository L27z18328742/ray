  95     def _try_ci_fault_injection(self):
  96         """Try to inject fault during generate (when health monitor is running)."""
  97         if not self._ci_fault_injection_pending:
  98             return
  99 
 100         # Only inject fault once
 101         self._ci_fault_injection_pending = False
 102 
 103         if (
 104             self.server
 105             and self.server.server_groups
 106             and self.server.server_groups[0].all_engines
 107             and self.server.server_groups[0].all_engines[0]
 108         ):
 109             logger.info("CI Fault Injection: Simulating crash on engine 0 during generate")
 110             try:
 111                 # This will cause the ray actor to exit
 112                 self.server.server_groups[0].all_engines[0].simulate_crash.remote()
 113                 # Wait for health monitor to detect the crash and mark engine as None
 114                 # health_check_interval + health_check_timeout + buffer
 115                 wait_time = self.args.rollout_health_check_interval + self.args.rollout_health_check_timeout + 5
 116                 logger.info(f"CI Fault Injection: Waiting {wait_time}s for health monitor to detect crash")
 117                 time.sleep(wait_time)
 118             except Exception as e:
 119                 logger.warning(f"CI Fault Injection failed: {e}")

 176     def generate(self, rollout_id):
 177         start_time = time.time()
 178         self.rollout_id = rollout_id
 179         set_current_rollout_id(rollout_id)
 180         self.health_monitoring_resume()
 181         if self.args.ci_test and self.args.use_fault_tolerance and rollout_id >= 2:
 182             self._try_ci_fault_injection()
 183         data, metrics = self._get_rollout_data(rollout_id=rollout_id)
 184         disk_routes = [
 185             sample.rollout_routed_experts
 186             for sample in data
 187             if isinstance(sample.rollout_routed_experts, DiskTensorRef)
 188         ]
 189         if disk_routes:
 190             self._active_routed_experts_rollouts.add(rollout_id)
 191             rss_gib, hwm_gib = get_process_host_memory_gib()
 192             logger.info(
 193                 "R3 manager spill profile: rollout_id=%d samples=%d files=%d disk_bytes=%.3f GiB "
 194                 "rss=%.3f GiB hwm=%.3f GiB host_available=%.3f GiB",
 195                 rollout_id,
 196                 len(data),
 197                 len(disk_routes),
 198                 sum(ref.nbytes for ref in disk_routes) / 1024**3,
 199                 rss_gib,
 200                 hwm_gib,
 201                 psutil.virtual_memory().available / 1024**3,
 202             )
 203         save_debug_rollout_data(
 204             self.args.save_debug_rollout_data,
 205             data,
 206             rollout_id=rollout_id,
 207             evaluation=False,
 208         )
 209         log_rollout_data(rollout_id, self.args, data, metrics, time.time() - start_time)
 210         if self.args.debug_rollout_only:
 211             # if debug rollout only, we don't convert samples to train data and directly return
 212             return
 213         data = self._convert_samples_to_train_data(data)
 214         return self._split_train_data_by_dp(data)

 286     def _get_rollout_data(self, rollout_id):
 287         if self.args.load_debug_rollout_data:
 288             data = load_debug_rollout_data(
 289                 self.args.load_debug_rollout_data,
 290                 rollout_id=rollout_id,
 291                 subsample_ratio=self.args.load_debug_rollout_data_subsample,
 292             )
 293             metrics = None
 294         else:
 295             if fully_async_metrics_enabled(self.args):
 296                 # The training loop keeps serving weights fixed while collecting
 297                 # this batch. Query only the updatable model.
 298                 server = self._get_updatable_server()
 299                 engines = [engine for engine in server.engines if engine is not None] if server else []
 300                 versions = ray.get([engine.get_weight_version.remote() for engine in engines])
 301                 valid = bool(versions) and all(
 302                     str(version).isascii() and str(version).isdigit() for version in versions
 303                 )
 304                 self.args._rollout_weight_version = max(map(int, versions)) if valid else None
 305             data = call_rollout_fn(self.generate_rollout, self.args, rollout_id, self.data_source, evaluation=False)
 306             metrics = data.metrics
 307             data = data.samples
 308             # Enforce the rollout_id contract before flattening: any list[Sample]
 309             # encountered in the nested output must have rollout_id set on every
 310             # element. Default rollouts inherit it from the data source; compact /
 311             # subagent paths that split one rollout into N training samples must
 312             # set the same rollout_id on every sibling so the loss reducer counts
 313             # the rollout once instead of N times.
 314             validate_rollout_id_annotated(data)
 315             # flatten the data if it is a list of lists
 316             while isinstance(data[0], list):
 317                 data = list(itertools.chain.from_iterable(data))
 318 
 319         return data, metrics

 321     def _post_process_rewards(self, samples: list[Sample] | list[list[Sample]]):
 322         if self.custom_reward_post_process_func is not None:
 323             return self.custom_reward_post_process_func(self.args, samples)
 324 
 325         raw_rewards = [sample.get_reward_value(self.args) for sample in samples]
 326         if (
 327             self.args.advantage_estimator in ["grpo", "gspo", "cispo", "reinforce_plus_plus_baseline"]
 328             and self.args.rewards_normalization
 329         ):
 330             # group norm
 331             rewards = torch.tensor(raw_rewards, dtype=torch.float)
 332             if rewards.shape[-1] == self.args.n_samples_per_prompt * self.args.rollout_batch_size:
 333                 rewards = rewards.reshape(-1, self.args.n_samples_per_prompt)
 334             else:
 335                 # when samples count are not equal in each group
 336                 rewards = rewards.view(-1, rewards.shape[-1])
 337             mean = rewards.mean(dim=-1, keepdim=True)
 338             rewards = rewards - mean
 339 
 340             if self.args.advantage_estimator in ["grpo", "gspo", "cispo"] and self.args.grpo_std_normalization:
 341                 std = rewards.std(dim=-1, keepdim=True)
 342                 rewards = rewards / (std + 1e-6)
 343 
 344             return raw_rewards, rewards.flatten().tolist()
 345 
 346         return raw_rewards, raw_rewards

 348     def _convert_samples_to_train_data(self, samples: list[Sample] | list[list[Sample]]):
 349         """
 350         Convert inference generated samples to training data.
 351         """
 352         if self.custom_convert_samples_to_train_data_func is not None:
 353             return self.custom_convert_samples_to_train_data_func(self.args, samples)
 354 
 355         raw_rewards, rewards = self._post_process_rewards(samples)
 356 
 357         assert len(raw_rewards) == len(samples)
 358         assert len(rewards) == len(samples)
 359 
 360         rollout_ids = [sample.rollout_id for sample in samples]
 361         existed_rollout_id_values = set(rid for rid in rollout_ids if rid is not None)
 362         tmp_id = 0
 363         for i in range(len(rollout_ids)):
 364             if rollout_ids[i] is None:
 365                 while tmp_id in existed_rollout_id_values:
 366                     tmp_id += 1
 367                 rollout_ids[i] = tmp_id
 368                 existed_rollout_id_values.add(tmp_id)
 369 
 370         train_data = {
 371             "tokens": [sample.tokens for sample in samples],
 372             "response_lengths": [sample.response_length for sample in samples],
 373             # some reward model, e.g. remote rm, may return multiple rewards,
 374             # we could use key to select the reward.
 375             "rewards": rewards,
 376             "raw_reward": raw_rewards,
 377             "truncated": [1 if sample.status == Sample.Status.TRUNCATED else 0 for sample in samples],
 378             "sample_indices": [sample.index for sample in samples],
 379             "rollout_ids": rollout_ids,
 380         }
 381 
 382         # loss mask
 383         # TODO: compress the loss mask
 384         loss_masks = []
 385         for sample in samples:
 386             # always instantiate loss_mask if not provided
 387             if sample.loss_mask is None:
 388                 sample.loss_mask = [1] * sample.response_length
 389 
 390             assert (
 391                 len(sample.loss_mask) == sample.response_length
 392             ), f"loss mask length {len(sample.loss_mask)} != response length {sample.response_length}"
 393             if sample.remove_sample:
 394                 sample.loss_mask = [0] * sample.response_length
 395             loss_masks.append(sample.loss_mask)
 396         train_data["loss_masks"] = loss_masks
 397 
 398         # Per-rollout aggregate, precomputed at the step level (where we can
 399         # see every sample of every rollout) and broadcast per-sample so the
 400         # per-mb loss reducer uses the correct whole-rollout denominator even
 401         # when a rollout's samples land in different micro-batches (first-fit
 402         # packing can split a rollout across mbs):
 403         #
 404         #   ``rollout_mask_sums[i]`` — sum of loss-mask totals over every
 405         #   sample in sample i's rollout. Used as the reducer's denominator
 406         #   so summing partial contributions across mbs yields one
 407         #   token-weighted mean per rollout.
 408         rollout_id_list = train_data["rollout_ids"]
 409         mask_sums_per_sample = [sum(m) for m in loss_masks]
 410         rollout_total_mask: dict[int, int] = {}
 411         for rid, ms in zip(rollout_id_list, mask_sums_per_sample, strict=True):
 412             rollout_total_mask[rid] = rollout_total_mask.get(rid, 0) + ms
 413         train_data["rollout_mask_sums"] = [rollout_total_mask[rid] for rid in rollout_id_list]
 414 
 415         # Overwrite raw_reward when available. Mixed-source batches may only
 416         # populate this field for a subset of samples (e.g. SWE but not code).
 417         if any(sample.metadata and "raw_reward" in sample.metadata for sample in samples):
 418             train_data["raw_reward"] = [
 419                 sample.metadata["raw_reward"] if sample.metadata and "raw_reward" in sample.metadata else sample.reward
 420                 for sample in samples
 421             ]
 422 
 423         # For rollout buffer
 424         if samples[0].metadata and "round_number" in samples[0].metadata:
 425             train_data["round_number"] = [sample.metadata["round_number"] for sample in samples]
 426 
 427         # Add rollout log probabilities for off-policy correction
 428         if samples[0].rollout_log_probs is not None:
 429             train_data["rollout_log_probs"] = [sample.rollout_log_probs for sample in samples]
 430 
 431         if getattr(self.args, "rollout_top_p", 1.0) != 1.0:
 432             for sample in samples:
 433                 assert sample.rollout_top_p_token_ids is not None
 434                 assert sample.rollout_top_p_token_offsets is not None
 435                 assert len(sample.rollout_top_p_token_offsets) == sample.response_length + 1, (
 436                     f"top-p token offsets length {len(sample.rollout_top_p_token_offsets)} "
 437                     f"!= response length + 1 {sample.response_length + 1}"
 438                 )
 439                 offset_end = int(sample.rollout_top_p_token_offsets[-1])
 440                 assert offset_end == len(sample.rollout_top_p_token_ids), (
 441                     f"top-p token offsets[-1] {offset_end} "
 442                     f"!= token ids length {len(sample.rollout_top_p_token_ids)}"
 443                 )
 444             train_data["rollout_top_p_token_ids"] = [sample.rollout_top_p_token_ids for sample in samples]
 445             train_data["rollout_top_p_token_offsets"] = [sample.rollout_top_p_token_offsets for sample in samples]
 446 
 447         routed_experts_present = [sample.rollout_routed_experts is not None for sample in samples]
 448         dead_sample_indices = {
 449             idx
 450             for idx, (present, sample) in enumerate(zip(routed_experts_present, samples, strict=True))
 451             if not present and sample.loss_mask is not None and not any(sample.loss_mask)
 452         }
 453         live_missing = [
 454             idx for idx, present in enumerate(routed_experts_present) if not present and idx not in dead_sample_indices
 455         ]
 456         routing_replay_enabled = getattr(self.args, "use_rollout_routing_replay", False)
 457         if live_missing and (routing_replay_enabled or any(routed_experts_present)):
 458             raise ValueError(f"Rollout routed experts are missing for live sample indices {live_missing[:32]}.")
 459 
 460         if any(routed_experts_present) or (routing_replay_enabled and dead_sample_indices):
 461             dtype = torch.uint8 if self.args.num_experts <= 256 else torch.int32
 462             routed_experts = [
 463                 (
 464                     (
 465                         sample.rollout_routed_experts
 466                         if isinstance(sample.rollout_routed_experts, DiskTensorRef)
 467                         else sample.materialize_rollout_routed_experts()
 468                     )
 469                     if present
 470                     else torch.zeros(
 471                         (max(0, len(sample.tokens) - 1), self.args.num_layers, self.args.moe_router_topk),
 472                         dtype=dtype,
 473                     )
 474                 )
 475                 for sample, present in zip(samples, routed_experts_present, strict=True)
 476             ]
 477             if routing_replay_enabled and any(routed_experts_present):
 478                 captured_pairs = [
 479                     (experts, max(0, len(sample.tokens) - 1))
 480                     for sample, experts, present in zip(samples, routed_experts, routed_experts_present, strict=True)
 481                     if present
 482                 ]
 483                 validate_rollout_routed_experts_for_replay(
 484                     [experts for experts, _ in captured_pairs],
 485                     self.args,
 486                     expected_rows=[rows for _, rows in captured_pairs],
 487                 )
 488             if dead_sample_indices:
 489                 logger.warning(
 490                     "Using all-zero routed experts for %d fully loss-masked samples (indices %s).",
 491                     len(dead_sample_indices),
 492                     sorted(dead_sample_indices)[:16],
 493                 )
 494             train_data["rollout_routed_experts"] = routed_experts
 495 
 496         if samples[0].train_metadata is not None:
 497             train_data["metadata"] = [sample.train_metadata for sample in samples]
 498 
 499         if any(sample.multimodal_train_inputs is not None for sample in samples):
 500             train_data["multimodal_train_inputs"] = [sample.multimodal_train_inputs for sample in samples]
 501 
 502         if samples[0].teacher_log_probs is not None:
 503             train_data["teacher_log_probs"] = [sample.teacher_log_probs for sample in samples]
 504 
 505         if samples[0].metadata is not None:
 506             train_data["source_names"] = [get_source(sample) for sample in samples]
 507 
 508         return train_data

 513     def _split_train_data_by_dp(self, data):
 514         """Compute the DP/mbs schedule and package each rank's rollout_data
 515         into a Ray Box. The schedule itself is computed by
 516         :func:`build_dp_schedule` so it stays unit-testable without Ray/sglang.
 517 
 518         Step split is by rollout id (``samples[i].rollout_id``, falling back
 519         to ``samples[i].index``); each step holds exactly
 520         ``args.global_batch_size`` rollouts so the training-step count per
 521         rollout is fixed at ``rollout_batch_size * n_samples_per_prompt //
 522         global_batch_size`` regardless of how many training samples each
 523         rollout produced.
 524         """
 525         dp_size = self.train_parallel_config["dp_size"]
 526         total_lengths = [len(t) for t in data["tokens"]]
 527         data["total_lengths"] = total_lengths
 528 
 529         partitions, micro_batch_indices, num_microbatches, global_batch_sizes = build_dp_schedule(
 530             self.args,
 531             self.train_parallel_config,
 532             total_lengths,
 533             global_batch_size=self.args.global_batch_size,
 534             rollout_indices=data["rollout_ids"],
 535         )
 536 
 537         # Package per-rank rollout_data
 538         rollout_data_refs = []
 539         for r in range(dp_size):
 540             partition = partitions[r]
 541             rollout_data = {"partition": partition}
 542             for key in [
 543                 "tokens",
 544                 "multimodal_train_inputs",
 545                 "response_lengths",
 546                 "rewards",
 547                 "truncated",
 548                 "loss_masks",
 549                 "round_number",
 550                 "sample_indices",
 551                 "rollout_ids",
 552                 "rollout_mask_sums",
 553                 "rollout_log_probs",
 554                 "rollout_top_p_token_ids",
 555                 "rollout_top_p_token_offsets",
 556                 "rollout_routed_experts",
 557                 "source_names",
 558                 "prompt",
 559                 "teacher_log_probs",
 560             ]:
 561                 if key not in data:
 562                     continue
 563                 rollout_data[key] = [data[key][j] for j in partition]
 564             # keys that need to be splited at train side
 565             for key in ["raw_reward", "total_lengths"]:
 566                 if key not in data:
 567                     continue
 568                 rollout_data[key] = data[key]
 569             rollout_data["global_batch_sizes"] = global_batch_sizes
 570             rollout_data["num_microbatches"] = num_microbatches
 571             rollout_data["micro_batch_indices"] = micro_batch_indices[r]
 572             tensorize_rollout_data_for_training(rollout_data)
 573             transport = getattr(self.args, "rollout_data_transport", "object-store")
 574             if transport == "nixl":
 575                 rollout_data_refs.append(Box(ray.put(rollout_data, _tensor_transport="nixl")))
 576             elif transport == "object-store":
 577                 rollout_data_refs.append(Box(ray.put(rollout_data)))
 578             else:
 579                 raise ValueError(f"Unsupported rollout data transport: {transport!r}")
 580         return rollout_data_refs
