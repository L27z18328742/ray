  57     def _allocate_gpus_for_actor(self, pg, num_gpus_per_actor):
  58         world_size = self._num_nodes * self._num_gpus_per_node
  59 
  60         # Use placement group to lock resources for models of same type
  61         assert pg is not None
  62         pg, reordered_bundle_indices, _reordered_gpu_ids = pg
  63 
  64         env_vars = {
  65             # because sglang will always set NCCL_CUMEM_ENABLE to 0
  66             # we need also set it to 0 to prevent nccl error.
  67             "NCCL_CUMEM_ENABLE": os.environ.get("NCCL_CUMEM_ENABLE", "0"),
  68             "NVTE_FP8_BLOCK_SCALING_FP32_SCALES": os.environ.get("NVTE_FP8_BLOCK_SCALING_FP32_SCALES", "1"),
  69             **{name: "1" for name in NOSET_VISIBLE_DEVICES_ENV_VARS_LIST},
  70             **self.args.train_env_vars,
  71         }
  72 
  73         if self.args.offload_train:
  74             import torch_memory_saver
  75 
  76             for path in [
  77                 "torch_memory_saver_hook_mode_preload_cu13.abi3.so",
  78                 "torch_memory_saver_hook_mode_preload_cu12.abi3.so",
  79                 "torch_memory_saver_hook_mode_preload.abi3.so",
  80             ]:
  81                 dynlib_path = os.path.join(
  82                     os.path.dirname(os.path.dirname(torch_memory_saver.__file__)),
  83                     path,
  84                 )
  85                 if os.path.exists(dynlib_path):
  86                     break
  87             else:
  88                 raise FileNotFoundError(
  89                     "Cannot find torch_memory_saver dynamic library. Please make sure torch_memory_saver is properly installed."
  90                 )
  91 
  92             env_vars["LD_PRELOAD"] = dynlib_path
  93             env_vars["TMS_INIT_ENABLE"] = "1"
  94             env_vars["TMS_INIT_ENABLE_CPU_BACKUP"] = "1"
  95 
  96         # We cannot do routing replay for critic.
  97         if self.args.use_routing_replay and self.role == "actor":
  98             env_vars["ENABLE_ROUTING_REPLAY"] = "1"
  99 
 100         if self._actor_cls is None:
 101             from slime.backends.megatron_utils.actor import MegatronTrainRayActor
 102 
 103             actor_impl = MegatronTrainRayActor
 104         else:
 105             actor_impl = self._actor_cls
 106 
 107         actor_options = {
 108             "num_gpus": 1,
 109             "runtime_env": {"env_vars": add_default_ray_env_vars(env_vars)},
 110         }
 111         if getattr(self.args, "rollout_data_transport", "object-store") == "nixl":
 112             actor_options["enable_tensor_transport"] = True
 113         TrainRayActor = ray.remote(**actor_options)(actor_impl)
 114 
 115         # Create worker actors
 116         self._actor_handlers = []
 117         master_addr, master_port = None, None
 118         for rank in range(world_size):
 119             actor = TrainRayActor.options(
 120                 num_cpus=num_gpus_per_actor,
 121                 num_gpus=num_gpus_per_actor,
 122                 scheduling_strategy=PlacementGroupSchedulingStrategy(
 123                     placement_group=pg,
 124                     placement_group_bundle_index=reordered_bundle_indices[rank],
 125                 ),
 126             ).remote(world_size, rank, master_addr, master_port)
 127             if rank == 0:
 128                 master_addr, master_port = ray.get(actor.get_master_addr_and_port.remote())
 129             self._actor_handlers.append(actor)

 131     def async_train(self, rollout_id, rollout_data_ref, external_data=None):
 132         """Do one rollout training. Returns a list of Ray refs (one per worker).
 133 
 134         For critics, each ref resolves to ``{"values": [cpu tensors...]}`` (or ``{}``
 135         for non-last-PP-stage workers). Actor refs resolve to ``None``.
 136 
 137         ``external_data`` may be a list (one item per worker) or a single dict
 138         broadcast to all workers.
 139         """
 140         if isinstance(external_data, list):
 141             assert len(external_data) == len(self._actor_handlers)
 142             return [
 143                 actor.train.remote(rollout_id, rollout_data_ref, external_data=ed)
 144                 for actor, ed in zip(self._actor_handlers, external_data, strict=False)
 145             ]
 146         return [
 147             actor.train.remote(rollout_id, rollout_data_ref, external_data=external_data)
 148             for actor in self._actor_handlers
 149         ]

 151     def save_model(self, rollout_id, force_sync=False):
 152         """Save actor model"""
 153         ret = ray.get([actor.save_model.remote(rollout_id, force_sync=force_sync) for actor in self._actor_handlers])
 154         if self._release_train_enabled():
 155             self.args.load = self.args.save
 156             self.args.ckpt_step = None
 157             self.args.finetune = False
 158             self.args.no_load_optim = self.args.no_save_optim
 159             self.args.no_load_rng = False
 160         return ret

 162     def update_weights(self):
 163         """Broadcast weights from rank 0 to all other ranks."""
 164         if not self._full_disk_weight_update_enabled():
 165             return ray.get([actor.update_weights.remote() for actor in self._actor_handlers])
 166 
 167         weight_version = self._disk_weight_version + 1
 168         disk_weight_dir = Path(self.args.update_weight_disk_dir) / f"weight_v{weight_version:06d}"
 169         ray.get([actor.update_weights.remote() for actor in self._actor_handlers])
 170         self._disk_weight_version = weight_version
 171         if self._release_train_enabled():
 172             self.release()
 173         self._reload_rollout_weights_from_disk(disk_weight_dir, str(weight_version))
 174 
 175     def onload(self):
 176         return ray.get([actor.wake_up.remote() for actor in self._actor_handlers])
 177 
 178     def offload(self):
 179         return ray.get([actor.sleep.remote() for actor in self._actor_handlers])
 180 
 181     def release(self):
 182         actors, self._actor_handlers = self._actor_handlers, []
 183         for actor in actors:
 184             ray.kill(actor, no_restart=True)
 185         if actors:
 186             time.sleep(5)

 188     def create(self, rollout_manager=None):
 189         if self._actor_handlers:
 190             return None
 191         if rollout_manager is not None:
 192             self._rollout_manager = rollout_manager
 193         self.args.update_weight_start_version = self._disk_weight_version
 194         self._allocate_gpus_for_actor(self._pg, self._num_gpus_per_actor)
 195         start_rollout_ids = ray.get(
 196             [
 197                 actor.init.remote(
 198                     self.args,
 199                     self.role,
 200                     with_ref=self._with_ref,
 201                     with_opd_teacher=self._with_opd_teacher,
 202                 )
 203                 for actor in self._actor_handlers
 204             ]
 205         )
 206         if self._rollout_manager is not None:
 207             self.set_rollout_manager(self._rollout_manager)
 208         return start_rollout_ids
 209

 227     def _reload_rollout_weights_from_disk(self, disk_weight_dir, weight_version):
 228         assert self._rollout_manager is not None, "disk weight update requires a rollout manager."
 229         if self.args.offload_rollout:
 230             ray.get(self._rollout_manager.onload_weights.remote())
 231         engines, *_ = ray.get(self._rollout_manager.get_updatable_engines_and_lock.remote())
 232         if not engines:
 233             if not self.args.update_weight_disk_keep_files:
 234                 shutil.rmtree(disk_weight_dir, ignore_errors=True)
 235             return
 236         if self.args.update_weight_local_checkpoint_dir:
 237             # each host pulls the published checkpoint onto local disk (e.g. NVMe) and
 238             # the engines reload from there; the pull is disk-only, so it runs before
 239             # pause and overlaps generation
 240             ray.get([engine.pull_weights.remote(int(weight_version)) for engine in engines])
 241             model_path = self.args.update_weight_local_checkpoint_dir
 242         else:
 243             model_path = str(disk_weight_dir)
 244         ray.get([engine.pause_generation.remote() for engine in engines])
 245         ray.get([engine.flush_cache.remote() for engine in engines])
 246         ray.get(
 247             [
 248                 engine.update_weights_from_disk.remote(
 249                     model_path=model_path,
 250                     weight_version=weight_version,
 251                 )
 252                 for engine in engines
 253             ]
 254         )
 255         if self.args.ci_test:
 256             engine_versions = ray.get([engine.get_weight_version.remote() for engine in engines])
 257             mismatches = [
 258                 f"engine {idx}: {engine_version}"
 259                 for idx, engine_version in enumerate(engine_versions)
 260                 if str(engine_version) != str(weight_version)
 261             ]
 262             if mismatches:
 263                 raise RuntimeError(
 264                     "Weight version mismatch after disk reload! "
 265                     f"Expected: {weight_version}; " + ", ".join(mismatches)
 266                 )
 267         if not self.args.update_weight_disk_keep_files:
 268             shutil.rmtree(disk_weight_dir, ignore_errors=True)
 269         ray.get([engine.continue_generation.remote() for engine in engines])
