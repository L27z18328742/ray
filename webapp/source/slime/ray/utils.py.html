  58 @ray.remote
  59 class Lock(RayActor):
  60     def __init__(self):
  61         self._locked = False  # False: unlocked, True: locked
  62 
  63     def acquire(self):
  64         """
  65         Try to acquire the lock. Returns True if acquired, False otherwise.
  66         Caller should retry until it returns True.
  67         """
  68         if not self._locked:
  69             self._locked = True
  70             return True
  71         return False
  72 
  73     def release(self):
  74         """Release the lock, allowing others to acquire."""
  75         assert self._locked, "Lock is not acquired, cannot release."
  76         self._locked = False
