  21 def sort_key(x):
  22     index, node_identifier, gpu_id = x
  23     # Sort by node IP number and then by GPU ID
  24     try:
  25         # try to parse it as an IP address.
  26         ip_address = node_identifier
  27         node_ip_parts = list(map(int, ip_address.split(".")))
  28     except ValueError:
  29         # Try to resolve the hostname to an IP address.
  30         try:
  31             ip_address = socket.gethostbyname(node_identifier)
  32             node_ip_parts = list(map(int, ip_address.split(".")))
  33         except (socket.gaierror, TypeError):
  34             # Instead, we convert each character of the original identifier string
  35             # to its ASCII value. This provides a stable and consistent numerical
  36             # representation that allows for sorting.
  37             node_ip_parts = [ord(c) for c in node_identifier]
  38 
  39     return (node_ip_parts, int(gpu_id))

  42 def _create_placement_group(num_gpus):
  43     """Create a placement group with the specified number of GPUs."""
  44     if num_gpus == 0:
  45         return None, [], []
  46 
  47     bundles = [{"GPU": 1, "CPU": 1} for _ in range(num_gpus)]
  48     pg = placement_group(bundles, strategy="PACK")
  49     num_bundles = len(bundles)
  50 
  51     # Wait for the placement group to be scheduled. Poll rather than a bare
  52     # ray.get(pg.ready()) so the wait is observable: when it can't be placed yet
  53     # (a node's GPUs haven't registered with the GCS, or an autoscaler is still
  54     # bringing nodes up) log the GPU counts periodically instead of hanging with no
  55     # output. The wait stays unbounded, so autoscaling clusters — where a pending
  56     # placement group is what drives scale-up — are unaffected.
  57     ready_ref = pg.ready()
  58     elapsed = 0
  59     log_interval = 30
  60     while not ray.wait([ready_ref], timeout=log_interval)[0]:
  61         elapsed += log_interval
  62         total = ray.cluster_resources().get("GPU", 0)
  63         available = ray.available_resources().get("GPU", 0)
  64         logger.info(
  65             f"Waiting for placement group of {num_gpus} GPUs (elapsed {elapsed}s): "
  66             f"{total:g} GPUs registered with Ray, {available:g} available."
  67         )
  68 
  69     # use info actor to get the GPU id
  70     info_actors = []
  71     for i in range(num_bundles):
  72         info_actors.append(
  73             InfoActor.options(
  74                 scheduling_strategy=PlacementGroupSchedulingStrategy(
  75                     placement_group=pg,
  76                     placement_group_bundle_index=i,
  77                 ),
  78             ).remote()
  79         )
  80     gpu_ids = ray.get([actor.get_ip_and_gpu_id.remote() for actor in info_actors])
  81     for actor in info_actors:
  82         ray.kill(actor)
  83 
  84     bundle_infos = [(i, gpu_ids[i][0], gpu_ids[i][1]) for i in range(num_bundles)]
  85     sorted_bundle_infos = sorted(bundle_infos, key=sort_key)
  86     pg_reordered_bundle_indices = [info[0] for info in sorted_bundle_infos]
  87     # Map from logical index -> physical GPU ID
  88     pg_reordered_gpu_ids = [gpu_ids[info[0]][1] for info in sorted_bundle_infos]
  89 
  90     for i in range(num_bundles):
  91         actual_bundle_index = pg_reordered_bundle_indices[i]
  92         logger.info(
  93             f"  bundle {i:4}, actual_bundle_index: {actual_bundle_index:4}, "
  94             f"node: {gpu_ids[actual_bundle_index][0]}, gpu: {gpu_ids[actual_bundle_index][1]}"
  95         )
  96 
  97     return pg, pg_reordered_bundle_indices, pg_reordered_gpu_ids

 100 def _get_placement_group_layout(args) -> tuple[int, int]:
 101     actor_num_gpus = args.actor_num_nodes * args.actor_num_gpus_per_node
 102 
 103     if args.debug_train_only:
 104         return actor_num_gpus, 0
 105 
 106     if args.rollout_external:
 107         if args.debug_rollout_only:
 108             return actor_num_gpus, 0
 109         return actor_num_gpus, actor_num_gpus
 110 
 111     if args.debug_rollout_only:
 112         return args.rollout_num_gpus, 0
 113 
 114     if args.colocate:
 115         return max(actor_num_gpus, args.rollout_num_gpus), 0
 116 
 117     return actor_num_gpus + args.rollout_num_gpus, actor_num_gpus

 120 def create_placement_groups(args):
 121     """Create placement groups for actor, critic, and rollout engines."""
 122 
 123     num_gpus, rollout_offset = _get_placement_group_layout(args)
 124 
 125     logger.info(f"Creating placement group with {num_gpus} GPUs...")
 126     pg, actor_pg_reordered_bundle_indices, actor_pg_reordered_gpu_ids = _create_placement_group(num_gpus)
 127     rollout_pg_reordered_bundle_indices = actor_pg_reordered_bundle_indices[rollout_offset:]
 128     rollout_pg_reordered_gpu_ids = actor_pg_reordered_gpu_ids[rollout_offset:]
 129 
 130     result = {
 131         "actor": (pg, actor_pg_reordered_bundle_indices, actor_pg_reordered_gpu_ids),
 132         "rollout": (pg, rollout_pg_reordered_bundle_indices, rollout_pg_reordered_gpu_ids),
 133     }
 134 
 135     result["critic"] = result["actor"] if args.use_critic else None
 136 
 137     return result

 163 def create_actor_model(args, pgs, rollout_manager, actor_cls=None):
 164     actor_args = args
 165     if args.megatron_config_path is not None:
 166         from slime.utils.arguments import parse_megatron_role_args
 167 
 168         actor_args = parse_megatron_role_args(args, args.megatron_config_path, role="actor")
 169 
 170     actor_model_kwargs = {}
 171     if actor_cls is not None:
 172         actor_model_kwargs["actor_cls"] = actor_cls
 173     actor_model = allocate_train_group(
 174         args=actor_args,
 175         num_nodes=args.actor_num_nodes,
 176         num_gpus_per_node=args.actor_num_gpus_per_node,
 177         pg=pgs["actor"],
 178         with_ref=actor_args.kl_coef != 0 or actor_args.use_kl_loss,
 179         with_opd_teacher=actor_args.use_opd and actor_args.opd_type == "megatron",

 182     actor_start_rollout_ids = actor_model.create(rollout_manager=rollout_manager)

 186 def create_training_models(args, pgs, rollout_manager, actor_cls=None):
 187     actor_model, actor_start_rollout_ids = create_actor_model(args, pgs, rollout_manager, actor_cls=actor_cls)
 188 
 189     critic_model = None
 190     if args.use_critic and args.num_rollout != 0:
 191         from slime.utils.arguments import parse_megatron_role_args
 192 
 193         critic_args = (
 194             parse_megatron_role_args(args, args.megatron_config_path, role="critic")
 195             if args.megatron_config_path is not None
 196             else copy.deepcopy(args)
 197         )
 198         if args.megatron_config_path is None:
 199             critic_args.disable_param_buffers_cpu_backup = False
 200 
 201         critic_model = allocate_train_group(
 202             args=critic_args,
 203             num_nodes=args.critic_num_nodes,
 204             num_gpus_per_node=args.critic_num_gpus_per_node,
 205             pg=pgs["critic"],
 206             role="critic",
 207         )
 208         critic_start_rollout_ids = critic_model.create(rollout_manager=rollout_manager)
 209 
 210     # TODO how to decide rollout start id when critic is involved? For now we just require user to specify it via args.
 211     if critic_model is not None:
 212         start_rollout_ids = critic_start_rollout_ids
 213     else:
 214         start_rollout_ids = actor_start_rollout_ids
 215 
 216     assert len(set(start_rollout_ids)) == 1
 217 
 218     if args.start_rollout_id is None:
 219         args.start_rollout_id = start_rollout_ids[0]
 220 
 221     if args.rollout_global_dataset:
 222         ray.get(rollout_manager.load.remote(args.start_rollout_id - 1))
 223 
 224     return actor_model, critic_model

 227 def create_rollout_manager(args, pg):
 228     from .rollout import RolloutManager
 229 
 230     rollout_manager_options = {
 231         "num_cpus": 1,
 232         "num_gpus": 0,
 233         "runtime_env": {"env_vars": add_default_ray_env_vars()},
 234     }
 235     if getattr(args, "rollout_data_transport", "object-store") == "nixl":
 236         rollout_manager_options["enable_tensor_transport"] = True
 237     rollout_manager = RolloutManager.options(**rollout_manager_options).remote(args, pg)
 238 
 239     # calculate num_rollout from num_epoch
 240     num_rollout_per_epoch = None
 241     if args.num_rollout is None:
 242         num_rollout_per_epoch = ray.get(rollout_manager.get_num_rollout_per_epoch.remote())
 243         args.num_rollout = num_rollout_per_epoch * args.num_epoch
 244         assert args.num_rollout > 0
 245 
 246     if args.check_weight_update_equal:
 247         ray.get(rollout_manager.check_weights.remote(action="snapshot"))
 248         ray.get(rollout_manager.check_weights.remote(action="reset_tensors"))
 249 
 250     if args.offload_rollout:
 251         ray.get(rollout_manager.offload.remote())
 252 
 253     return rollout_manager, num_rollout_per_epoch
