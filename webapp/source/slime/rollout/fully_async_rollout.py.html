slime/rollout/fully_async_rollout.py

   1 """Fully-async rollout for slime.
   2 
   3 Decouples ``max_concurrent_tasks`` from ``rollout_batch_size``: a background
   4 asyncio worker keeps a fixed pool of in-flight trajectories across rollout
   5 boundaries, so the next training step doesn't have to wait for the slowest
   6 in-flight sample to finish.
   7 
   8 Use with ``--rollout-function-path slime.rollout.fully_async_rollout.generate_rollout_fully_async``.
   9 Plug in per-sample logic via ``--custom-generate-function-path`` and
  10 per-sample reward via ``--custom-rm-path`` — the worker calls slime's stock
  11 :func:`generate_and_rm_group` which dispatches to those.
  12 
  13 Concurrency is sourced from ``args.sglang_server_concurrency`` and scaled by
  14 the number of sglang engines to match the per-sample semaphore cap in
  15 :mod:`slime.rollout.sglang_rollout`.
  16 
  17 The worker is intentionally oblivious to slime's higher-level pause /
  18 weight-update signalling (e.g. ``GenerateState.aborted``). Each in-flight
  19 generation short-circuits on those signals on its own and surfaces
  20 :data:`Sample.Status.ABORTED`; the only piece the worker owns is
  21 **redirecting ABORTED groups back to ``data_buffer``** instead of shipping
  22 them to training, so the next rollout (with refreshed weights) can pick
  23 them up.
  24 
  25 ``--dynamic-sampling-filter-path`` is honoured DAPO-style: rejected groups do
  26 not count toward ``rollout_batch_size`` and are replaced from the warm queue.
  27 """

  54 # Global worker, shared across rollout calls so the queue stays warm.
  55 _global_worker: AsyncRolloutWorker | None = None
  56 _worker_lock = threading.Lock()
  57 
  58 
  59 def _get_global_worker(args, data_buffer) -> AsyncRolloutWorker:
  60     global _global_worker
  61     with _worker_lock:
  62         if _global_worker is None or not _global_worker.worker_thread.is_alive():
  63             logger.info("starting fully-async rollout worker")
  64             _global_worker = AsyncRolloutWorker(
  65                 args, data_buffer, concurrency=args.sglang_server_concurrency * get_rollout_num_engines(args)
  66             )
  67             _global_worker.start()
  68         return _global_worker
  69 
  70 
  71 def _stop_global_worker() -> None:
  72     global _global_worker
  73     with _worker_lock:
  74         if _global_worker is not None:
  75             _global_worker.stop()
  76             _global_worker = None
  77 
  78 
  79 atexit.register(_stop_global_worker)

  82 class AsyncRolloutWorker:
  83     """Background thread + asyncio loop that continuously consumes groups
  84     from ``data_buffer`` and runs :func:`generate_and_rm_group` on each."""
  85 
  86     def __init__(self, args, data_buffer, concurrency: int = 10):
  87         self.args = args
  88         self.data_buffer = data_buffer
  89         self.concurrency = max(1, concurrency // getattr(args, "n_samples_per_prompt", 1))
  90         self.running = True
  91         # Unbounded on purpose: put() runs inside the event-loop thread (task
  92         # done-callback), so a bounded queue that fills up would block the loop
  93         # and freeze every in-flight generation. Backpressure lives in _loop()
  94         # instead, which stops topping up while a full pool of completed groups
  95         # is already waiting to be consumed.
  96         self.output_queue: queue.Queue[tuple[int, list[Sample]]] = queue.Queue()
  97         self.poll_interval = 1.0
  98         self.worker_thread: threading.Thread | None = None
  99         self.state = GenerateState(args)
 100 
 101     # -- public --------------------------------------------------------------
 102 
 103     def start(self) -> None:
 104         if self.worker_thread is None or not self.worker_thread.is_alive():
 105             self.worker_thread = threading.Thread(target=self._thread_main, name="fully-async-rollout", daemon=True)
 106             self.worker_thread.start()
 107 
 108     def stop(self) -> None:
 109         self.running = False
 110         if self.worker_thread and self.worker_thread.is_alive():
 111             self.worker_thread.join(timeout=5)
 112 
 113     def get_completed_groups(self, limit: int | None = None) -> list[tuple[int, list[Sample]]]:
 114         """Pop up to ``limit`` completed groups (all of them when ``None``).
 115 
 116         Callers that only need a fixed number of groups must pass ``limit`` —
 117         anything popped beyond it would otherwise have to be thrown away, and
 118         these groups are fully generated and reward-scored, with their prompts
 119         already consumed from ``data_buffer``.
 120         """
 121         completed: list[tuple[int, list[Sample]]] = []
 122         while limit is None or len(completed) < limit:
 123             try:
 124                 completed.append(self.output_queue.get_nowait())
 125             except queue.Empty:
 126                 break
 127         return completed
 128 
 129     def queue_size(self) -> int:
 130         return self.output_queue.qsize()
 131 
 132     # -- internals -----------------------------------------------------------
 133 
 134     def _thread_main(self) -> None:
 135         asyncio.run(self._loop())
 136 
 137     async def _loop(self) -> None:
 138         active_tasks: set[asyncio.Task] = set()
 139         max_concurrent = self.concurrency
 140         gid_counter = 0
 141 
 142         while self.running:
 143             try:
 144                 # Reap done tasks
 145                 if active_tasks:
 146                     done = {t for t in active_tasks if t.done()}
 147                     for t in done:
 148                         try:
 149                             t.result()  # results already handled in callback
 150                         except Exception as e:  # noqa: BLE001
 151                             logger.warning("fully-async task crashed: %r", e)
 152                     active_tasks -= done
 153                     if done:
 154                         # Done callbacks requeue ABORTED groups. Let them run
 155                         # before asking the data source for replacement work.
 156                         await asyncio.sleep(0)
 157 
 158                 # Top up. The qsize gate is the queue's backpressure: once a
 159                 # full pool of completed groups is waiting, stop pulling new
 160                 # prompts until the training side drains some.
 161                 while (
 162                     len(active_tasks) < max_concurrent and self.output_queue.qsize() < max_concurrent and self.running
 163                 ):
 164                     groups = self.data_buffer.get_samples(1)
 165                     if not groups:
 166                         break
 167                     for group in groups:
 168                         gid = gid_counter
 169                         gid_counter += 1
 170                         task = asyncio.create_task(
 171                             generate_and_rm_group(
 172                                 self.args,
 173                                 group,
 174                                 sampling_params=self.state.sampling_params.copy(),
 175                                 evaluation=False,
 176                             )
 177                         )
 178                         task.add_done_callback(self._make_done_cb(gid))
 179                         active_tasks.add(task)
 180 
 181                 await asyncio.sleep(self.poll_interval)
 182             except Exception as e:  # noqa: BLE001
 183                 logger.exception("fully-async loop iteration error: %s", e)
 184                 await asyncio.sleep(self.poll_interval)
 185 
 186         if active_tasks:
 187             logger.info(
 188                 "fully-async: waiting for %d in-flight tasks to drain",
 189                 len(active_tasks),
 190             )
 191             try:
 192                 await asyncio.wait(active_tasks, timeout=30)
 193             except Exception:  # noqa: BLE001
 194                 pass
 195 
 196     def _make_done_cb(self, gid: int):
 197         def _cb(done_task: asyncio.Task) -> None:
 198             try:
 199                 result = done_task.result()
 200             except Exception:  # noqa: BLE001
 201                 logger.exception("fully-async: process task raised")
 202                 return
 203             if not isinstance(result, list):
 204                 logger.warning(
 205                     "fully-async: generate_and_rm_group returned %r, expected list[Sample]; dropping",
 206                     type(result).__name__,
 207                 )
 208                 return
 209             # Aborted group → requeue, don't ship to training.
 210             if any(getattr(s, "status", None) == Sample.Status.ABORTED for s in result):
 211                 try:
 212                     self.data_buffer.add_samples([result])
 213                 except Exception:  # noqa: BLE001
 214                     logger.exception("fully-async: failed to requeue aborted group")
 215                 return
 216             self.output_queue.put((gid, result))
 217 
 218         return _cb

 221 async def _generate_rollout_async(args, rollout_id: int, data_buffer) -> RolloutFnTrainOutput | list[list[Sample]]:
 222     assert args.rollout_global_dataset
 223     filters_enabled = bool(
 224         getattr(args, "dynamic_sampling_filter_path", None) or getattr(args, "rollout_sample_filter_path", None)
 225     )
 226     worker = _get_global_worker(args, data_buffer)
 227 
 228     target = args.rollout_batch_size
 229     logger.info(
 230         "fully-async rollout %d: target=%d queue_warm=%d",
 231         rollout_id,
 232         target,
 233         worker.queue_size(),
 234     )
 235 
 236     collected: dict[int, list[Sample]] = {}
 237     dropped: list[list[Sample]] = []
 238     drop_reasons: dict[str, int] = {}
 239     dynamic_filter = (
 240         load_function(args.dynamic_sampling_filter_path)
 241         if getattr(args, "dynamic_sampling_filter_path", None) is not None
 242         else None
 243     )
 244     started = time.time()
 245     last_log = started
 246     LOG_EVERY = 30.0
 247 
 248     while len(collected) < target:
 249         # Pull only what this rollout still needs; the surplus stays queued for
 250         # the next rollout (that is the "queue stays warm" contract).
 251         drained = 0
 252         for gid, group in worker.get_completed_groups(limit=target - len(collected)):
 253             drained += 1
 254             if not filters_enabled:
 255                 collected[gid] = group
 256                 continue
 257 
 258             verdict = call_dynamic_filter(dynamic_filter, args, group)
 259             if verdict.keep:
 260                 collected[gid] = group
 261                 continue
 262 
 263             reason = verdict.reason or "dynamic_filter"
 264             for sample in group:
 265                 sample.remove_sample = True
 266                 if sample.metadata is not None:
 267                     sample.metadata["removed_reason"] = reason
 268             dropped.append(group)
 269             drop_reasons[reason] = drop_reasons.get(reason, 0) + 1
 270 
 271         if not drained:
 272             await asyncio.sleep(0.05)
 273 
 274         now = time.time()
 275         if now - last_log > LOG_EVERY:
 276             logger.info(
 277                 "fully-async rollout %d: collected %d/%d (dropped %d), queue=%d, elapsed=%.1fs",
 278                 rollout_id,
 279                 len(collected),
 280                 target,
 281                 len(dropped),
 282                 worker.queue_size(),
 283                 now - started,
 284             )
 285             last_log = now
 286 
 287     # Order by sample.index for determinism (slime convention).
 288     def _key(group: list[Sample]) -> int:
 289         for s in group:
 290             idx = getattr(s, "index", None)
 291             if idx is not None:
 292                 return int(idx)
 293         return 0
 294 
 295     out = sorted(collected.values(), key=_key)
 296     dropped = sorted(dropped, key=_key)
 297     if (filter_path := getattr(args, "rollout_sample_filter_path", None)) is not None:
 298         load_function(filter_path)(args, out)
 299     logger.info(
 300         "fully-async rollout %d: done in %.1fs, kept=%d dropped=%d (%s), queue_left=%d",
 301         rollout_id,
 302         time.time() - started,
 303         len(out),
 304         len(dropped),
 305         drop_reasons,
 306         worker.queue_size(),
 307     )
 308     if not filters_enabled:
 309         return out
 310     metrics = {f"rollout/dynamic_filter/drop_{reason}": count for reason, count in drop_reasons.items()}
 311     metrics["rollout/dynamic_filter/dropped_groups"] = len(dropped)
 312     metrics["rollout/dynamic_filter/dropped_ratio"] = len(dropped) / (len(out) + len(dropped))
 313     metrics["_dropped_samples"] = dropped
 314     return RolloutFnTrainOutput(samples=out, metrics=metrics)

 317 def generate_rollout_fully_async(args, rollout_id, data_buffer, evaluation: bool = False):
 318     """Slime ``--rollout-function-path`` entrypoint."""
 319 
 320     if evaluation:
 321         raise ValueError("fully-async rollout doesn't support evaluation mode")
 322     return run(_generate_rollout_async(args, rollout_id, data_buffer))
