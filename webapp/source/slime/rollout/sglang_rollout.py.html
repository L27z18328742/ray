  42 def _prepare_prompt_ids(sample: Sample, tokenizer, processor: Any) -> list[int]:
  43     raw_multimodal_inputs = sample.multimodal_inputs or {}
  44     has_multimodal_inputs = any(value is not None for value in raw_multimodal_inputs.values())
  45     reuse_existing_input_ids = bool(sample.tokens) and (
  46         sample.multimodal_train_inputs is not None or not has_multimodal_inputs
  47     )
  48 
  49     if processor and has_multimodal_inputs and not reuse_existing_input_ids:
  50         processor_output = processor(text=sample.prompt, **build_processor_kwargs(raw_multimodal_inputs))
  51         prompt_ids = processor_output["input_ids"][0]
  52         if sample.multimodal_train_inputs is None:
  53             sample.multimodal_train_inputs = {
  54                 k: v for k, v in processor_output.items() if k not in _PROCESSOR_PROMPT_KEYS
  55             } or None
  56         return prompt_ids
  57 
  58     if reuse_existing_input_ids:
  59         return sample.tokens
  60 
  61     return tokenizer.encode(sample.prompt, add_special_tokens=False)

  64 def get_model_url(args: Namespace, model_name: str, endpoint: str = "/generate") -> str:
  65     """Return the router URL for a named model.
  66 
  67     Use this in custom rollout functions to route requests to a specific
  68     model when multiple models are deployed via ``--sglang-config``::
  69 
  70         url = get_model_url(args, "ref", "/generate")
  71         resp = await post(url, json=payload)
  72 
  73     Falls back to the default router if *model_name* is not found or
  74     ``sglang_model_routers`` is not set.
  75     """
  76     routers = getattr(args, "sglang_model_routers", None)
  77     if routers and model_name in routers:
  78         ip, port = routers[model_name]
  79         return f"http://{ip}:{port}{endpoint}"
  80     return f"http://{args.sglang_router_ip}:{args.sglang_router_port}{endpoint}"

 119     @contextmanager
 120     def dp_rank_context(self):
 121         candidates = [i for i, count in enumerate(self.dp_counts) if count == min(self.dp_counts)]
 122         dp_rank = int(np.random.choice(candidates))
 123         self.dp_counts[dp_rank] += 1
 124         self.dp_rank = dp_rank
 125         try:
 126             yield dp_rank
 127         finally:
 128             self.dp_counts[dp_rank] -= 1
 129             assert self.dp_counts[dp_rank] >= 0

 138     def submit_generate_tasks(self, samples: list[list[Sample]]) -> None:
 139         for group in samples:

 154 async def generate(args: Namespace, sample: Sample, sampling_params: dict[str, Any]) -> Sample:
 155     """Generate using traditional SGLang router with token-based workflow"""
 156     if args.ci_test:
 157         assert isinstance(sample.prompt, str)
 158 
 159     state = GenerateState(args)
 160     url = f"http://{args.sglang_router_ip}:{args.sglang_router_port}/generate"
 161 
 162     assert (
 163         sample.status == Sample.Status.PENDING or sample.status == Sample.Status.ABORTED
 164     ), f"Sample status is {sample.status}"
 165 
 166     prompt_ids = _prepare_prompt_ids(sample, state.tokenizer, state.processor)
 167 
 168     sampling_params["max_new_tokens"] -= sample.response_length
 169 
 170     assert (
 171         sampling_params["max_new_tokens"] >= 0
 172     ), f"max_new_tokens: {sampling_params['max_new_tokens']} should not be less than 0"
 173     if sampling_params["max_new_tokens"] == 0:
 174         sample.status = Sample.Status.TRUNCATED
 175         return sample
 176 
 177     # Prepare payload for sglang server
 178     payload = {
 179         "sampling_params": sampling_params,
 180         "return_logprob": True,
 181     }
 182 
 183     if args.use_rollout_routing_replay:
 184         payload["return_routed_experts"] = True
 185 
 186     images = sample.multimodal_inputs.get("images") if sample.multimodal_inputs else None
 187     if images:
 188         payload["image_data"] = [encode_image_for_rollout_engine(image) for image in images]
 189         # For single-turn multimodal requests, send text so SGLang expands the
 190         # image placeholders with its own processor rules.
 191         payload["text"] = sample.prompt
 192     else:
 193         payload["input_ids"] = prompt_ids
 194 
 195     if not sample.tokens:
 196         sample.tokens = prompt_ids
 197 
 198     # Use session_id for consistent hashing routing (SGLang Model Gateway)
 199     headers = None
 200     if sample.session_id:
 201         if getattr(args, "router_policy", None) == "consistent_hashing":
 202             headers = {"X-SMG-Routing-Key": sample.session_id}
 203 
 204     with trace_span(sample, "sglang_generate", attrs={"max_new_tokens": sampling_params["max_new_tokens"]}) as span:
 205         output = await post(url, payload, headers=headers)
 206         span.update(build_sglang_meta_trace_attrs(output["meta_info"]))
 207 
 208     if "output_token_logprobs" in output["meta_info"]:
 209         new_response_tokens = [item[1] for item in output["meta_info"]["output_token_logprobs"]]
 210         new_response_log_probs = [item[0] for item in output["meta_info"]["output_token_logprobs"]]
 211     else:
 212         new_response_tokens, new_response_log_probs = [], []
 213 
 214     sample.append_response_tokens(
 215         args,
 216         tokens=new_response_tokens,
 217         log_probs=new_response_log_probs,
 218         trainable=True,
 219         meta_info=output["meta_info"],
 220         text=output["text"],
 221     )
 222 
 223     return sample

 226 async def _run_request_abortable_generate(
 227     state: GenerateState,
 228     sample: Sample,
 229     generate_call: Awaitable[Sample | list[Sample]],
 230 ) -> Sample | list[Sample]:
 231     task = asyncio.current_task()
 232     assert task is not None
 233     state.cancellable_tasks.add(task)
 234     try:
 235         return await generate_call
 236     except asyncio.CancelledError:
 237         if task in state.cancellable_tasks:
 238             raise
 239         sample.status = Sample.Status.ABORTED
 240         return sample
 241     finally:
 242         state.cancellable_tasks.discard(task)

 256 @trace_function("generate_and_rm", target="sample")
 257 async def generate_and_rm(
 258     args: Namespace,
 259     sample: Sample | list[Sample],
 260     sampling_params: dict[str, Any],
 261     evaluation: bool = False,
 262 ) -> Sample | list[Sample]:
 263     # mask previous off-policy generation for partial rollout
 264     if args.partial_rollout and args.mask_offpolicy_in_partial_rollout and sample.response_length > 0:
 265         sample.loss_mask = [0] * sample.response_length
 266 
 267     # For samples with existing response, check if they're complete
 268     if sample.status == Sample.Status.COMPLETED or sample.status == Sample.Status.TRUNCATED:
 269         assert sample.response is not None
 270         if not args.group_rm:
 271             assert sample.reward is not None
 272         return sample
 273 
 274     state = GenerateState(args)
 275 
 276     # generate
 277     async with state.semaphore:
 278         if state.aborted:
 279             sample.status = Sample.Status.ABORTED
 280             return sample
 281 
 282         with state.dp_rank_context() as _:
 283             custom_func_path = sample.generate_function_path or args.custom_generate_function_path
 284             generate_func = load_function(custom_func_path) if custom_func_path is not None else generate
 285 
 286             if custom_func_path is not None and "evaluation" in inspect.signature(generate_func).parameters:
 287                 generate_call = generate_func(args, sample, sampling_params, evaluation=evaluation)
 288             else:
 289                 generate_call = generate_func(args, sample, sampling_params)
 290 
 291             if getattr(generate_func, "abort_mode", None) == "request":
 292                 sample = await _run_request_abortable_generate(state, sample, generate_call)
 293             else:
 294                 sample = await _run_server_abort_generate(state, generate_call)
 295 
 296     sample = await apply_rollout_sample_hooks(args, sample, evaluation=evaluation)
 297 
 298     # for the rm that need the whole group, we will not do the rm here
 299     if args.group_rm:
 300         return sample
 301 
 302     if isinstance(sample, list):
 303         samples = sample
 304         if any(sample.status == Sample.Status.ABORTED for sample in samples):
 305             return samples
 306 
 307         samples_need_reward = [sample for sample in samples if sample.reward is None]
 308         with trace_span(samples_need_reward, "reward_model"):
 309             rewards = await batched_async_rm(args, samples_need_reward)
 310         for sample, reward in zip(samples_need_reward, rewards, strict=False):
 311             sample.reward = reward
 312         return samples
 313     else:
 314         if sample.status == Sample.Status.ABORTED:
 315             return sample
 316         # Some custom generate paths may have already filled the reward.
 317         if sample.reward is None:
 318             with trace_span(sample, "reward_model"):
 319                 sample.reward = await async_rm(args, sample)
 320 
 321     return sample

 324 @trace_function(
 325     "generate_and_rm_group",
 326     target="group",
 327     attrs_getter=lambda args, group, sampling_params, evaluation=False: {"group_size": len(group)},
 328 )
 329 async def generate_and_rm_group(
 330     args: Namespace, group: list[Sample], sampling_params: dict[str, Any], evaluation: bool = False
 331 ) -> list[Sample] | list[list[Sample]]:
 332     # ``generate_and_rm`` may return either a ``Sample`` or a ``list[Sample]``
 333     # depending on whether the ``--custom-generate-function-path`` callable
 334     # emits one trainable sample or several (e.g. multi-turn agent rollouts
 335     # that fan out into multiple prefix-chained samples). The asyncio.gather
 336     # below preserves whichever shape each task produced, so the group is
 337     # ``list[Sample]`` for plain rollouts and ``list[list[Sample]]`` for
 338     # the fan-out case.
 339     state = GenerateState(args)
 340 
 341     if state.aborted:
 342         return group
 343 
 344     # Generate a unique session_id for each sample in the group
 345     for sample in group:
 346         if sample.session_id is None:
 347             sample.session_id = str(uuid.uuid4())
 348 
 349     tasks = []
 350     for idx, sample in enumerate(group):
 351         current_sampling_params = sampling_params.copy()
 352         if getattr(args, "sglang_enable_deterministic_inference", False):
 353             seed = state.group_sampling_seeds[idx]
 354             current_sampling_params["sampling_seed"] = seed
 355         tasks.append(
 356             asyncio.create_task(generate_and_rm(args, sample, current_sampling_params, evaluation=evaluation))
 357         )
 358 
 359     group = await asyncio.gather(*tasks)
 360 
 361     # for the rm that need the whole group, we will do the rm here
 362     if not state.aborted and args.group_rm:
 363         with trace_span(group, "group_reward_model"):
 364             rewards = await batched_async_rm(args, group)
 365         for sample, reward in zip(group, rewards, strict=False):
 366             sample.reward = reward
 367 
 368     return group

 371 async def abort(args: Namespace, rollout_id: int) -> list[list[Sample]]:
 372     aborted_samples = []
 373 
 374     state = GenerateState(args)
 375     assert not state.aborted
 376     state.aborted = True
 377 
 378     cancellable_tasks = list(state.cancellable_tasks)
 379     state.cancellable_tasks.difference_update(cancellable_tasks)
 380     for task in cancellable_tasks:
 381         task.cancel()
 382 
 383     if state.active_server_generations:
 384         response = await get(f"http://{args.sglang_router_ip}:{args.sglang_router_port}/workers")
 385         urls = [worker["url"] for worker in response["workers"]]
 386         await abort_servers_until_idle(urls)
 387 
 388     await asyncio.gather(*cancellable_tasks, return_exceptions=True)
 389 
 390     # make sure all the pending tasks are finished
 391     count = 0
 392     while state.pendings:
 393         done, state.pendings = await asyncio.wait(state.pendings, return_when=asyncio.FIRST_COMPLETED)
 394 
 395         if not args.partial_rollout:
 396             continue
 397 
 398         # for partial rollout, collect the partial samples into the data buffer
 399         for task in done:
 400             group = task.result()
 401             if not any(sample.status == Sample.Status.ABORTED and sample.response_length > 0 for sample in group):
 402                 continue
 403             for sample in group:
 404                 if sample.response and "start_rollout_id" not in sample.metadata:
 405                     sample.metadata["start_rollout_id"] = rollout_id
 406             aborted_samples.append(group)
 407             count += len(group)
 408 
 409     if args.partial_rollout:
 410         logger.info(f"Collected {count} partial samples into the data buffer")
 411 
 412     return aborted_samples

 415 async def generate_rollout_async(
 416     args: Namespace, rollout_id: int, data_source: Callable[[int], list[list[Sample]]]
 417 ) -> tuple[RolloutFnTrainOutput, list[list[Sample]]]:
 418     """An example to implement the generate_rollout function for an rule based rm rollout generation.
 419 
 420     Args:
 421         args: the whole args
 422         rollout_id: int, the id of the rollout, used for deterministic data generation
 423         data_source: the data source to fetch
 424 
 425     Returns:
 426         tuple[RolloutFnTrainOutput, list[list[Sample]]]:
 427             - data: a list of groups of samples generated by the rollout, length equals `rollout_batch_size`
 428             - aborted_samples: any partial groups collected during abort when partial_rollout is enabled
 429     """
 430     assert args.rollout_global_dataset
 431 
 432     state = GenerateState(args)
 433 
 434     # instantiate data filters
 435     dynamic_filter = (
 436         load_function(args.dynamic_sampling_filter_path) if args.dynamic_sampling_filter_path is not None else None
 437     )
 438 
 439     metric_gatherer = MetricGatherer()
 440 
 441     # target_data_size is the total number of valid samples to get
 442     target_data_size = args.rollout_batch_size
 443 
 444     data = []
 445     all_data = []
 446     do_print = True
 447     pbar = tqdm(total=target_data_size * args.n_samples_per_prompt, desc="Rollout generation")
 448     while len(data) < target_data_size:
 449         while state.remaining_batch_size < target_data_size:
 450             # get samples from the buffer and submit the generation requests.
 451             samples = data_source(args.over_sampling_batch_size)
 452             state.submit_generate_tasks(samples)
 453 
 454         # wait for the generation to finish
 455         done, state.pendings = await asyncio.wait(state.pendings, return_when=asyncio.FIRST_COMPLETED)
 456         for task in done:
 457             group: list[Sample] = task.result()
 458 
 459             if do_print:
 460                 sample = group[0][0] if isinstance(group[0], list) else group[0]
 461                 logger.info(
 462                     f"First rollout sample: {[str(sample.prompt) + sample.response]}, label: {str(sample.label)[:100]}, reward: {sample.reward}",
 463                 )
 464                 do_print = False
 465 
 466             assert len(group) == args.n_samples_per_prompt
 467             all_data.append(group)
 468 
 469             dynamic_filter_output = call_dynamic_filter(dynamic_filter, args, group)
 470             if should_drop_dynamic_filter_output(
 471                 dynamic_filter_output,
 472                 remaining_batch_size=state.remaining_batch_size,
 473                 target_data_size=target_data_size,
 474             ):
 475                 metric_gatherer.on_dynamic_filter_drop(reason=dynamic_filter_output.reason)
 476                 state.remaining_batch_size -= 1
 477                 continue
 478 
 479             # add the samples to the data
 480             # NOTE: here we have not stored all the unused samples back to the data buffer.
 481             if len(data) < target_data_size:
 482                 data.append(group)
 483                 pbar.update(args.n_samples_per_prompt)
 484 
 485     pbar.close()
 486     sample = data[-1][0][0] if isinstance(data[-1][0], list) else data[-1][0]
 487     logger.info(
 488         f"Finish rollout: {[str(sample.prompt) + sample.response]}, label: {str(sample.label)[:100]}, reward: {sample.reward}",
 489     )
 490 
 491     # there are still some unfinished requests, abort them
 492     aborted_samples = await abort(args, rollout_id)
 493 
 494     assert len(data) == args.rollout_batch_size, f"Got {len(data)} samples, expected {args.rollout_batch_size}"
 495     data = sorted(data, key=lambda group: group[0][0].index if isinstance(group[0], list) else group[0].index)
 496     all_samples = sorted(
 497         all_data, key=lambda group: group[0][0].index if isinstance(group[0], list) else group[0].index
 498     )
 499 
 500     # reset the global state to prevent effects on the next rollout or eval.
 501     state.reset()
 502     if args.rollout_sample_filter_path is not None:
 503         filter_func = load_function(args.rollout_sample_filter_path)
 504         filter_func(args, data)
 505 
 506     # There can be circumstances where users want to process all samples including filtered ones.
 507     if args.rollout_all_samples_process_path is not None:
 508         process_func = load_function(args.rollout_all_samples_process_path)
 509         process_func(args, all_samples, data_source)
 510 
 511     return RolloutFnTrainOutput(samples=data, metrics=metric_gatherer.collect()), aborted_samples

 584     base_sampling_params = dict(
 585         temperature=dataset_cfg.temperature,
 586         top_p=dataset_cfg.top_p,
 587         top_k=dataset_cfg.top_k,
 588         max_new_tokens=dataset_cfg.max_response_len,
 589         stop=dataset_cfg.stop if dataset_cfg.stop is not None else args.rollout_stop,
 590         stop_token_ids=(
 591             dataset_cfg.stop_token_ids if dataset_cfg.stop_token_ids is not None else args.rollout_stop_token_ids
 592         ),
 593         skip_special_tokens=(
 594             dataset_cfg.skip_special_tokens
 595             if dataset_cfg.skip_special_tokens is not None
 596             else args.rollout_skip_special_tokens
 597         ),
 598         no_stop_trim=dataset_cfg.no_stop_trim if dataset_cfg.no_stop_trim is not None else True,
 599         spaces_between_special_tokens=False,
 600     )
 601     if dataset_cfg.repetition_penalty is not None:
 602         base_sampling_params["repetition_penalty"] = dataset_cfg.repetition_penalty
 603     min_new_tokens = dataset_cfg.min_new_tokens
 604     if min_new_tokens is None:
 605         min_new_tokens = getattr(args, "eval_min_new_tokens", None)
 606     if min_new_tokens is not None:
 607         base_sampling_params["min_new_tokens"] = min_new_tokens
 608 
 609     tasks = []
 610     # do multiple samples for eval prompts
 611     sample_index = 0
 612     for _i, prompt_sample in enumerate(dataset.samples):
 613         for j in range(dataset_cfg.n_samples_per_eval_prompt):
 614             # use the same prompt for multiple samples
 615             sample = copy.deepcopy(prompt_sample)
 616             sample.index = sample_index
 617             sample_index += 1
 618             sample.metadata = dataset_cfg.inject_metadata(getattr(sample, "metadata", None))
 619             sample.custom_rm_path = dataset_cfg.custom_rm_path
 620             sample.generate_function_path = getattr(dataset_cfg, "custom_generate_function_path", None)
 621             sampling_params = base_sampling_params
 622             if getattr(args, "sglang_enable_deterministic_inference", False):
 623                 sampling_params = base_sampling_params.copy()
 624                 sampling_params["sampling_seed"] = args.rollout_seed + j
 625             tasks.append(
 626                 asyncio.create_task(
 627                     generate_and_rm(
 628                         args,
 629                         sample,
 630                         sampling_params=sampling_params,
 631                         evaluation=True,
 632                     )
 633                 )
 634             )
 635 
 636     data = []
 637     do_print = True
 638     pbar = tqdm(total=len(tasks), desc=f"Eval {dataset_cfg.name}", disable=not do_print)
 639     for coro in asyncio.as_completed(tasks):
 640         sample = await coro
 641         if do_print:
 642             logged_sample = sample[0] if isinstance(sample, list) else sample
 643             logger.info(
 644                 "eval_rollout_single_dataset example data: "
 645                 f"{[str(logged_sample.prompt) + logged_sample.response]} "
 646                 f"reward={logged_sample.reward}"
 647             )
 648             do_print = False
 649         if isinstance(sample, list):
 650             data.extend(sample)
 651         else:
 652             data.append(sample)
 653         pbar.update(1)
 654     pbar.close()
 655 
 656     data.sort(key=lambda sample: sample.index)
 657 
 658     reward_key = args.eval_reward_key or args.reward_key
 659     return {
 660         dataset_cfg.name: {
 661             "rewards": [sample.reward if not reward_key else sample.reward[reward_key] for sample in data],
 662             "truncated": [sample.status == Sample.Status.TRUNCATED for sample in data],
 663             "samples": data,
 664         }
 665     }
