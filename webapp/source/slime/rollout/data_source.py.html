 138     def load(self, rollout_id=None):
 139         if not self.args.rollout_global_dataset:

 168 class RolloutDataSourceWithBuffer(RolloutDataSource):
 169     def __init__(self, args):
 170         super().__init__(args)
 171         self.buffer = []
 172         if self.args.buffer_filter_path is None:
 173             self.buffer_filter = pop_oldest if getattr(args, "buffer_sort_by_staleness", False) else pop_first
 174         else:
 175             self.buffer_filter = load_function(self.args.buffer_filter_path)
 176 
 177     def get_samples(self, num_samples: int) -> list[list[Sample]]:
 178         """
 179         Return num_samples samples
 180         """
 181 
 182         samples = self._get_samples_from_buffer(num_samples)
 183         num_samples -= len(samples)
 184 
 185         if num_samples == 0:
 186             return samples
 187 
 188         samples += super().get_samples(num_samples=num_samples)
 189         return samples
 190 
 191     def _get_samples_from_buffer(self, num_samples: int) -> list[list[Sample]]:
 192         if len(self.buffer) == 0 or num_samples == 0:
 193             return []
 194 
 195         samples = self.buffer_filter(self.args, None, self.buffer, num_samples)
 196         return samples
 197 
 198     def add_samples(self, samples: list[list[Sample]]):
 199         """
 200         Add a sample group to buffer.
 201         """
 202         if not samples:
 203             return
 204         assert isinstance(samples, list), f"samples must be a list, got {type(samples)}"
 205         assert isinstance(samples[0], list), f"the elements of samples must be list, got {type(samples[0])}"
 206         for i in range(0, len(samples)):
 207             assert (
 208                 len(samples[i]) == self.args.n_samples_per_prompt
 209             ), f"the length of the elements of samples must be equal to n_samples_per_prompt, got {len(samples[i])} != {self.args.n_samples_per_prompt}"
 210             group = samples[i]  # type: ignore
 211             self.buffer.append(group)

 225 def pop_oldest(args, rollout_id, buffer: list[list[Sample]], num_samples: int) -> list[list[Sample]]:
 226     """Resume groups with the oldest generated tokens first; preserve ties."""
 227 
 228     def oldest_version(group):
 229         versions = (str(v) for sample in group for v in (sample.weight_versions or []))
 230         return min(
 231             (int(v) for v in versions if v.isascii() and v.removeprefix("-").isdigit()),
 232             default=float("inf"),
 233         )
 234 
 235     buffer.sort(key=oldest_version)
 236     return pop_first(args, rollout_id, buffer, num_samples)

 239 def pop_first(args, rollout_id, buffer: list[list[Sample]], num_samples: int) -> list[list[Sample]]:
 240     num_to_pop = min(len(buffer), num_samples)
 241     samples = buffer[:num_to_pop]
 242     del buffer[:num_to_pop]
 243     return samples
